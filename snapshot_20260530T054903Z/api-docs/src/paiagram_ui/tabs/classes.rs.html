3use egui::{Button, Layout, Panel, ScrollArea, Ui, WidgetText, vec2};

6use serde::{Deserialize, Serialize};

8use super::Tab;
9use crate::OpenOrFocus;
10use crate::tabs::trip::TripTab;

12#[derive(Default, PartialEq, Clone, Serialize, Deserialize, MapEntities)]
13pub(crate) struct ClassesTab {
14    #[serde(skip)]
15    selected_class: Option<Entity>,
16    #[serde(skip)]
17    hovered_trip: Option<Entity>,
18}
19
20impl Tab for ClassesTab {
21    const NAME: &'static str = "Classes";
22    fn title(&self) -> WidgetText {
23        tr!("tab-classes").into()
24    }
25    fn main_display(&mut self, world: &mut bevy::ecs::world::World, ui: &mut Ui) {
26        world
27            .run_system_cached_with(list_classes, (ui, self))
28            .unwrap();
29    }
30}
31
32fn list_classes(
33    (InMut(ui), InMut(tab)): (InMut<Ui>, InMut<ClassesTab>),
34    mut class_q: Query<(Entity, &Class, &Name, &mut DisplayedStroke)>,
35    entity_name_q: Query<(Entity, &Name)>,
36    mut commands: Commands,
37) {
38    Panel::right(ui.id().with("first"))
39        .default_size(ui.available_width() / 3.0)
40        .resizable(true)
41        .show_inside(ui, |ui| {
42            ScrollArea::vertical().show(ui, |ui| {
43                ui.label("Trips");
44                let Some(class_entity) = tab.selected_class else {
45                    return;
46                };
47                let Ok((_, class, _, _)) = class_q.get(class_entity) else {
48                    return;
49                };
50                let mut hovered = false;
51                ui.with_layout(Layout::default().with_cross_justify(true), |ui| {
52                    for (trip_entity, name) in
53                        entity_name_q.iter_many(class.as_trips().iter().copied())
54                    {
55                        let res = ui.button(name.as_str());
56                        if res.hovered() {
57                            hovered = true;
58                            tab.hovered_trip = Some(trip_entity);
59                        }
60                        if res.clicked() {
61                            commands.write_message(OpenOrFocus(crate::MainTab::Trip(
62                                TripTab::new(trip_entity),
63                            )));
64                        }
65                    }
66                });
67                if !hovered {
68                    tab.hovered_trip = None;
69                }
70            });
71        });
72
73    let mut itoa_buffer = itoa::Buffer::new();
74    ScrollArea::vertical().auto_shrink(false).show(ui, |ui| {
75        egui::Grid::new("class grid")
76            .num_columns(3)
77            .striped(true)
78            .show(ui, |ui| {
79                ui.label(tr!("classes-name"));
80                ui.label(tr!("classes-count"));
81                ui.label(tr!("classes-color"));
82                ui.end_row();
83                for (class_entity, class, class_name, mut stroke) in class_q.iter_mut() {
84                    ui.allocate_ui_with_layout(
85                        vec2(200.0, 24.0),
86                        Layout::default().with_cross_justify(true),
87                        |ui| {
88                            ui.selectable_value(
89                                &mut tab.selected_class,
90                                Some(class_entity),
91                                class_name.as_str(),
92                            );
93                        },
94                    );
95                    let printed = itoa_buffer.format(class.as_trips().len());
96                    ui.label(printed);
97                    ui.add(&mut stroke.color);
98                    ui.end_row();
99                }
100            });
101    });
102
103    ScrollArea::vertical().id_salt("third").show(ui, |ui| {
104        let Some(trip_entity) = tab.hovered_trip else {
105            return;
106        };
107        ui.label(trip_entity.to_string());
108    });
109}
