9#[derive(FromMeta)]
10enum OuDiaType {
11    /// [key => Val]
12    /// return Val
13    SinglePairSingleEntry(String),
14    /// [key => Val1, Val2, Val3, ...]
15    /// return Vec<Val>
16    SinglePairManyEntries(String),
17    /// [Struct::key => Struct]
18    /// return Struct
19    SingleStruct(String),
20    /// [Struct::key => Struct, Struct::key => Struct, ...]
21    /// return Vec<Struct>
22    ManyStructs(String),
23    /// [key => [Struct::key => Struct, Struct::key => Struct, ...]]
24    /// return Vec<Struct>
25    SingleStructManyEntries(String),
26    /// [key1 => [Struct::key => Struct, ...], key2 => [Struct::key => Struct, ...]]
27    /// return Vec<Struct>
28    TwinStructMultipleEntries { first: String, second: String },
29}

31#[derive(FromMeta)]
32#[darling(derive_syn_parse)]
33struct StructArgs {
34    key: Option<String>,
35    alias: Option<String>,
36}

38/// A field-level `default` annotation, mirroring serde's `#[serde(default)]`.
39///
40/// `#[oudia(default)]` falls back to `Default::default()`, while
41/// `#[oudia(default = <expr>)]` uses the given expression.
42enum DefaultValue {
43    UseDefault,
44    Value(syn::Expr),
45}
46
47impl FromMeta for DefaultValue {
48    fn from_word() -> darling::Result<Self> {
49        Ok(DefaultValue::UseDefault)
50    }
51    fn from_expr(expr: &syn::Expr) -> darling::Result<Self> {
52        Ok(DefaultValue::Value(expr.clone()))
53    }
54}

56#[derive(FromField)]
57#[darling(attributes(oudia))]
58struct FieldOpts {
59    #[darling(rename = "type")]
60    kind: OuDiaType,
61    alias: Option<String>,
62    parse_fn: Option<syn::Expr>,
63    silence_fn: Option<syn::Expr>,
64    serialize_fn: Option<syn::Expr>,
65    default: Option<DefaultValue>,
66}

68enum OuterType<'a> {
69    Option(&'a Type),
70    Vec(&'a Type),
71    Plain(&'a Type),
72}
73
74impl<'a> OuterType<'a> {
75    fn into_inner(&self) -> &Type {
76        match *self {
77            Self::Option(ty) => ty,
78            Self::Vec(ty) => ty,
79            Self::Plain(ty) => ty,
80        }
81    }
82}
83
84fn inspect_type(ty: &Type) -> OuterType<'_> {
85    if let Type::Path(TypePath { qself: None, path, .. }) = ty
86        && let Some(segment) = path.segments.last()
87        && let PathArguments::AngleBracketed(args) = &segment.arguments
88        && let Some(GenericArgument::Type(inner_ty)) = args.args.first()
89    {
90        if segment.ident == "Option" {
91            return OuterType::Option(inner_ty);
92        } else if segment.ident == "Vec" {
93            return OuterType::Vec(inner_ty);
94        }
95    }
96
97    OuterType::Plain(ty)
98}
99
100fn is_bool_type(ty: &Type) -> bool {
101    if let Type::Path(TypePath { qself: None, path, .. }) = ty {
102        return path.is_ident("bool");
103    }
104    false
105}

107fn doc_aliases(names: &[String], skip: &str) -> Vec<syn::Attribute> {
108    let names: Vec<&String> = names.iter().filter(|name| name.as_str() != skip).collect();
109    if names.is_empty() {
110        return Vec::new();
111    }
112
113    let first = names[0];
114    let rest = &names[1..];
115    let doc_desc = quote! {
116        concat!{
117            "\n\nAlso known as `",
118            #first,
119            #( "`, `", #rest, )*
120            "`."
121        }
122    };
123
124    let mut attrs = vec![parse_quote!(#[doc = #doc_desc])];
125    for name in names {
126        attrs.push(parse_quote!(#[doc(alias = #name)]));
127    }
128    attrs
129}

131#[proc_macro_attribute]
132pub fn oudia(args: TokenStream, input: TokenStream) -> TokenStream {
133    let args = parse_macro_input!(args as StructArgs);
134    let mut input = parse_macro_input!(input as DeriveInput);
135
136    let struct_ident = input.ident.clone();
137    let (impl_generics, ty_generics, where_clause) = input.generics.split_for_impl();
138
139    let struct_oudia_key = args.key.clone().unwrap_or_else(|| struct_ident.to_string());
140
141    let mut struct_names = Vec::new();
142    if let Some(key) = &args.key {
143        if !key.is_empty() {
144            struct_names.push(key.clone());
145        }
146    }
147    if let Some(alias) = &args.alias {
148        struct_names.push(alias.clone());
149    }
150    input.attrs.extend(doc_aliases(&struct_names, &struct_ident.to_string()));
151
152    let mut initializers = proc_macro2::TokenStream::new();
153    let mut matchers = proc_macro2::TokenStream::new();
154    let mut silencers = proc_macro2::TokenStream::new();
155    let mut assembler = proc_macro2::TokenStream::new();
156    let mut fixups = proc_macro2::TokenStream::new();
157    let mut serializers = proc_macro2::TokenStream::new();
158
159    // Fields used to synthesize a `Default` impl. We only emit one when every
160    // field has a resolvable default.
161    let mut all_fields_default = true;
162    let mut default_fields = proc_macro2::TokenStream::new();
163
164    {
165        let Data::Struct(data_struct) = &mut input.data else {
166            return Error::new_spanned(&input.ident, "You should apply `oudia` on a struct")
167                .to_compile_error()
168                .into();
169        };
170
171        for field in &mut data_struct.fields {
172            let opts = match FieldOpts::from_field(&*field) {
173                Ok(opts) => opts,
174                Err(e) => {
175                    return Error::new_spanned(
176                        &*field,
177                        format!("failed to parse oudia attributes on field: {e}"),
178                    )
179                    .to_compile_error()
180                    .into();
181                }
182            };
183
184            let field_ident = field.ident.as_ref();
185            let outer = inspect_type(&field.ty);
186            let inner_ty = outer.into_inner();
187            let is_bool = is_bool_type(inner_ty);
188            let has_default = opts.default.is_some();
189
190            // The value used for both deserialization fallback and the `Default` impl.
191            let default_expr = match &opts.default {
192                Some(DefaultValue::UseDefault) => Some(quote! { Default::default() }),
193                Some(DefaultValue::Value(expr)) => Some(quote! { #expr }),
194                None => match &outer {
195                    OuterType::Option(_) => Some(quote! { None }),
196                    OuterType::Vec(_) => Some(quote! { Vec::new() }),
197                    OuterType::Plain(_) => None,
198                },
199            };
200
201            if let (Some(ident), Some(expr)) = (field_ident, &default_expr) {
202                default_fields.extend(quote! {
203                    #ident: #expr,
204                });
205            } else {
206                all_fields_default = false;
207            }
208
209            if let Some(parse_fn) = &opts.parse_fn {
210                initializers.extend(quote! {
211                    let #field_ident = (#parse_fn)(input)?;
212                });
213            } else {
214                // How to parse a single string value (a `&str`) into `inner_ty`.
215                let parse_value = |value: &syn::Expr| {
216                    if is_bool {
217                        quote! { (#value == "1") }
218                    } else {
219                        quote! { #value.parse::<#inner_ty>()? }
220                    }
221                };
222
223                // Uniform initializer: every field accumulates into `Option<T>`
224                // or `Vec<T>`; defaults are applied later, in the fixups.
225                match (&outer, &opts.kind) {
226                    (
227                        OuterType::Option(_) | OuterType::Plain(_),
228                        OuDiaType::SingleStruct(..) | OuDiaType::SinglePairSingleEntry(..),
229                    ) => {
230                        initializers.extend(quote! {
231                            let mut #field_ident: Option<#inner_ty> = None;
232                        });
233                    }
234                    (
235                        OuterType::Vec(_),
236                        OuDiaType::ManyStructs(..)
237                        | OuDiaType::SinglePairManyEntries(..)
238                        | OuDiaType::SingleStructManyEntries(..)
239                        | OuDiaType::TwinStructMultipleEntries { .. },
240                    ) => {
241                        initializers.extend(quote! {
242                            let mut #field_ident: Vec<#inner_ty> = Vec::new();
243                        });
244                    }
245                    _ => {
246                        return Error::new_spanned(
247                            &field.ty,
248                            "OuDia type don't match with field type.",
249                        )
250                        .to_compile_error()
251                        .into();
252                    }
253                }
254
255                matchers.extend(match &opts.kind {
256                    OuDiaType::SingleStruct(key) => quote! {
257                        crate::ast::Structure::Struct(k, v) if k == #key => {
258                            #field_ident = Some(<#inner_ty as crate::OuDiaIo>::from_structure(v)?);
259                        }
260                    },
261                    OuDiaType::SinglePairSingleEntry(key) => {
262                        let first = parse_quote!(first);
263                        let parse_expr = parse_value(&first);
264                        quote! {
265                            crate::ast::Structure::Pair(k, v) if k == #key && let Some(first) = v.first() => {
266                                #field_ident = Some(#parse_expr);
267                            }
268                        }
269                    }
270                    OuDiaType::ManyStructs(key) => quote! {
271                        crate::ast::Structure::Struct(k, v) if k == #key => {
272                            #field_ident.push(<#inner_ty as crate::OuDiaIo>::from_structure(v)?);
273                        }
274                    },
275                    OuDiaType::SinglePairManyEntries(key) => {
276                        let val = parse_quote!(val);
277                        let parse_expr = parse_value(&val);
278                        quote! {
279                            crate::ast::Structure::Pair(k, v) if k == #key => for val in v {
280                                #field_ident.push(#parse_expr);
281                            }
282                        }
283                    }
284                    OuDiaType::SingleStructManyEntries(key) => quote! {
285                        crate::ast::Structure::Struct(k, v) if k == #key => for node in v {
286                            match node {
287                                crate::ast::Structure::Struct(k2, v2) if k2 == <#inner_ty as crate::OuDiaIo>::OUDIA_KEY => {
288                                    #field_ident.push(<#inner_ty as crate::OuDiaIo>::from_structure(v2)?);
289                                }
290                                // Unknown field. Give error in log.
291                                crate::ast::Structure::Pair(k2, _) | crate::ast::Structure::Struct(k2, _) => {
292                                    log::debug!("Encountered unknown field `{k2}` when parsing `{}`", std::any::type_name::<Self>());
293                                }
294                            }
295                        }
296                    },
297                    OuDiaType::TwinStructMultipleEntries { first, second } => quote! {
298                        crate::ast::Structure::Struct(k, v) if k == #first || k == #second => for node in v {
299                            match node {
300                                crate::ast::Structure::Struct(k2, v2) if k2 == <#inner_ty as crate::OuDiaIo>::OUDIA_KEY => {
301                                    #field_ident.push(<#inner_ty as crate::OuDiaIo>::from_structure(v2)?);
302                                }
303                                // Unknown field. Give error in log.
304                                crate::ast::Structure::Pair(k2, _) | crate::ast::Structure::Struct(k2, _) => {
305                                    log::debug!("Encountered unknown field `{k2}` when parsing `{}`", std::any::type_name::<Self>());
306                                }
307                            }
308                        }
309                    },
310                });
311
312                // Fixups are the only place defaults / required-ness differ.
313                match (&outer, &opts.kind) {
314                    (
315                        OuterType::Plain(_),
316                        OuDiaType::SingleStruct(..) | OuDiaType::SinglePairSingleEntry(..),
317                    ) => {
318                        if has_default {
319                            let expr = default_expr.as_ref().unwrap();
320                            fixups.extend(quote! {
321                                let #field_ident = #field_ident.unwrap_or(#expr);
322                            });
323                        } else {
324                            let missing = match &opts.kind {
325                                OuDiaType::SingleStruct(key) => quote! { #key },
326                                OuDiaType::SinglePairSingleEntry(key) => quote! { #key },
327                                _ => unreachable!(),
328                            };
329                            fixups.extend(quote! {
330                                let Some(#field_ident) = #field_ident else {
331                                    return Err(crate::IrConversionError::MissingField {
332                                        processing: std::any::type_name::<Self>(),
333                                        missing: #missing,
334                                    });
335                                };
336                            });
337                        }
338                    }
339                    (OuterType::Vec(_), _) => {
340                        if has_default {
341                            let expr = default_expr.as_ref().unwrap();
342                            fixups.extend(quote! {
343                                if #field_ident.is_empty() {
344                                    #field_ident = #expr;
345                                }
346                            });
347                        }
348                    }
349                    _ => {}
350                }
351            }
352
353            if let Some(silence_fn) = &opts.silence_fn {
354                silencers.extend(quote! {
355                    crate::ast::Structure::Pair(k, _) | crate::ast::Structure::Struct(k, _) if (#silence_fn)(k)=> {}
356                })
357            }
358
359            if let Some(serialize_fn) = &opts.serialize_fn {
360                serializers.extend(quote! {
361                    (#serialize_fn)(&mut __oudia_items);
362                });
363            } else {
364                serializers.extend(match (&outer, &opts.kind) {
365                    (OuterType::Option(_), OuDiaType::SinglePairSingleEntry(key)) => {
366                        if is_bool {
367                            quote! {
368                                if let Some(value) = &self.#field_ident {
369                                    __oudia_items.push(crate::pair!(#key => if *value { "1" } else { "0" }));
370                                }
371                            }
372                        } else {
373                            quote! {
374                                if let Some(value) = &self.#field_ident {
375                                    __oudia_items.push(crate::pair!(#key => value.to_string()));
376                                }
377                            }
378                        }
379                    }
380                    (OuterType::Plain(_), OuDiaType::SinglePairSingleEntry(key)) => {
381                        if is_bool {
382                            quote! {
383                                __oudia_items.push(crate::pair!(#key => if self.#field_ident { "1" } else { "0" }));
384                            }
385                        } else {
386                            quote! {
387                                __oudia_items.push(crate::pair!(#key => self.#field_ident.to_string()));
388                            }
389                        }
390                    }
391                    (OuterType::Vec(_), OuDiaType::SinglePairManyEntries(key)) => {
392                        if is_bool {
393                            quote! {
394                                __oudia_items.push(crate::pair!(#key => .. self.#field_ident.iter().map(|value| if *value { "1" } else { "0" })));
395                            }
396                        } else {
397                            quote! {
398                                __oudia_items.push(crate::pair!(#key => .. self.#field_ident.iter().map(|value| value.to_string())));
399                            }
400                        }
401                    }
402                    (OuterType::Option(_), OuDiaType::SingleStruct(..)) => quote! {
403                        if let Some(value) = &self.#field_ident {
404                            __oudia_items.push(value.to_structure());
405                        }
406                    },
407                    (OuterType::Plain(_), OuDiaType::SingleStruct(..)) => quote! {
408                        __oudia_items.push(self.#field_ident.to_structure());
409                    },
410                    (OuterType::Vec(_), OuDiaType::ManyStructs(..)) => quote! {
411                        __oudia_items.extend(self.#field_ident.iter().map(|value| value.to_structure()));
412                    },
413                    (OuterType::Vec(_), OuDiaType::SingleStructManyEntries(key)) => quote! {
414                        __oudia_items.push(crate::structure!(#key => .. self.#field_ident.iter().map(|value| value.to_structure())));
415                    },
416                    // We don't have a generic way to split the values back into the two
417                    // keys, so serialize everything under the first key for now.
418                    (OuterType::Vec(_), OuDiaType::TwinStructMultipleEntries { first, .. }) => quote! {
419                        __oudia_items.push(crate::structure!(#first => .. self.#field_ident.iter().map(|value| value.to_structure())));
420                    },
421                    _ => unreachable!(),
422                });
423            }
424
425            assembler.extend(quote! {
426                #field_ident,
427            });
428
429            field.attrs.retain(|attr| !attr.path().is_ident("oudia"));
430
431            let mut field_names = Vec::new();
432            if let Some(key) = match &opts.kind {
433                OuDiaType::SinglePairSingleEntry(key)
434                | OuDiaType::SinglePairManyEntries(key)
435                | OuDiaType::SingleStruct(key)
436                | OuDiaType::ManyStructs(key)
437                | OuDiaType::SingleStructManyEntries(key) => Some(key.clone()),
438                OuDiaType::TwinStructMultipleEntries { .. } => None,
439            } {
440                field_names.push(key);
441            }
442            if let Some(alias) = &opts.alias {
443                field_names.push(alias.clone());
444            }
445            field.attrs.extend(doc_aliases(
446                &field_names,
447                &field_ident.map(|ident| ident.to_string()).unwrap_or_default(),
448            ));
449        }
450    }
451
452    let default_impl = if all_fields_default {
453        quote! {
454            impl #impl_generics Default for #struct_ident #ty_generics #where_clause {
455                fn default() -> Self {
456                    Self {
457                        #default_fields
458                    }
459                }
460            }
461        }
462    } else {
463        proc_macro2::TokenStream::new()
464    };
465
466    let expanded = quote! {
467        #input
468
469        #default_impl
470
471        impl #impl_generics crate::OuDiaIo for #struct_ident #ty_generics #where_clause {
472            const OUDIA_KEY: &'static str = #struct_oudia_key;
473
474            fn from_structure(input: &[crate::ast::Structure<'_>]) -> Result<Self, crate::IrConversionError> {
475                #initializers
476                for node in input {
477                    match node {
478                        #matchers
479                        #silencers
480                        // Unknown field. Give error in log.
481                        crate::ast::Structure::Pair(k, _) | crate::ast::Structure::Struct(k, _) => {
482                            log::debug!("Encountered unknown field `{k}` when parsing `{}`", std::any::type_name::<Self>());
483                        }
484                    }
485                }
486                #fixups
487                Ok(Self { #assembler })
488            }
489
490            fn to_structure(&self) -> crate::ast::Structure<'static> {
491                let mut __oudia_items: Vec<crate::ast::Structure<'static>> = Vec::new();
492                #serializers
493                crate::ast::Structure::Struct(Self::OUDIA_KEY.into(), __oudia_items)
494            }
495        }
496    };
497
498    TokenStream::from(expanded)
499}
