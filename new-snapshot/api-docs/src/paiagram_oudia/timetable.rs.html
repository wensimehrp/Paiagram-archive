8/// Also known as `駅扱`.
9#[doc(alias = "駅扱")]
10#[repr(u32)]
11#[derive(Default, Clone, Copy, Eq, PartialEq, Debug)]
12#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
13pub enum ServiceMode {
14    /// Also known as `運行なし`.
15    ///
16    /// The trip does not visit the station
17    #[default]
18    #[doc(alias = "運行なし")]
19    NoOperation = 0,
20    /// Also known as `停車`.
21    ///
22    /// The trip visits the station and stops
23    #[doc(alias = "停車")]
24    Stop = 1,
25    /// Also known as `通過`.
26    ///
27    /// The trip visits the station but does not stop
28    #[doc(alias = "通過")]
29    Pass = 2,
30}

32make_ir_type! {
33    /// A timetable entry
34    #[derive(Default)]
35    struct TimetableEntry as ["Ekijikoku", "駅時刻"];
36    pub service_mode as ["駅扱"]: ServiceMode,
37    pub arrival_time as ["着時刻"]: Option<Time>,
38    pub departure_time as ["発時刻"]: Option<Time>,
39    pub track_index as ["着発番線"]: Option<usize>,
40    /// Operations associated with this timetable entry.
41    operations as ["作業"]: Option<Box<RootOperationTree>>,
42}

49        self.operations.get_or_insert_default()

53impl std::fmt::Display for TimetableEntry {
54    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
55        write!(f, "{}", self.service_mode as u32)?;
56        if self.arrival_time.is_some() || self.departure_time.is_some() {
57            write!(f, ";")?;
58            if let Some(arrival) = self.arrival_time {
59                write!(f, "{}/", arrival.to_oud_string())?;
60            }
61            if let Some(departure) = self.departure_time {
62                write!(f, "{}", departure.to_oud_string())?;
63            }
64        }
65        if let Some(track) = self.track_index {
66            write!(f, "${}", track)?;
67        }
68        Ok(())
69    }
70}

79pub mod time {
80    use pest_consume::{Error, Parser};
81
82    use super::{ServiceMode, Time, TimetableEntry};
83
84    #[derive(Parser)]
85    #[grammar = "timetable.pest"]
86    pub struct TimeParser;
87
88    type Result<T> = std::result::Result<T, Error<Rule>>;
89    type Node<'i> = pest_consume::Node<'i, Rule, ()>;
90
91    #[pest_consume::parser]
92    impl TimeParser {
93        fn service_mode(input: Node<'_>) -> Result<ServiceMode> {
94            match input.as_str() {
95                "0" => Ok(ServiceMode::NoOperation),
96                "1" => Ok(ServiceMode::Stop),
97                "2" => Ok(ServiceMode::Pass),
98                _ => unreachable!(),
99            }
100        }
101        fn arrival_time(input: Node<'_>) -> Result<Time> {
102            Time::from_oud_str(input.as_str()).map_err(|e| input.error(e))
103        }
104        fn departure_time(input: Node<'_>) -> Result<Time> {
105            Time::from_oud_str(input.as_str()).map_err(|e| input.error(e))
106        }
107        fn track_index(input: Node<'_>) -> Result<usize> {
108            input.as_str().parse::<usize>().map_err(|e| input.error(e))
109        }
110        pub fn timetable_entry(input: Node<'_>) -> Result<TimetableEntry> {
111            let mut service_mode: ServiceMode = ServiceMode::default();
112            let mut arrival_time: Option<Time> = None;
113            let mut departure_time: Option<Time> = None;
114            let mut track_index: Option<usize> = None;
115            for node in input.into_children() {
116                match node.as_rule() {
117                    Rule::service_mode => service_mode = Self::service_mode(node)?,
118                    Rule::arrival_time => arrival_time = Some(Self::arrival_time(node)?),
119                    Rule::departure_time => departure_time = Some(Self::departure_time(node)?),
120                    Rule::track_index => track_index = Some(Self::track_index(node)?),
121                    _ => unreachable!(),
122                }
123            }
124            Ok(TimetableEntry {
125                service_mode,
126                arrival_time,
127                departure_time,
128                track_index,
129                ..Default::default()
130            })
131        }
132    }
133}

135pub(crate) fn normalize_times<'a>(mut time_iter: impl Iterator<Item = &'a mut Time> + 'a) {
136    let Some(mut previous_time) = time_iter.next().copied() else {
137        return;
138    };
139    for time in time_iter {
140        while *time < previous_time {
141            *time += Time::from_hms(24, 0, 0);
142        }
143        previous_time = *time;
144    }
145}

147pub(crate) fn parse_to_timetable_entry(
148    input: &'_ str,
149) -> Result<TimetableEntry, pest::error::Error<time::Rule>> {
150    let a = time::TimeParser::parse(time::Rule::timetable_entry, input)?.single()?;
151    Ok(time::TimeParser::timetable_entry(a)?)
152}
