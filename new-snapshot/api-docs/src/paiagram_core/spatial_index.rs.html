10/// Spatial caches used for fast geographic lookups.
11///
12/// Every tree is derived from the rest of the world, so the cache is never
13/// persisted. Instead it is rebuilt from scratch whenever the world changes.
14///
15/// The trees index projected positions ([`XyPos`]) rather than stored coordinates ([`Coord`]), so
16/// they can be queried directly with positions obtained from the screen.
17#[derive(Default, Clone, Debug)]
18pub struct SpatialCache {
19    pub intervals: IntervalsRTree,
20    pub nodes: NodesRTree,
21    pub trip_legs: TripLegsRTree,
22}
23
24impl SpatialCache {
25    pub fn build(world: &World) -> Self {
26        puffin::profile_function!();
27        // temp solution
28        let (intervals, cache) = IntervalsRTree::from_world(world.clone());
29        let (nodes, trip_legs) = join(
30            || NodesRTree::from_world(world.clone()),
31            || TripLegsRTree::from_world_and_cache(world.clone(), cache),
32        );
33        Self { intervals, nodes, trip_legs }
34    }
35}

37/// Normalise the two corners of a query and build the matching 2D envelope.
38fn envelope_2d(xy_min: XyPos, xy_max: XyPos) -> AABB<[i64; 2]> {
39    AABB::from_corners([xy_min.0.min(xy_max.0) as i64, xy_min.1.min(xy_max.1) as i64], [
40        xy_min.0.max(xy_max.0) as i64,
41        xy_min.1.max(xy_max.1) as i64,
42    ])
43}
44
45/// Like [`envelope_2d`], but collapsed onto the given instant on the time axis.
46fn envelope_3d(xy_min: XyPos, xy_max: XyPos, time: TimetableTime) -> AABB<[i64; 3]> {
47    let t = i64::from(time.0);
48    AABB::from_corners([xy_min.0.min(xy_max.0) as i64, xy_min.1.min(xy_max.1) as i64, t], [
49        xy_min.0.max(xy_max.0) as i64,
50        xy_min.1.max(xy_max.1) as i64,
51        t,
52    ])
53}

56pub struct IntervalRTreeObject {
57    pub interval_key: IntervalKey,
58    pub nodes: EcoVec<(XyPos, f32)>,
59}
60
61impl RTreeObject for IntervalRTreeObject {
62    type Envelope = AABB<[i64; 2]>;
63    fn envelope(&self) -> Self::Envelope {
64        self.nodes.iter().fold(AABB::new_empty(), |mut aabb, &(xy, _)| {
65            aabb.merge(&AABB::from_point([i64::from(xy.0), i64::from(xy.1)]));
66            aabb
67        })
68    }
69}

71/// Builds the progress and projected positions for an interval.
72///
73/// Distances are measured on the stored [`Coord`]s, since the projection may distort lengths;
74/// the positions themselves are projected so the index can be queried with [`XyPos`].
75fn build_progress(
76    graph: &Graph,
77    key: IntervalKey,
78    interval: Option<&Wfc<Interval, IntervalCache>>,
79) -> EcoVec<(XyPos, f32)> {
80    // stupid path here...
81    if let Some(interval) = interval.or(graph.intervals.get(&key)) {
82        let mut positions: EcoVec<(XyPos, f32)> = EcoVec::new();
83        let mut prev: Option<Coord> = None;
84        let mut total_distance = 0.0f32;
85        for coord in interval.positions(key, graph.nodes()) {
86            if let Some(prev) = prev {
87                total_distance += graph.coord_type.distance_to_meters(prev, coord) as f32;
88            }
89            prev = Some(coord);
90            positions.push((graph.coord_type.project(coord), total_distance));
91        }
92        // normalize the processes
93        for (_, progress) in positions.make_mut().iter_mut() {
94            *progress = *progress / total_distance
95        }
96        positions
97    } else {
98        // Short path if the interval has not been created
99        let hi_pos = graph.nodes().get(&key.hi).map_or_default(|wfc| wfc.pos);
100        let lo_pos = graph.nodes().get(&key.lo).map_or_default(|wfc| wfc.pos);
101        eco_vec![(graph.coord_type.project(hi_pos), 0.0), (graph.coord_type.project(lo_pos), 1.0)]
102    }
103}

118#[derive(Clone, Debug)]
119pub struct TripLegRTreeObject {
120    /// shared progress in high-key to low-key order
121    pub interval: EcoVec<(XyPos, f32)>,
122    pub is_hi_to_lo: bool,
123    pub trip_key: TripKey,
124    pub curr_arr: TimetableTime,
125    pub curr_dep: TimetableTime,
126    pub next_arr: TimetableTime,
127}
128
129impl RTreeObject for TripLegRTreeObject {
130    type Envelope = AABB<[i64; 3]>;
131    fn envelope(&self) -> Self::Envelope {
132        self.interval.iter().fold(AABB::new_empty(), |mut aabb, (xy, _)| {
133            let x = i64::from(xy.0);
134            let y = i64::from(xy.1);
135            let p1 = [x, y, i64::from(self.curr_arr.0)];
136            let p2 = [x, y, i64::from(self.curr_dep.0)];
137            let p3 = [x, y, i64::from(self.next_arr.0)];
138            aabb.merge(&AABB::from_point(p1));
139            aabb.merge(&AABB::from_point(p2));
140            aabb.merge(&AABB::from_point(p3));
141            aabb
142        })
143    }
144}

146#[derive(Default, Clone, Debug)]
147pub struct IntervalsRTree(RTree<IntervalRTreeObject>);
148
149impl IntervalsRTree {
150    /// Build the tree from every interval in the world's graph.
151    fn from_world(world: World) -> (Self, FxHashMap<IntervalKey, EcoVec<(XyPos, f32)>>) {
152        puffin::profile_function!();
153        let mut cache = FxHashMap::default();
154        let objects = world
155            .graph
156            .intervals
157            .iter()
158            .map(|(interval_key, interval)| IntervalRTreeObject {
159                interval_key: *interval_key,
160                nodes: {
161                    let interval_progress =
162                        build_progress(&world.graph, *interval_key, Some(interval));
163                    cache.insert(*interval_key, interval_progress.clone());
164                    interval_progress
165                },
166            })
167            .collect();
168        (Self(RTree::bulk_load(objects)), cache)
169    }
170
171    /// Every interval whose geometry intersects the given box.
172    pub fn get(&self, xy_min: XyPos, xy_max: XyPos) -> impl Iterator<Item = &IntervalRTreeObject> {
173        self.0.locate_in_envelope_intersecting(&envelope_2d(xy_min, xy_max))
174    }
175}

177#[derive(Default, Clone, Debug)]
178pub struct NodesRTree(RTree<NodeRTreeObject>);
179
180impl NodesRTree {
181    /// Build the tree from every node in the world's graph.
182    fn from_world(world: World) -> Self {
183        puffin::profile_function!();
184        let objects = world
185            .graph
186            .nodes()
187            .iter()
188            .map(|(node_key, node)| NodeRTreeObject {
189                node_key: *node_key,
190                xy: world.graph.coord_type.project(node.pos),
191            })
192            .collect();
193        Self(RTree::bulk_load(objects))
194    }
195
196    /// Every node inside the given box.
197    pub fn get(&self, xy_min: XyPos, xy_max: XyPos) -> impl Iterator<Item = NodeRTreeObject> {
198        self.0.locate_in_envelope_intersecting(&envelope_2d(xy_min, xy_max)).cloned()
199    }
200}

202#[derive(Default, Clone, Debug)]
203pub struct TripLegsRTree(RTree<TripLegRTreeObject>);
204
205impl TripLegsRTree {
206    /// Build the tree from the estimated legs of every trip.
207    fn from_world_and_cache(
208        world: World,
209        progresses: FxHashMap<IntervalKey, EcoVec<(XyPos, f32)>>,
210    ) -> Self {
211        puffin::profile_function!();
212        let mut objects = Vec::with_capacity(world.trips.len() * 10);
213        for (trip_key, trip) in &world.trips {
214            trip.schedule.estimates(&world.graph, |e| {
215                for [(curr_est, curr), (next_est, next)] in e.array_windows() {
216                    let (Some(curr_est), Some(next_est)) = (curr_est, next_est) else {
217                        continue;
218                    };
219                    if curr.node_key() == next.node_key() {
220                        continue;
221                    }
222                    let interval_key = IntervalKey::new(curr.node_key(), next.node_key());
223                    let progresses = progresses
224                        .get(&interval_key)
225                        .cloned()
226                        .unwrap_or_else(|| build_progress(&world.graph, interval_key, None));
227                    objects.push(TripLegRTreeObject {
228                        interval: progresses,
229                        is_hi_to_lo: curr.node_key() == interval_key.hi,
230                        trip_key: *trip_key,
231                        curr_arr: curr_est.arr,
232                        curr_dep: curr_est.dep,
233                        next_arr: next_est.arr,
234                    });
235                }
236            })
237        }
238        Self(RTree::bulk_load(objects))
239    }
240
241    /// Every trip leg whose geometry intersects the given box and whose time
242    /// span contains the given instant.
243    pub fn get(
244        &self,
245        xy_min: XyPos,
246        xy_max: XyPos,
247        time: TimetableTime,
248    ) -> impl Iterator<Item = TripLegRTreeObject> {
249        self.0.locate_in_envelope_intersecting(&envelope_3d(xy_min, xy_max, time)).cloned()
250    }
