14/// Travel mode. Travel mode defines how the vehicle travels.
15#[derive(Clone, Copy, Serialize, Deserialize, Debug, PartialEq)]
16pub enum TravelMode {
17    At(TTime),
18    For(TDuration),
19    Flexible,
20}
21
22impl TravelMode {
23    pub fn shifted_by(self, dd: TDuration) -> TravelMode {
24        match self {
25            Self::At(t) => Self::At(t + dd),
26            Self::For(d) => Self::For(d + dd),
27            Self::Flexible => Self::Flexible,
28        }
29    }
30    pub fn to_oud2_string(self, show_seconds: bool) -> String {
31        match self {
32            Self::At(t) => t.to_oud2_string(show_seconds),
33            Self::For(d) => "temp".to_string(),
34            Self::Flexible => "O".into(),
35        }
36    }
37}

39#[derive(Clone, Serialize, Deserialize, Copy, Debug, PartialEq, Hash)]
40pub struct TEntryId(u32);
41static NEXT_ENTRY_ID: AtomicU32 = AtomicU32::new(1);
42
43impl TEntryId {
44    /// Create a new dummy ID.
45    pub unsafe fn dummy() -> Self {
46        Self(0)
47    }
48    /// Create a new unique entry ID.
49    pub fn new() -> Self {
50        Self(NEXT_ENTRY_ID.fetch_add(1, Ordering::Relaxed))
51    }
52}

54/// An entry in the schedule. `TEntry` is short for `TimetableEntry`.
55///
56/// The schedule conatins an ordered list of entries. There are two categories for an entry:
57/// `derived`, which is automatically generated by the system and may be invalidated, and `pinned`,
58/// which can be only generated by sending explicit [`crate::Command`]s.
59///
60/// `derived` and `pinned` entries have different semantics. A derived entry may or may not visit
61/// the station, while a pinned entry must visit the station. Each pinned entry has an ID.
62#[derive(Clone, Serialize, Deserialize, Copy, Debug, PartialEq)]
63pub struct TEntry {
64    /// Which node does the trip visit?
65    pub node: NodeKey,
66    /// How does the trip arrive at, or pass the node?
67    pub arr_or_pass: TravelMode,
68    /// How does the trip leave the node?
69    pub dep: Option<TravelMode>,
70    /// Id of the trip
71    pub id: TEntryId,
72}
73
74/// The trip's schedule.
75#[derive(Clone, Serialize, Deserialize, PartialEq, Debug)]
76pub struct TripSchedule {
77    entries: EcoVec<TEntry>,
78}
79
80impl TripSchedule {
81    /// Create a new trip schedule
82    pub fn new(entries: EcoVec<TEntry>) -> Self {
83        Self { entries }
84    }
85
86    /// The schedule entries.
87    pub fn entries(&self) -> &[TEntry] {
88        &self.entries
89    }
90
91    /// Push a new entry to the end of the schedule
92    pub fn push(&mut self, new: TEntry) {
93        self.entries.push(new);
94    }
95
96    /// The duration of an entry since the previuos stable timepoint.
97    pub fn dur_since_prev_stable(
98        &self,
99        estimates: &[(Option<TEstimate>, EstimateEntry)],
100        id: TEntryId,
101    ) -> Option<TDuration> {
102        let Some(pos) = estimates
103            .iter()
104            .position(|(_, it)| matches!(it, EstimateEntry::Pinned(entry) if entry.id == id))
105        else {
106            return None;
107        };
108        let e1 = estimates[pos].0?.arr;
109        let e0 = estimates[..pos]
110            .iter()
111            .rfind(|(_, it)| {
112                matches!(
113                    make_se(*it),
114                    StackElem::AtAt(..) | StackElem::ForAt(..) | StackElem::ForFor(..)
115                )
116            })
117            .map(|(es, _)| *es)??
118            .dep;
119        Some(e1 - e0)
120    }
121
122    /// Shift the arr or pass mode of an entry, given the id.
123    pub fn shift_arr_or_pass(&mut self, id: TEntryId, duration: TimetableDuration) {
124        let Some(entry) = self.entries.make_mut().iter_mut().find(|e| e.id == id) else {
125            return;
126        };
127        match &mut entry.arr_or_pass {
128            TravelMode::At(t) => *t += duration,
129            TravelMode::For(d) => *d += duration,
130            TravelMode::Flexible => {}
131        }
132    }
133
134    pub fn shift_dep(&mut self, id: TEntryId, duration: TimetableDuration) {
135        let Some(entry) = self.entries.make_mut().iter_mut().find(|e| e.id == id) else {
136            return;
137        };
138        match &mut entry.dep {
139            Some(TravelMode::At(t)) => *t += duration,
140            Some(TravelMode::For(d)) => *d += duration,
141            Some(TravelMode::Flexible) | None => {}
142        }
143    }
144
145    /// Set the arr or pass mode of an entry, given the id.
146    pub fn set_arr_or_pass_mode(&mut self, id: TEntryId, mode: TravelMode) {
147        let Some(changed) = self.entries.iter().position(|e| e.id == id) else {
148            return;
149        };
150        let old_arrivals = self.arrival_times();
151        let entries = self.entries.make_mut();
152        entries[changed].arr_or_pass = mode;
153
154        // `For` arrivals are measured from the previous stable departure. Changing this
155        // entry's mode can move or remove that anchor, so re-anchor the following `For`
156        // entries to keep their absolute arrival times unchanged.
157        let mut prev_stable: Option<TTime> = None;
158        for (index, entry) in entries.iter_mut().enumerate() {
159            let arr = match entry.arr_or_pass {
160                TravelMode::At(t) => Some(t),
161                TravelMode::For(d) => {
162                    let arr = prev_stable.map(|p| p + d);
163                    match (index > changed, prev_stable, old_arrivals[index]) {
164                        (true, Some(prev), Some(old_arr)) => {
165                            entry.arr_or_pass = TravelMode::For(old_arr - prev);
166                            Some(old_arr)
167                        }
168                        _ => arr,
169                    }
170                }
171                TravelMode::Flexible => None,
172            };
173            let dep = match entry.dep.unwrap_or(TravelMode::Flexible) {
174                TravelMode::At(t) => Some(t),
175                TravelMode::For(d) => arr.map(|a| a + d),
176                TravelMode::Flexible => arr,
177            };
178            if let Some(dep) = dep {
179                prev_stable = Some(dep);
180            }
181        }
182    }
183
184    /// The absolute arrival time of each entry under the current modes.
185    fn arrival_times(&self) -> Vec<Option<TTime>> {
186        let mut prev_stable: Option<TTime> = None;
187        let mut arrivals = Vec::with_capacity(self.entries.len());
188        for entry in &self.entries {
189            let arr = match entry.arr_or_pass {
190                TravelMode::At(t) => Some(t),
191                TravelMode::For(d) => prev_stable.map(|p| p + d),
192                TravelMode::Flexible => None,
193            };
194            arrivals.push(arr);
195            let dep = match entry.dep.unwrap_or(TravelMode::Flexible) {
196                TravelMode::At(t) => Some(t),
197                TravelMode::For(d) => arr.map(|a| a + d),
198                TravelMode::Flexible => arr,
199            };
200            if let Some(dep) = dep {
201                prev_stable = Some(dep);
202            }
203        }
204        arrivals
205    }
206
207    /// Set the departure mode of an entry in the schedule, given the id.
208    pub fn set_dep_mode(&mut self, id: TEntryId, mode: Option<TravelMode>) {
209        let Some(entry) = self.entries.make_mut().iter_mut().find(|e| e.id == id) else {
210            return;
211        };
212        entry.dep = mode;
213    }
214
215    /// Removes an entry from the schedule, given the id.
216    pub fn remove(&mut self, id: TEntryId) -> Option<TEntry> {
217        self.entries.iter().position(|entry| entry.id == id).map(|idx| self.entries.remove(idx))
218    }
219
220    /// Push a new entry into the schedule, given the id.
221    pub fn insert(&mut self, idx: usize, entry: TEntry) {
222        self.entries.insert(idx, entry);
223    }
224}

226#[derive(Clone, Copy, Debug, Serialize)]
227pub struct TEstimate {
228    pub arr: TTime,
229    pub dep: TTime,
230}
231
232impl TEstimate {
233    pub fn duration(self) -> TDuration {
234        self.dep - self.arr
235    }

238#[derive(Clone, Copy, Debug, Serialize)]
239pub enum EstimateEntry {
240    Pinned(TEntry),
241    Derived(NodeKey),
242}
243
244impl EstimateEntry {
245    pub fn node_key(&self) -> NodeKey {
246        match self {
247            Self::Pinned(entry) => entry.node,
248            Self::Derived(node) => *node,
249        }
250    }
251    pub fn to_oud2_string_arr(self, show_seconds: bool) -> String {
252        match self {
253            Self::Pinned(entry) => entry.arr_or_pass.to_oud2_string(show_seconds),
254            Self::Derived(_) => "⇂".into(),
255        }
256    }
257    pub fn to_oud2_string_dep(self, show_seconds: bool) -> String {
258        match self {
259            Self::Pinned(entry) if let Some(dep) = entry.dep => dep.to_oud2_string(show_seconds),
260            _ => "⇂".into(),
261        }
262    }
263}

265/// A pending flexible entry, whose exact timepoint is only known once the next
266/// stable timepoint shows up.
267struct EstimateSketch {
268    /// The node the entry passes.
269    node: NodeKey,
270    /// The index of the entry's placeholder in the output buffer.
271    slot: usize,
272    /// The duration the vehicle dwells at the node.
273    dur: TDuration,
274}
275
276type OutputInner = (Option<TEstimate>, EstimateEntry);
277
278thread_local! {
279    static ESTIMATE_BUFFER: RefCell<Vec<EstimateSketch>> = RefCell::new(Vec::with_capacity(100));
280    static OUTPUT_BUFFER: RefCell<Vec<OutputInner>> = RefCell::new(Vec::with_capacity(100));
281}
282
283enum StackElem {
284    In(NodeKey, TDuration),
285    AtAt(NodeKey, TTime, TTime),
286    ForAt(NodeKey, TDuration, TTime),
287    ForFor(NodeKey, TDuration, TDuration),
288}
289
290impl TripSchedule {
291    /// Output the estimates of the trip at all places
292    pub fn estimates<F, R>(&self, graph: &Graph, f: F) -> R
293    where
294        F: FnMut(&[OutputInner]) -> R,
295    {
296        puffin::profile_function!();
297        ESTIMATE_BUFFER.with(|r| {
298            let mut estimate_buf = r.borrow_mut();
299            estimate_buf.clear();
300            OUTPUT_BUFFER.with(|r| {
301                let mut output_buf = r.borrow_mut();
302                output_buf.clear();
303                self.estimates_inner(graph, &mut estimate_buf, &mut output_buf, f)
304            })
305        })
306    }
307
308    pub fn full_route<'a>(&'a self, graph: &'a Graph) -> impl Iterator<Item = EstimateEntry> + 'a {
309        self.entries
310            .array_windows()
311            .flat_map(move |[curr, next]| {
312                let shortest_path = graph
313                    .shortest_path(curr.node, next.node)
314                    .map_or_default(|(mut nodes, _)| {
315                        nodes.pop(); // cut the end
316                        nodes
317                    })
318                    .into_iter()
319                    .skip(1) // cut the start
320                    .map(EstimateEntry::Derived);
321                std::iter::chain([EstimateEntry::Pinned(*curr)], shortest_path)
322            })
323            .chain(self.entries.last().copied().map(EstimateEntry::Pinned))
324    }
325
326    fn estimates_inner<F, R>(
327        &self,
328        graph: &Graph,
329        esb: &mut Vec<EstimateSketch>,
330        otb: &mut Vec<OutputInner>,
331        mut f: F,
332    ) -> R
333    where
334        F: FnMut(&[OutputInner]) -> R,
335    {
336        // The departure time and node of the last stable timepoint. `For` modes are
337        // relative to it, and it anchors the distance-based estimates.
338        let mut prev_stable: Option<(TTime, NodeKey)> = None;
339        for entry in self.full_route(graph) {
340            let se = make_se(entry);
341            match se {
342                StackElem::In(node, dur) => {
343                    // The estimate depends on the next stable timepoint, so only push
344                    // a placeholder for now and record where it lives.
345                    let slot = otb.len();
346                    otb.push((None, entry));
347                    esb.push(EstimateSketch { node, slot, dur });
348                }
349                // A stable timepoint unwinds all pending flexible entries.
350                StackElem::AtAt(node, at, dt) => {
351                    if let Some((prev_t, prev_n)) = prev_stable {
352                        unwind(graph, esb, otb, prev_t, prev_n, node, at - prev_t);
353                    } else {
354                        esb.clear();
355                    }
356                    otb.push((Some(TEstimate { arr: at, dep: dt }), entry));
357                    prev_stable = Some((dt, node));
358                }
359                StackElem::ForAt(node, ad, dt) => {
360                    if let Some((prev_t, prev_n)) = prev_stable {
361                        let ok = unwind(graph, esb, otb, prev_t, prev_n, node, ad);
362                        otb.push((
363                            ok.then(|| TEstimate {
364                                arr: prev_t + ad,
365                                dep: dt,
366                            }),
367                            entry,
368                        ));
369                    } else {
370                        esb.clear();
371                        otb.push((None, entry));
372                    }
373                    // `dt` is absolute, so this entry still anchors the next segment
374                    // even if its own estimate could not be resolved.
375                    prev_stable = Some((dt, node));
376                }
377                StackElem::ForFor(node, ad, dd) => {
378                    if let Some((prev_t, prev_n)) = prev_stable {
379                        let ok = unwind(graph, esb, otb, prev_t, prev_n, node, ad);
380                        if ok {
381                            let arr = prev_t + ad;
382                            let dep = arr + dd;
383                            otb.push((Some(TEstimate { arr, dep }), entry));
384                            prev_stable = Some((dep, node));
385                        } else {
386                            otb.push((None, entry));
387                        }
388                    } else {
389                        esb.clear();
390                        otb.push((None, entry));
391                    }
392                }
393            };
394        }
395        // finalize
396        f(otb.as_slice())
397    }
398}

400/// Resolve the pending flexible entries in `esb`, which lie between the previous
401/// stable timepoint `prev_t`/`prev_n` and the current stable timepoint at `curr_n`.
402/// `total` is the time from the previous departure to the current arrival.
403///
404/// The time is distributed proportionally to the lengths of the intervals along the
405/// way; the time spent dwelling at the flexible nodes does not count towards the
406/// travel time. The resolved estimates are written back into the corresponding output
407/// slots. Returns `false` if some interval along the way cannot be found in the graph; the
408/// pending entries are then left without estimates.
409fn unwind(
410    graph: &Graph,
411    esb: &mut Vec<EstimateSketch>,
412    otb: &mut Vec<OutputInner>,
413    prev_t: TTime,
414    prev_n: NodeKey,
415    curr_n: NodeKey,
416    total: TDuration,
417) -> bool {
418    let interval_length = |a: NodeKey, b: NodeKey| -> Option<Distance> {
419        if a == b {
420            return Some(Distance::ZERO);
421        }
422        let key = IntervalKey::new(a, b);
423        graph.intervals.get(&key).map(|e| e.length(key, graph))
424    };
425    // Dwelling does not count towards the average velocity.
426    let stop_dur: TDuration = esb.iter().map(|s| s.dur).sum();
427    let travel_dur = total - stop_dur;
428
429    // The total length of the segment, including the two stable endpoints.
430    let mut total_dis = 0i64;
431    let mut cur = prev_n;
432    for s in esb.iter() {
433        let Some(dis) = interval_length(cur, s.node) else {
434            esb.clear();
435            return false;
436        };
437        total_dis += dis.0 as i64;
438        cur = s.node;
439    }
440    let Some(dis) = interval_length(cur, curr_n) else {
441        esb.clear();
442        return false;
443    };
444    total_dis += dis.0 as i64;
445
446    // Distribute the travel time proportionally to the interval lengths. The lookups
447    // here cannot fail, as every pair was already checked above.
448    let total_dis_f = total_dis as f64;
449    let travel_s = travel_dur.0 as f64;
450    let mut t_f = prev_t.0 as f64;
451    let mut cur = prev_n;
452    for s in esb.drain(..) {
453        let dis = interval_length(cur, s.node).unwrap().0 as f64;
454        let leg = if total_dis_f == 0.0 {
455            0.0
456        } else {
457            travel_s * dis / total_dis_f
458        };
459        t_f += leg;
460        let arr = TimetableTime(t_f.round() as i32);
461        let dep = arr + s.dur;
462        otb[s.slot].0 = Some(TEstimate { arr, dep });
463        t_f += s.dur.0 as f64;
464        cur = s.node;
465    }
466    true
467}

469fn make_se(estimate_entry: EstimateEntry) -> StackElem {
470    use StackElem as Se;
471    use TravelMode as Tm;
472    let node = estimate_entry.node_key();
473    let match_entry = match estimate_entry {
474        EstimateEntry::Pinned(tentry) => tentry,
475        EstimateEntry::Derived(node) => TEntry {
476            node,
477            arr_or_pass: TravelMode::Flexible,
478            dep: None,
479            id: TEntryId(0),
480        },
481    };
482    match (
483        match_entry.arr_or_pass,
484        match_entry.dep.unwrap_or(Tm::Flexible),
485    ) {
486        (Tm::At(at), Tm::At(dt)) => Se::AtAt(node, at, dt),
487        (Tm::At(at), Tm::For(dd)) => Se::AtAt(node, at, at + dd),
488        (Tm::At(at), Tm::Flexible) => Se::AtAt(node, at, at),
489        (Tm::For(ad), Tm::At(dt)) => Se::ForAt(node, ad, dt),
490        (Tm::For(ad), Tm::For(dd)) => Se::ForFor(node, ad, dd),
491        (Tm::For(ad), Tm::Flexible) => Se::ForFor(node, ad, TDuration::ZERO),
492        (Tm::Flexible, Tm::At(dt)) => Se::AtAt(node, dt, dt),
493        (Tm::Flexible, Tm::For(dd)) => Se::In(node, dd),
494        (Tm::Flexible, Tm::Flexible) => Se::In(node, TDuration::ZERO),
495    }
