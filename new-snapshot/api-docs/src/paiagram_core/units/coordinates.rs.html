3#[derive(Clone, Copy, Debug, PartialEq)]
4pub struct Wgs84LonLat {
5    pub lon: f64,
6    pub lat: f64,
7}
8
9impl Wgs84LonLat {
10    pub const fn new(lon: f64, lat: f64) -> Self {
11        Self {
12            lon: lon.clamp(-180.0, 180.0),
13            lat: lat.clamp(-90.0, 90.0),
14        }
15    }
16    pub fn distance_to_meters(self, other: Self) -> f64 {
17        // Convert degrees to radians
18        let self_lat_rad = self.lat.to_radians();
19        let self_lon_rad = self.lon.to_radians();
20        let other_lat_rad = other.lat.to_radians();
21        let other_lon_rad = other.lon.to_radians();
22
23        // Differences in coordinates
24        let delta_lat = other_lat_rad - self_lat_rad;
25        let delta_lon = other_lon_rad - self_lon_rad;
26
27        // Haversine formula
28        let a = (delta_lat / 2.0).sin().powi(2)
29            + self_lat_rad.cos() * other_lat_rad.cos() * (delta_lon / 2.0).sin().powi(2);
30
31        let c = 2.0 * a.sqrt().atan2((1.0 - a).sqrt());
32
33        XyPos::EARTH_RADIUS_METERS * c
34    }
35}

54#[derive(Clone, Copy, Debug, PartialEq)]
55pub struct XyPos {
56    pub x: f64,
57    pub y: f64,
58}

60impl XyPos {
61    pub fn new(x: f64, y: f64) -> Self {
62        Self { x, y }
63    }
64    // EPSG:3857
65    /// The constant defined in EPSG:3857
66    const EARTH_RADIUS_METERS: f64 = 6_378_137.0;
67    const WEB_MERCATOR_MAX_LAT: f64 = 85.051_128_779_806_59;
68}

70impl From<Wgs84LonLat> for LonLat {
71    fn from(value: Wgs84LonLat) -> Self {
72        let lon = value.lon.clamp(-180.0, 180.0) * Self::CONVERSION_FACTOR_F64;
73        let lon = lon.round() as i32;
74        let lat = value.lat.clamp(-90.0, 90.0) * Self::CONVERSION_FACTOR_F64;
75        let lat = lat.round() as i32;
76        Self { lon, lat }
77    }
78}

80impl From<LonLat> for Wgs84LonLat {
81    fn from(value: LonLat) -> Self {
82        let lon = value.lon as f64 / LonLat::CONVERSION_FACTOR_F64;
83        let lon = lon.clamp(-180.0, 180.0);
84        let lat = value.lat as f64 / LonLat::CONVERSION_FACTOR_F64;
85        let lat = lat.clamp(-90.0, 90.0);
86        Self { lon, lat }
87    }
88}

90impl From<Wgs84LonLat> for XyPos {
91    fn from(value: Wgs84LonLat) -> Self {
92        let x = Self::EARTH_RADIUS_METERS * value.lon.to_radians();
93        let lat = value.lat.clamp(-Self::WEB_MERCATOR_MAX_LAT, Self::WEB_MERCATOR_MAX_LAT);
94        let lat_rad = lat.to_radians();
95        let y =
96            -Self::EARTH_RADIUS_METERS * (std::f64::consts::FRAC_PI_4 + lat_rad / 2.0).tan().ln();
97        Self { x, y }
98    }
99}

101impl From<XyPos> for Wgs84LonLat {
102    fn from(value: XyPos) -> Self {
103        let lon = (value.x / XyPos::EARTH_RADIUS_METERS).to_degrees();
104        let lat = (2.0 * (-value.y / XyPos::EARTH_RADIUS_METERS).exp().atan()
105            - std::f64::consts::FRAC_PI_2)
106            .to_degrees();
107        Self { lon, lat }
108    }
109}
