23/// What to account as a part of the station
24#[derive(Clone, Serialize, Deserialize, Debug, PartialEq)]
25pub enum StationRecord {
26    /// All platforms
27    All(StationKey),
28    /// Some platforms
29    Some(EcoVec<NodeKey>),
30}
31
32impl StationRecord {
33    pub fn nodes<'a>(&'a self, stations: &'a StationCollection) -> &'a [NodeKey] {
34        match *self {
35            StationRecord::All(station_key) => {
36                stations.get(&station_key).map_or_default(|wfc| wfc.cache.nodes.as_slice())
37            }
38            StationRecord::Some(ref nodes) => nodes.as_slice(),
39        }
40    }
41    pub fn station_key<'a>(
42        &'a self,
43        node_collection: &'a NodeCollection,
44    ) -> Option<&'a StationKey> {
45        match self {
46            StationRecord::All(station_key) => Some(station_key),
47            StationRecord::Some(nodes) => {
48                nodes.first().and_then(|key| node_collection.get(key)).map(|wfc| &wfc.parent)
49            }
50        }
51    }
52}

54/// An interval on the route
55#[derive(Clone, Serialize, Deserialize, Debug, PartialEq)]
56pub struct RouteInterval {
57    pub station_record: StationRecord,
58    /// Optional milestone to be displayed
59    pub milestone: Option<Distance>,
60    pub canvas_length: Option<CanvasLength>,
61    pub nodes: EcoVec<NodeKey>,
62}
63
64impl RouteInterval {
65    fn progresses<'a>(
66        &'a self,
67        world: &'a World,
68        next_stn_nodes: &'a [NodeKey],
69    ) -> impl Iterator<Item = (NodeKey, Option<f32>)> + 'a {
70        let curr_stn_nodes = self.station_record.nodes(&world.stations);
71        std::iter::chain(&self.nodes, curr_stn_nodes).map(|node_key| {
72            (
73                *node_key,
74                gen_progress(node_key, &world.graph, &self.nodes, curr_stn_nodes, next_stn_nodes),
75            )
76        })
77    }
78}

80// suboptimal implementation to generate progress
81fn gen_progress(
82    current_node: &NodeKey,
83    graph: &Graph,
84    interval_nodes: &[NodeKey],
85    curr_stn_nodes: &[NodeKey],
86    next_stn_nodes: &[NodeKey],
87) -> Option<f32> {
88    let is_part_of_interval = |key: &NodeKey| {
89        curr_stn_nodes.contains(key) || interval_nodes.contains(key) || next_stn_nodes.contains(key)
90    };
91    // roughly the same as the implementation in graph.rs
92    let successors = |source: &NodeKey| {
93        let source = *source;
94        graph.outgoing_intervals(source).filter_map(move |(interval_key, interval)| {
95            let target = interval_key.other(source);
96            is_part_of_interval(&target).then_some((target, interval.length(interval_key, graph)))
97        })
98    };
99    let distance_to_source = curr_stn_nodes
100        .into_iter()
101        .filter_map(|source| dijkstra(source, successors, |node| *node == *current_node))
102        .map(|(_, distance)| distance)
103        .min()?
104        .0 as f32;
105    let distance_to_target = next_stn_nodes
106        .into_iter()
107        .filter_map(|source| dijkstra(source, successors, |node| *node == *current_node))
108        .map(|(_, distance)| distance)
109        .min()?
110        .0 as f32;
111    let ratio = distance_to_source / (distance_to_source + distance_to_target);
112    Some(if ratio.is_nan() { 0.0 } else { ratio })
113}

118pub type BucketPoint = (TEstimate, EstimateEntry, u32, f32);
119pub type TripBucket = SmallVec<[Vec<BucketPoint>; 1]>;

121#[derive(Clone, Default, Debug, Serialize)]
122pub struct MareyCache {
123    pub map: TripKeyHashMap<TripBucket>,
124}
125
126impl RouteIntervals {
127    fn progresses<'a>(
128        &'a self,
129        world: &'a World,
130    ) -> impl Iterator<Item = impl Iterator<Item = (NodeKey, Option<f32>)>> + 'a {
131        self.0.array_windows().map(|[curr, next]| {
132            let next_stn_nodes = next.station_record.nodes(&world.stations);
133            curr.progresses(world, next_stn_nodes)
134        })
135    }
136
137    fn prepare_lookup(&self, world: &World) -> NodeKeyHashMap<Vec<(u32, f32)>> {
138        puffin::profile_function!();
139        // Where each graph node sits on the Marey chart: every `(line index, progress)` placement.
140        let mut node_lookup: NodeKeyHashMap<Vec<(u32, f32)>> = NodeKeyHashMap::default();
141        for (key, progress) in self.progresses(world).enumerate().flat_map(|(idx, it)| {
142            it.filter_map(move |(key, distance)| Some((key, (idx as u32, distance?))))
143        }) {
144            match node_lookup.get_mut(&key) {
145                Some(entries) => {
146                    if !entries.contains(&progress) {
147                        entries.push(progress);
148                    }
149                }
150                None => {
151                    node_lookup.insert(key, vec![progress]);
152                }
153            }
154        }
155
156        // `progresses` only walks pairs of stations, so the last station is never the `curr`
157        // station of a segment and would otherwise be missing from the lookup. Place it at the
158        // end of the final segment so trips connect to it.
159        if let Some(last) = self.0.last()
160            && let Some(line_index) = self.0.len().checked_sub(2)
161        {
162            let progress = (line_index as u32, 1.0);
163            for &node_key in last.station_record.nodes(&world.stations) {
164                let entries = node_lookup.entry(node_key).or_default();
165                if !entries.contains(&progress) {
166                    entries.push(progress);
167                }
168            }
169        }
170
171        node_lookup
172    }
173
174    pub fn all_passing_trips(&self, world: &World) -> Vec<TripKey> {
175        puffin::profile_function!();
176        let mut trip_keys: Vec<TripKey> = self
177            .0
178            .iter()
179            .flat_map(|route_interval| {
180                std::iter::chain(
181                    route_interval.station_record.nodes(&world.stations),
182                    &route_interval.nodes,
183                )
184            })
185            .flat_map(|node_key| world.graph.outgoing_intervals(*node_key))
186            .flat_map(|(_, wfc)| &wfc.cache.trips)
187            .copied()
188            .collect();
189        trip_keys.sort_unstable();
190        trip_keys.dedup();
191        trip_keys
192    }
193
194    /// Populates trips into a Marey cache
195    pub fn populate_trips(
196        &self,
197        world: &World,
198        cache: &mut MareyCache,
199        trips_to_rebuild: impl Iterator<Item = TripKey>,
200    ) {
201        puffin::profile_function!();
202        let node_lookup = self.prepare_lookup(world);
203
204        for trip_key in trips_to_rebuild {
205            let Some(trip) = world.trips.get(&trip_key) else {
206                continue;
207            };
208            let bucket = cache.map.entry(trip_key).or_default();
209            prep_marey::prep_single_segment(&trip.schedule, bucket, &world.graph, |entry| {
210                node_lookup.get(&entry.node_key()).map_or(&[], Vec::as_slice)
211            });
212            if bucket.is_empty() {
213                cache.map.remove(&trip_key);
214            }
215        }
216    }
217
218    /// Populates a single schedule into the Marey cache
219    pub fn populate_schedule(&self, world: &World, schedule: &TripSchedule) -> Option<TripBucket> {
220        let node_lookup = self.prepare_lookup(world);
221        let mut bucket = TripBucket::new();
222        prep_marey::prep_single_segment(schedule, &mut bucket, &world.graph, |entry| {
223            node_lookup.get(&entry.node_key()).map_or(&[], Vec::as_slice)
224        });
225        Some(bucket)
226    }
227
228    pub fn populate_route_timetable(&self, world: &World) -> RouteTimetable {
229        puffin::profile_function!();
230        let mut marey_cache = MareyCache::default();
231        self.populate_trips(world, &mut marey_cache, self.all_passing_trips(world).into_iter());
232        RouteTimetable::new(marey_cache, self.0.iter().map(|entry| &entry.station_record))
233    }
