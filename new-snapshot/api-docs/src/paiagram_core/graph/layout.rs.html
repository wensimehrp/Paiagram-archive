7impl Source {
8    pub fn auto_layout(&mut self) {
9        auto_layout(self);
10    }
11}
12
13fn auto_layout(source: &mut Source) {
14    puffin::profile_function!();
15    let mut graph: ForceGraph<NodeKey, ()> = ForceGraph::default();
16    let mut idx_map = HashMap::with_capacity(source.graph.nodes().len());
17    for &node in source.graph.nodes().keys() {
18        let idx = graph.add_node(Node::new("", node));
19        idx_map.insert(node, idx);
20    }
21    for edge_key in source.graph.intervals().keys() {
22        graph.add_edge(
23            idx_map.get(&edge_key.hi).copied().unwrap(),
24            idx_map.get(&edge_key.lo).copied().unwrap(),
25            (),
26        );
27    }
28    let mut simulation = Simulation::from_graph(graph, SimulationParameters::default());
29    for _ in 0..1000 {
30        simulation.update(0.1);
31        let max_step = simulation
32            .get_graph()
33            .node_weights()
34            .map(|node| node.location.distance(node.old_location))
35            .fold(0.0_f32, f32::max);
36        if max_step < 0.1 {
37            break;
38        }
39    }
40    let _ = source.mutate(|mut world| {
41        for node in simulation.get_graph().node_weights() {
42            let xy = XyPos(node.location.x.round() as i32, node.location.y.round() as i32);
43            let new_coord = world.graph.coord_type.unproject(xy);
44            world.graph.update_node_coordinate(node.data, new_coord);
45        }
46        Ok(world)
47    });
48}
