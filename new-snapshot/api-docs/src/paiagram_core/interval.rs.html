1use super::Interval;
2use crate::graph::Graph;
3use crate::{Coord, Distance, IntervalKey};

5impl Interval {
6    /// The length of the interval in metres.
7    ///
8    /// Uses the explicitly-stored length when present; otherwise sums the distances between
9    /// consecutive points of [`Interval::positions`] according to the graph's coordinate type.
10    pub fn length(&self, key: IntervalKey, graph: &Graph) -> Distance {
11        if let Some(d) = self.length {
12            return Distance(d.get() as i32);
13        };
14
15        let mut total = 0.0;
16        let mut prev: Option<Coord> = None;
17        for pos in self.positions(key, graph.nodes()) {
18            if let Some(prev) = prev {
19                total += graph.coord_type.distance_to_meters(prev, pos);
20            }
21            prev = Some(pos);
22        }
23        Distance(total.round() as i32)
24    }
25}
