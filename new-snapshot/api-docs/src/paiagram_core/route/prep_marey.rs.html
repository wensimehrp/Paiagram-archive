7pub(super) struct TripAllEdges<'a> {
8    /// Active edges that may be expanded.
9    pub active: TripBucket,
10    /// Finalized edges that are either invalidated because of an invalid estimate or finalized
11    /// because it's impossible to extend the edge any more
12    pub finalized: &'a mut TripBucket,
13}
14
15/// Prepare the segments for a trip on a marey chart
16pub(super) fn make_trip_segments<'a>(
17    mut all_edges: TripAllEdges<'a>,
18    (estimate, entry): &(Option<TEstimate>, EstimateEntry),
19    mut node_lookup: impl FnMut(NodeKey) -> &'a [(u32, f32)],
20) -> TripAllEdges<'a> {
21    let &Some(estimate) = estimate else {
22        // invalid estimate invalidates the working set
23        all_edges.finalized.extend(all_edges.active.drain(..));
24        return all_edges;
25    };
26
27    let curr_idxs = node_lookup(entry.node_key());
28    let mut new_active = SmallVec::new();
29
30    for &(curr_idx, curr_progress) in curr_idxs {
31        if let Some(pos) = all_edges.active.iter().position(|edge| {
32            let (_estimate, _node, station_idx, _progress) = edge.last().unwrap();
33            station_idx.abs_diff(curr_idx) <= 1
34        }) {
35            let mut edge = all_edges.active.swap_remove(pos);
36            edge.push((estimate, *entry, curr_idx, curr_progress));
37            new_active.push(edge);
38        } else {
39            new_active.push(vec![(estimate, *entry, curr_idx, curr_progress)]);
40        }
41    }
42
43    all_edges.finalized.extend(all_edges.active.drain(..));
44    all_edges.active = new_active;
45    all_edges
46}
