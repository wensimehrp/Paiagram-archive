2//! Module for handling GTFS Static feeds (<https://gtfs.org/>).
3//!
4//! GTFS is more general than the other supported formats: it has no notion of a
5//! single ordered line, and stops are referenced by id from the stop times. We
6//! map it onto Paiagram as follows:
7//!
8//! - every GTFS stop becomes a [`Node`]; stops sharing a `parent_station` are grouped under a
9//!   single [`Station`];
10//! - every route becomes a [`Route`], whose station sequence is taken from the first of its trips;
11//! - every trip becomes a [`Trip`], with one schedule entry per stop time;
12//! - an [`Interval`] is created for every pair of consecutive stops in a trip.
13

32/// The position of a GTFS stop. Missing coordinates default to `(0, 0)`.
33fn stop_lonlat(stop: &Stop) -> LonLat {
34    Wgs84LonLat::new(stop.longitude.unwrap_or(0.0), stop.latitude.unwrap_or(0.0)).into()
35}

37pub(super) fn parse_gtfs(data: &[u8]) -> Result<World, Box<dyn std::error::Error>> {
38    puffin::profile_function!();
39    let gtfs = Gtfs::from_reader(Cursor::new(data))?;
40    let mut world = World::default();
41
42    // Stops with a `parent_station` become nodes of that station; standalone
43    // stops become a station of their own.
44    let mut station_keys: HashMap<String, StationKey> = HashMap::new();
45    let mut node_keys: HashMap<String, NodeKey> = HashMap::new();
46    let mut node_station: HashMap<String, StationKey> = HashMap::new();
47
48    for (stop_id, stop) in &gtfs.stops {
49        let station_id = stop.parent_station.clone().unwrap_or_else(|| stop_id.clone());
50        let station_key = *station_keys.entry(station_id.clone()).or_insert_with(|| {
51            let parent = gtfs.stops.get(&station_id);
52            let name = parent
53                .and_then(|s| s.name.clone())
54                .or_else(|| stop.name.clone())
55                .unwrap_or_else(|| station_id.clone());
56            // Prefer the parent's coordinates, but fall back to the platform's when
57            // the parent has none.
58            let pos = parent
59                .filter(|s| s.longitude.is_some() && s.latitude.is_some())
60                .map_or_else(|| stop_lonlat(stop), |s| stop_lonlat(s));
61            let key = StationKey::new();
62            world.stations.insert(
63                key,
64                Wfc::new(Station {
65                    name: name.to_eco_string(),
66                    pos,
67                }),
68            );
69            key
70        });
71
72        let pos = stop_lonlat(stop);
73        let name = stop.name.clone().unwrap_or_else(|| stop_id.clone());
74        let node_key = NodeKey::new();
75        world.graph.insert_node(
76            node_key,
77            Node {
78                name: name.to_eco_string(),
79                parent: station_key,
80                pos,
81            },
82        );
83        node_keys.insert(stop_id.clone(), node_key);
84        node_station.insert(stop_id.clone(), station_key);
85    }
86
87    let mut route_classes: HashMap<String, ServiceClassKey> = HashMap::new();
88    for (route_id, route) in &gtfs.routes {
89        let color = route.color();
90        let key = ServiceClassKey::new();
91        let name = route
92            .short_name
93            .clone()
94            .or_else(|| route.long_name.clone())
95            .unwrap_or_else(|| route_id.clone());
96        world.service_classes.insert(
97            key,
98            Wfc::new(ServiceClass {
99                name: name.to_eco_string(),
100                style: StrokeStyle {
101                    color: Color32::from_rgb(color.r, color.g, color.b),
102                    width: 1,
103                },
104            }),
105        );
106        route_classes.insert(route_id.clone(), key);
107    }
108
109    let mut trips_by_route: HashMap<&str, Vec<&gtfs_structures::Trip>> = HashMap::new();
110    for trip in gtfs.trips.values() {
111        trips_by_route.entry(trip.route_id.as_str()).or_default().push(trip);
112    }
113    for (route_id, route) in &gtfs.routes {
114        let Some(trip) = trips_by_route.get(route_id.as_str()).and_then(|trips| trips.first())
115        else {
116            continue;
117        };
118
119        for window in trip.stop_times.windows(2) {
120            let (Some(&a), Some(&b)) = (
121                node_keys.get(&window[0].stop.id),
122                node_keys.get(&window[1].stop.id),
123            ) else {
124                continue;
125            };
126            if a == b {
127                continue;
128            }
129            // whoops I forgot to set up nodes
130            world.graph.insert_interval(
131                IntervalKey::new(a, b),
132                Wfc::new(Interval {
133                    intermediate_nodes: EcoVec::new(),
134                    length: None,
135                    direction: IntervalDirection::Both,
136                }),
137            );
138        }
139
140        let intervals = trip
141            .stop_times
142            .iter()
143            .filter_map(|stop_time| {
144                let station_key = node_station.get(&stop_time.stop.id)?;
145                Some(RouteInterval {
146                    station_record: StationRecord::All(*station_key),
147                    milestone: None,
148                    canvas_length: None,
149                    nodes: EcoVec::new(),
150                })
151            })
152            .collect::<EcoVec<_>>();
153        let name = route
154            .short_name
155            .clone()
156            .or_else(|| route.long_name.clone())
157            .unwrap_or_else(|| route_id.clone());
158        world.routes.insert(
159            RouteKey::new(),
160            Wfc::new(Route {
161                name: name.to_eco_string(),
162                intervals: RouteIntervals(intervals),
163            }),
164        );
165    }
166
167    // 5. Trips.
168    for (trip_id, trip) in &gtfs.trips {
169        let mut entries = EcoVec::new();
170        for stop_time in &trip.stop_times {
171            let Some(&node) = node_keys.get(&stop_time.stop.id) else {
172                continue;
173            };
174            let dep = stop_time.departure_time.map(|s| TimetableTime(s as i32));
175            let arr = stop_time.arrival_time.map(|s| TimetableTime(s as i32));
176            let (arr_or_pass, dep) = match (arr, dep) {
177                (Some(arr), Some(dep)) if arr != dep => {
178                    (TravelMode::At(arr), Some(TravelMode::At(dep)))
179                }
180                // A stop with `arrival == departure` is a pass-through.
181                (Some(arr), Some(_)) => (TravelMode::At(arr), None),
182                (Some(arr), None) => (TravelMode::At(arr), None),
183                // The first stop of a trip often only has a departure time.
184                (None, Some(dep)) => (TravelMode::At(dep), None),
185                (None, None) => (TravelMode::Flexible, None),
186            };
187            entries.push(TEntry {
188                node,
189                arr_or_pass,
190                dep,
191                id: TEntryId::new(),
192            });
193        }
194        if entries.is_empty() {
195            continue;
196        }
197        let name = trip
198            .trip_short_name
199            .clone()
200            .or_else(|| trip.trip_headsign.clone())
201            .unwrap_or_else(|| trip_id.clone());
202        world.trips.insert(
203            TripKey::new(),
204            Wfc::new(Trip {
205                name: name.to_eco_string(),
206                schedule: TripSchedule::new(entries),
207                service_class: route_classes.get(&trip.route_id).copied(),
208                vehicles: SmallVec::new(),
209            }),
210        );
211    }
212
213    Ok(world)
214}
