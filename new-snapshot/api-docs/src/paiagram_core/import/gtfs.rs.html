2//! Module for handling GTFS Static feeds (<https://gtfs.org/>).
3//!
4//! GTFS is more general than the other supported formats: it has no notion of a
5//! single ordered line, and stops are referenced by id from the stop times. We
6//! map it onto Paiagram as follows:
7//!
8//! - every GTFS stop becomes a [`Node`]; stops sharing a `parent_station` are grouped under a
9//!   single [`Station`];
10//! - every route becomes a [`Route`], whose station sequence is taken from the first of its trips;
11//! - every trip becomes a [`Trip`], with one schedule entry per stop time;
12//! - an [`Interval`] is created for every pair of consecutive stops in a trip.
13

32/// The position of a GTFS stop. Missing coordinates default to `(0, 0)`.
33fn stop_lonlat(stop: &Stop) -> Coord {
34    Coord::from_wgs84(stop.longitude.unwrap_or(0.0), stop.latitude.unwrap_or(0.0))
35}
36
37pub(super) fn parse_gtfs(data: &[u8]) -> Result<World, Box<dyn std::error::Error>> {
38    puffin::profile_function!();
39    let gtfs = Gtfs::from_reader(Cursor::new(data))?;
40    let mut world = World::default();
41
42    // Stops with a `parent_station` become nodes of that station; standalone
43    // stops become a station of their own.
44    let mut station_keys: HashMap<String, StationKey> = HashMap::new();
45    let mut node_keys: HashMap<String, NodeKey> = HashMap::new();
46    let mut node_station: HashMap<String, StationKey> = HashMap::new();
47
48    for (stop_id, stop) in &gtfs.stops {
49        let station_id = stop.parent_station.clone().unwrap_or_else(|| stop_id.clone());
50        let station_key = *station_keys.entry(station_id.clone()).or_insert_with(|| {
51            let parent = gtfs.stops.get(&station_id);
52            let name = parent
53                .and_then(|s| s.name.clone())
54                .or_else(|| stop.name.clone())
55                .unwrap_or_else(|| station_id.clone());
56            // Prefer the parent's coordinates, but fall back to the platform's when
57            // the parent has none.
58            let pos = parent
59                .filter(|s| s.longitude.is_some() && s.latitude.is_some())
60                .map_or_else(|| stop_lonlat(stop), |s| stop_lonlat(s));
61            let key = StationKey::new();
62            world.stations.insert(key, Wfc::new(Station { name: name.to_eco_string(), pos }));
63            key
64        });
65
66        let pos = stop_lonlat(stop);
67        let name = stop.name.clone().unwrap_or_else(|| stop_id.clone());
68        let node_key = NodeKey::new();
69        world.graph.insert_node(node_key, Node {
70            name: name.to_eco_string(),
71            parent: station_key,
72            pos,
73        });
74        node_keys.insert(stop_id.clone(), node_key);
75        node_station.insert(stop_id.clone(), station_key);
76    }
77
78    let mut route_classes: HashMap<String, ServiceClassKey> = HashMap::new();
79    for (route_id, route) in &gtfs.routes {
80        let color = route.color();
81        let key = ServiceClassKey::new();
82        let name = route
83            .short_name
84            .clone()
85            .or_else(|| route.long_name.clone())
86            .unwrap_or_else(|| route_id.clone());
87        world.service_classes.insert(
88            key,
89            Wfc::new(ServiceClass {
90                name: name.to_eco_string(),
91                style: StrokeStyle {
92                    color: Color32::from_rgb(color.r, color.g, color.b),
93                    width: 1,
94                },
95            }),
96        );
97        route_classes.insert(route_id.clone(), key);
98    }
99
100    let mut trips_by_route: HashMap<&str, Vec<&gtfs_structures::Trip>> = HashMap::new();
101    for trip in gtfs.trips.values() {
102        trips_by_route.entry(trip.route_id.as_str()).or_default().push(trip);
103    }
104    for (route_id, route) in &gtfs.routes {
105        let Some(trip) = trips_by_route.get(route_id.as_str()).and_then(|trips| trips.first())
106        else {
107            continue;
108        };
109
110        for window in trip.stop_times.windows(2) {
111            let (Some(&a), Some(&b)) =
112                (node_keys.get(&window[0].stop.id), node_keys.get(&window[1].stop.id))
113            else {
114                continue;
115            };
116            if a == b {
117                continue;
118            }
119            // whoops I forgot to set up nodes
120            world.graph.insert_interval(
121                IntervalKey::new(a, b),
122                Wfc::new(Interval {
123                    intermediate_nodes: EcoVec::new(),
124                    length: None,
125                    direction: IntervalDirection::Both,
126                }),
127            );
128        }
129
130        let intervals = trip
131            .stop_times
132            .iter()
133            .filter_map(|stop_time| {
134                let station_key = node_station.get(&stop_time.stop.id)?;
135                Some(RouteInterval {
136                    station_record: StationRecord::All(*station_key),
137                    milestone: None,
138                    canvas_length: None,
139                    nodes: EcoVec::new(),
140                })
141            })
142            .collect::<EcoVec<_>>();
143        let name = route
144            .short_name
145            .clone()
146            .or_else(|| route.long_name.clone())
147            .unwrap_or_else(|| route_id.clone());
148        world.routes.insert(
149            RouteKey::new(),
150            Wfc::new(Route { name: name.to_eco_string(), intervals: RouteIntervals(intervals) }),
151        );
152    }
153
154    // 5. Trips.
155    for (trip_id, trip) in &gtfs.trips {
156        let mut entries = EcoVec::new();
157        for stop_time in &trip.stop_times {
158            let Some(&node) = node_keys.get(&stop_time.stop.id) else {
159                continue;
160            };
161            let dep = stop_time.departure_time.map(|s| TimetableTime(s as i32));
162            let arr = stop_time.arrival_time.map(|s| TimetableTime(s as i32));
163            let (arr_or_pass, dep) = match (arr, dep) {
164                (Some(arr), Some(dep)) if arr != dep => {
165                    (TravelMode::At(arr), Some(TravelMode::At(dep)))
166                }
167                // A stop with `arrival == departure` is a pass-through.
168                (Some(arr), Some(_)) => (TravelMode::At(arr), None),
169                (Some(arr), None) => (TravelMode::At(arr), None),
170                // The first stop of a trip often only has a departure time.
171                (None, Some(dep)) => (TravelMode::At(dep), None),
172                (None, None) => (TravelMode::Flexible, None),
173            };
174            entries.push(TEntry { node, arr_or_pass, dep, id: TEntryId::new() });
175        }
176        if entries.is_empty() {
177            continue;
178        }
179        let name = trip
180            .trip_short_name
181            .clone()
182            .or_else(|| trip.trip_headsign.clone())
183            .unwrap_or_else(|| trip_id.clone());
184        world.trips.insert(
185            TripKey::new(),
186            Wfc::new(Trip {
187                name: name.to_eco_string(),
188                schedule: TripSchedule::new(entries),
189                service_class: route_classes.get(&trip.route_id).copied(),
190                vehicles: SmallVec::new(),
191            }),
192        );
193    }
194
195    Ok(world)
196}
