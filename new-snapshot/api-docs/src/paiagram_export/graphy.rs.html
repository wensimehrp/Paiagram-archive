19/// 1 PDF point is 1/72 inch, whereas [`CanvasLength`] is measured in millimetres.
20const PT_PER_MM: f64 = 72.0 / 25.4;
21/// Horizontal scale: five minutes of timetable span one centimetre of canvas, matching the
22/// Marey tile's default zoom.
23const TICKS_PER_5_MIN: f64 = 5.0 * 60.0 * 100.0;
24/// Padding (in PDF points) around the Marey chart.
25const MARGIN: f64 = 20.0;
26/// Extra time shown on either side of the Marey chart, in ticks (ten minutes).
27const TIME_PADDING_TICKS: i64 = 10 * 60 * 100;
28/// The largest allowed PDF page dimension (200 inches).
29const MAX_PAGE: f64 = 14400.0;
30/// Text size in PDF points; 13 egui points (1/96 inch) converted to 1/72 inch.
31const TEXT_SIZE: f32 = 13.0 * 72.0 / 96.0;
32
33/// Candidate time-grid spacings, in seconds.
34const TIME_SPACINGS_S: [i64; 10] = [1, 10, 30, 60, 300, 600, 1800, 3600, 14400, 86400];
35const TIME_LINE_MIN_WIDTH: f64 = 32.0;
36const TIME_LINE_MAX_WIDTH: f64 = 64.0;
37
38#[derive(Clone, Copy)]
39pub enum GraphyFormat {
40    Pdf,
41    Svg,
42    Png,
43}

45#[derive(Clone)]
46pub struct ExportGraphy {
47    pub world: World,
48    pub route: RouteKey,
49    pub cache: MareyCache,
50    pub format: GraphyFormat,
51    pub fonts: Vec<Cow<'static, [u8]>>,
52}

54impl paiagram_rw::ExportObject for ExportGraphy {
55    fn extension(&self) -> &str {
56        match self.format {
57            GraphyFormat::Pdf => "pdf",
58            GraphyFormat::Svg => "svg",
59            GraphyFormat::Png => "png",
60        }
61    }
62    fn default_filename(&self) -> &str {
63        "exported"
64    }
65    fn write_to_vec(&mut self, mut buf: Vec<u8>) -> io::Result<Vec<u8>> {
66        match self.format {
67            GraphyFormat::Pdf => {
68                let font = self
69                    .fonts
70                    .first()
71                    .and_then(|data| Font::new(krilla_data(data), 0))
72                    .ok_or_else(|| io::Error::other("no usable font was provided"))?;
73                let document = new_document(&self.world, self.route, &self.cache, font)
74                    .ok_or_else(|| io::Error::other("failed to build the PDF document"))?;
75                buf.extend_from_slice(&document);
76                Ok(buf)
77            }
78            GraphyFormat::Svg | GraphyFormat::Png => {
79                todo!()
80            }
81        }
82    }
83}

85fn krilla_data(data: &Cow<'static, [u8]>) -> Data {
86    match data {
87        Cow::Borrowed(bytes) => Data::from(*bytes),
88        Cow::Owned(bytes) => Data::from(bytes.clone()),
89    }
90}

92fn new_document(
93    world: &World,
94    route_key: RouteKey,
95    cache: &MareyCache,
96    font: Font,
97) -> Option<Vec<u8>> {
98    let route = world.routes.get(&route_key)?;
99    let (station_heights, station_names, total_height_mm) =
100        station_lines(world, &route.intervals.0);
101
102    // Time span covered by the Marey chart, padded a little on both sides. When the cache is empty,
103    // fall back to a whole day so there is still something to look at.
104    let (mut t_min, mut t_max) = time_range(cache).unwrap_or((0, 24 * 3600 * 100));
105    t_min -= TIME_PADDING_TICKS;
106    t_max += TIME_PADDING_TICKS;
107    let time_span = (t_max - t_min).max(1) as f64;
108
109    // Five minutes per centimetre horizontally, and 1:1 vertically (so one millimetre of canvas
110    // is one millimetre on the page).
111    let mut x_per_tick = CanvasLength::from_cm(1.0).to_postscript_pts() / TICKS_PER_5_MIN;
112    let mut y_per_mm = PT_PER_MM;
113
114    // Shrink uniformly if the drawing would exceed the maximum PDF page size.
115    let shrink = {
116        let width = time_span * x_per_tick + MARGIN * 2.0;
117        let height = total_height_mm * y_per_mm + MARGIN * 2.0;
118        (MAX_PAGE / width).min(MAX_PAGE / height).min(1.0)
119    };
120    x_per_tick *= shrink;
121    y_per_mm *= shrink;
122
123    let width = (time_span * x_per_tick + MARGIN * 2.0) as f32;
124    let height = (total_height_mm * y_per_mm + MARGIN * 2.0) as f32;
125
126    let to_x = |tick: i64| (MARGIN + (tick - t_min) as f64 * x_per_tick) as f32;
127    let to_y = |mm: f64| (MARGIN + mm * y_per_mm) as f32;
128
129    let content_left = to_x(t_min);
130    let content_right = to_x(t_max);
131    let content_top = to_y(0.0);
132    let content_bottom = to_y(total_height_mm);
133
134    let mut document = Document::new();
135    document.set_metadata(
136        Metadata::new()
137            .creator(concat!("Paiagram ", env!("CARGO_PKG_VERSION")).into())
138            .keywords(vec![
139                "Timetable".into(),
140                "Paiagram".into(),
141                "Train".into(),
142                "Transport".into(),
143                "Marey Chart".into(),
144            ])
145            .title("Paiagram Timetable".into()),
146    );
147    let mut page = document.start_page_with(PageSettings::from_wh(width, height)?);
148    let mut surface = page.surface();
149
150    // White background.
151    let background = {
152        let mut pb = PathBuilder::new();
153        pb.push_rect(Rect::from_xywh(0.0, 0.0, width, height)?);
154        pb.finish()?
155    };
156    surface.set_stroke(None);
157    surface.set_fill(Some(Fill {
158        paint: rgb::Color::new(255, 255, 255).into(),
159        opacity: NormalizedF32::ONE,
160        rule: Default::default(),
161    }));
162    surface.draw_path(&background);
163
164    // Station lines.
165    let station_stroke = gray_stroke(0.6, 1.0);
166    for height_mm in &station_heights {
167        let y = to_y(height_mm.0);
168        draw_line(&mut surface, content_left, y, content_right, y, &station_stroke);
169    }
170
171    // Time scale (lines and labels).
172    draw_time_lines(
173        &mut surface,
174        &font,
175        t_min,
176        t_max,
177        content_top,
178        content_bottom,
179        x_per_tick,
180        to_x,
181    );
182
183    // Trip (Marey) lines.
184    draw_trips(world, cache, &station_heights, &mut surface, to_x, to_y);
185
186    // Station labels, drawn last so they stay legible on top of the Marey lines.
187    if station_names.iter().any(|name| !name.is_empty()) {
188        surface.set_stroke(None);
189        surface.set_fill(Some(gray_fill(1.0)));
190        for (name, height_mm) in station_names.iter().zip(&station_heights) {
191            if name.is_empty() {
192                continue;
193            }
194            draw_text(
195                &mut surface,
196                &font,
197                content_left + 2.0,
198                to_y(height_mm.0) - 2.0,
199                name,
200                false,
201            );
202        }
203    }
204
205    surface.finish();
206    page.finish();
207    document.finish().ok()
208}

210/// The vertical position (in canvas millimetres) of the start of every route interval, the name
211/// of the station at that position, and the total height of the Marey chart.
212fn station_lines(
213    world: &World,
214    intervals: &[RouteInterval],
215) -> (Vec<CanvasLength>, Vec<String>, f64) {
216    let mut heights = Vec::with_capacity(intervals.len());
217    let mut names = Vec::with_capacity(intervals.len());
218    let mut total = 0.0_f64;
219    for interval in intervals {
220        heights.push(CanvasLength(total));
221        names.push(station_name(world, &interval.station_record).unwrap_or_default());
222        total += interval.canvas_length.unwrap_or_else(|| CanvasLength::from_cm(1.5)).0;
223    }
224    (heights, names, total)
225}

227/// The name of the station an interval's [`StationRecord`] belongs to.
228fn station_name(world: &World, record: &StationRecord) -> Option<String> {
229    let station = match record {
230        StationRecord::All(key) => *key,
231        StationRecord::Some(nodes) => {
232            let node = nodes.first()?;
233            world.graph.nodes().get(node)?.parent
234        }
235    };
236    world.stations.get(&station).map(|station| station.name.to_string())
237}

239/// The earliest and latest time (in ticks) covered by the cached trip polylines.
240fn time_range(cache: &MareyCache) -> Option<(i64, i64)> {
241    let mut min = i64::MAX;
242    let mut max = i64::MIN;
243    for polylines in cache.map.values() {
244        for polyline in polylines {
245            for (estimate, _entry, _idx, _progress) in polyline {
246                let arr = estimate.arr.to_ticks().0;
247                let dep = estimate.dep.to_ticks().0;
248                min = min.min(arr).min(dep);
249                max = max.max(arr).max(dep);
250            }
251        }
252    }
253    (min <= max).then_some((min, max))
254}

256fn draw_time_lines(
257    surface: &mut Surface<'_>,
258    font: &Font,
259    t_min: i64,
260    t_max: i64,
261    top: f32,
262    bottom: f32,
263    x_per_tick: f64,
264    to_x: impl Fn(i64) -> f32,
265) {
266    // How many ticks a single PDF point spans.
267    let ticks_per_point = 1.0 / x_per_tick;
268    let sizes: Vec<i64> = TIME_SPACINGS_S.iter().map(|secs| secs * 100).collect();
269    let first = sizes
270        .iter()
271        .position(|spacing| *spacing as f64 / ticks_per_point * 1.5 > TIME_LINE_MIN_WIDTH)
272        .unwrap_or(0);
273
274    let mut drawn: Vec<i64> = Vec::new();
275    // Coarsest spacing first, so finer grids skip the ticks that are already drawn.
276    for (level, &spacing) in sizes.iter().enumerate().skip(first).rev() {
277        let strength = ((spacing as f64 / ticks_per_point * 1.5 - TIME_LINE_MIN_WIDTH)
278            / (TIME_LINE_MAX_WIDTH - TIME_LINE_MIN_WIDTH))
279            .clamp(0.0, 1.0);
280        if strength < 0.1 {
281            continue;
282        }
283        let stroke = gray_stroke(0.5, strength as f32);
284        let mut tick = t_min - t_min.rem_euclid(spacing) - spacing;
285        while tick <= t_max {
286            tick += spacing;
287            if drawn.contains(&tick) {
288                continue;
289            }
290            drawn.push(tick);
291            let x = to_x(tick);
292            draw_line(surface, x, top, x, bottom, &stroke);
293            let label = time_label(Tick(tick).to_timetable_time(), level);
294            surface.set_stroke(None);
295            surface.set_fill(Some(gray_fill(1.0)));
296            draw_text(surface, font, x, top - 3.0, &label, true);
297        }
298    }
299}

301/// The label of a time-scale tick, matching the Marey tile: seconds for the finest levels,
302/// `H:MM` for the minute levels and the full `HH:MM:SS` for the day level.
303fn time_label(time: paiagram_core::time::TimetableTime, level: usize) -> String {
304    let (hours, minutes, seconds, _) = time.to_hmsd();
305    match level {
306        0..=2 => seconds.to_string(),
307        3..=8 => format!("{hours}:{minutes:02}"),
308        _ => time.to_string(),
309    }
310}

312fn draw_trips(
313    world: &World,
314    cache: &MareyCache,
315    station_heights: &[CanvasLength],
316    surface: &mut Surface<'_>,
317    to_x: impl Fn(i64) -> f32,
318    to_y: impl Fn(f64) -> f32,
319) {
320    for (trip_key, polylines) in &cache.map {
321        let Some(trip) = world.trips.get(trip_key) else {
322            continue;
323        };
324        let stroke =
325            trip.service_class.and_then(|key| world.service_classes.get(&key)).map_or_else(
326                || gray_stroke(1.0, 1.0),
327                |class| {
328                    let [r, g, b, a] = class.style.color.to_array();
329                    Stroke {
330                        paint: rgb::Color::new(r, g, b).into(),
331                        width: class.style.width as f32,
332                        opacity: NormalizedF32::new(a as f32 / 255.0).unwrap_or(NormalizedF32::ONE),
333                        ..Stroke::default()
334                    }
335                },
336            );
337
338        for polyline in polylines {
339            // Each entry spans `arr`..`dep` at one placement, so it contributes two points at
340            // the same height (mirroring the Marey tile's CPU drawing).
341            let mut points = Vec::with_capacity(polyline.len() * 2);
342            for (estimate, _entry, idx, progress) in polyline {
343                let (Some(start), Some(end)) =
344                    (station_heights.get(*idx as usize), station_heights.get(*idx as usize + 1))
345                else {
346                    continue;
347                };
348                let y_mm = start.0 + (end.0 - start.0) * *progress as f64;
349                let y = to_y(y_mm);
350                points.push((to_x(estimate.arr.to_ticks().0), y));
351                points.push((to_x(estimate.dep.to_ticks().0), y));
352            }
353            draw_polyline(surface, &points, &stroke);
354        }
355    }
356}

358fn gray_fill(opacity: f32) -> Fill {
359    Fill {
360        paint: rgb::Color::new(128, 128, 128).into(),
361        opacity: NormalizedF32::new(opacity).unwrap_or(NormalizedF32::ONE),
362        rule: Default::default(),
363    }
364}
365
366fn gray_stroke(width: f32, opacity: f32) -> Stroke {
367    Stroke {
368        paint: rgb::Color::new(128, 128, 128).into(),
369        width,
370        line_cap: LineCap::Round,
371        line_join: LineJoin::Round,
372        opacity: NormalizedF32::new(opacity).unwrap_or(NormalizedF32::ONE),
373        ..Stroke::default()
374    }
375}

377fn draw_line(surface: &mut Surface<'_>, x1: f32, y1: f32, x2: f32, y2: f32, stroke: &Stroke) {
378    draw_polyline(surface, &[(x1, y1), (x2, y2)], stroke);
379}
380
381fn draw_polyline(surface: &mut Surface<'_>, points: &[(f32, f32)], stroke: &Stroke) {
382    if points.len() < 2 {
383        return;
384    }
385    let mut pb = PathBuilder::new();
386    pb.move_to(points[0].0, points[0].1);
387    for &(x, y) in &points[1..] {
388        pb.line_to(x, y);
389    }
390    if let Some(path) = pb.finish() {
391        surface.set_fill(None);
392        surface.set_stroke(Some(stroke.clone()));
393        surface.draw_path(&path);
394    }
395}

397/// Draws a single line of text using the current fill. `y` is the baseline, and `centered`
398/// centres the text horizontally on `x`.
399fn draw_text(surface: &mut Surface<'_>, font: &Font, x: f32, y: f32, text: &str, centered: bool) {
400    if text.is_empty() {
401        return;
402    }
403    let x = if centered { x - estimate_text_width(text) * 0.5 } else { x };
404    surface.draw_text(
405        Point::from_xy(x, y),
406        font.clone(),
407        TEXT_SIZE,
408        text,
409        false,
410        TextDirection::Auto,
411    );
412}

414/// A rough advance-width estimate for a line of text, in PDF points.
415///
416/// krilla does not expose text measurement, so we approximate the width (half an em per ASCII
417/// character, a full em otherwise) to centre the time-scale labels.
418fn estimate_text_width(text: &str) -> f32 {
419    text.chars().map(|c| if c.is_ascii() { 0.5 } else { 1.0 }).sum::<f32>() * TEXT_SIZE
