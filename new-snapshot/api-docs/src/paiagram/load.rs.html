1use std::sync::mpsc;
2
3use paiagram_core::World;
4use paiagram_core::import::{ImportMode, ImportType, make_world};
5use paiagram_rw::LoadResult;
6use rfd::AsyncFileDialog;
7
8pub(crate) fn load_file(
9    dialog: AsyncFileDialog,
10    import_type: ImportType,
11    import_mode: ImportMode,
12    world_tx: mpsc::Sender<LoadResult<(World, ImportMode)>>,
13) {
14    let pick_data = async {
15        let handle = dialog.pick_file().await;
16        if let Some(handle) = handle {
17            Some(handle.read().await)
18        } else {
19            None
20        }
21    };
22    let process = async move {
23        #[cfg(debug_assertions)]
24        let data = if matches!(import_type, ImportType::BuiltinOud2) {
25            Some(Vec::new())
26        } else {
27            pick_data.await
28        };
29        #[cfg(not(debug_assertions))]
30        let data = pick_data.await;
31        let Some(data) = data else {
32            let _ = world_tx.send(LoadResult::Aborted);
33            return;
34        };
35        // for some reason egui's Context doesn't implement Send on wasm32. This means it can't be
36        // send to the rayon thread.
37        // Use a tx rx pair from futures_channel instead.
38        rayon::spawn(move || {
39            let _ = match make_world(&data, import_type) {
40                Ok(world) => world_tx.send(LoadResult::Success((world, import_mode))),
41                Err(e) => world_tx.send(LoadResult::Failed(anyhow::anyhow!("{e}"))),
42            };
43        });
44    };
45    #[cfg(target_arch = "wasm32")]
46    {
47        wasm_bindgen_futures::spawn_local(process);
48    }
49    #[cfg(not(target_arch = "wasm32"))]
50    {
51        let _ = rayon::spawn(move || pollster::block_on(process));
52    }
53}
