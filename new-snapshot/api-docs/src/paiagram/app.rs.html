59        let (world_read_tx, world_read_rx) = mpsc::channel();

89    /// Saves the current world to the stored path, or prompts for a location if none is set.
90    pub(crate) fn save(&mut self) {
91        let save_data = SaveData::from(self.world().clone());
92        #[cfg(not(target_arch = "wasm32"))]
93        if let Some(path) = self.save_path.clone() {
94            save_data.write_to_path(path, self.save_write_tx.clone());
95        } else {
96            save_data.write_via_interface(self.save_write_tx.clone());
97        }
98        #[cfg(target_arch = "wasm32")]
99        {
100            let name = self.save_path.as_ref().and_then(|p| p.to_str()).map(str::to_owned);
101            SaveCategory::Save(name).save(save_data, self.save_write_tx.clone());
102        }
103    }
104    /// Saves the current world to a new location.
105    pub(crate) fn save_as(&mut self) {
106        let save_data = SaveData::from(self.world().clone());
107        #[cfg(not(target_arch = "wasm32"))]
108        save_data.write_via_interface(self.save_write_tx.clone());
109        #[cfg(target_arch = "wasm32")]
110        save_data.write_via_interface(self.file_write_tx.clone());
111    }
112    /// Initializes the custom WGPU resources used by the Marey tile's render callback.
113    ///
114    /// Call once at startup, once the WGPU backend exists. `msaa_samples` must match the render
115    /// pass egui uses (eframe's `NativeOptions::multisampling`).
116    pub fn init_gpu(&mut self, render_state: &eframe::egui_wgpu::RenderState, msaa_samples: u32) {
117        crate::tiles::marey::gpu_draw::init(render_state, msaa_samples);
118        crate::tiles::map::gpu_draw::init(render_state, msaa_samples);
119    }
120    /// Apply UI commands and change the main ui state
121    pub(super) fn begin_frame(&mut self, ui: &mut Ui, mus: &mut UiTiles) {
122        puffin::profile_function!();
123        ui.set_theme(self.preferences.theme_preference);
124        self.source.begin_frame();
125        for cmd in self.ui_command_rx.try_iter() {
126            match cmd {
127                UiCommand::OpenOrFocus(tile) => mus.open_or_focus(tile),
128            }
129        }
130        for result in self.file_write_rx.try_iter() {
131            ui.request_repaint();
132            match result {
133                WriteResult::Success(path) => info!("Saved `{}`", path.display()),
134                WriteResult::Error(error) => {
135                    error!("Failed to write file: {error:#}");
136                }
137                WriteResult::Aborted => {
138                    warn!("Write operation aborted")
139                }
140            }
141        }
142        for result in self.save_write_rx.try_iter() {
143            ui.request_repaint();
144            match result {
145                WriteResult::Success(path) => {
146                    info!("Saved `{}`", path.display());
147                    self.save_path = Some(path);
148                }
149                WriteResult::Error(error) => {
150                    error!("Failed to write file: {error:#}");
151                }
152                WriteResult::Aborted => {
153                    warn!("Write operation aborted")
154                }
155            }
156        }
157        for result in self.world_read_rx.try_iter() {
158            ui.request_repaint();
159            puffin::profile_scope!("file_load");
160            match result {
161                LoadResult::Aborted => {
162                    info!("Load aborted");
163                }
164                LoadResult::Success((new_world, mode)) => {
165                    let _ = self.source.import(new_world, mode);
166                }
167                LoadResult::Failed(e) => {
168                    error!("Failed to load world: {e}");
169                }
170            }
171        }
172    }
173    /// Send a UI command to the UI command queue.
174    fn send_ui_cmd(&self, cmd: UiCommand) {
175        self.ui_command_tx
176            .send(cmd)
177            .map_err(|e| format!("Failed to send UI message to app: {e}"))
178            .unwrap();
179    }
180    /// Open or focus a tile. Shortcut for most tile-opening operations
181    pub(crate) fn open_or_focus(&self, tile: impl Into<UiTile>) {
182        self.send_ui_cmd(UiCommand::OpenOrFocus(tile.into()));
183    }
