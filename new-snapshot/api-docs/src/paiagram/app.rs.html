16pub struct App {
17    pub(crate) source: Source,
18    pub(crate) timer: GlobalTimer,
19    pub(crate) preferences: Preferences,
20    pub(crate) settings: Settings,
21    pub(crate) selected_items: SelectedItems,
22
23    // action and file operation transmitters and receivers
24    ui_command_rx: mpsc::Receiver<UiCommand>,
25    world_read_rx: mpsc::Receiver<LoadResult<(World, ImportMode)>>,
26    file_write_rx: mpsc::Receiver<WriteResult>,
27    ui_command_tx: mpsc::Sender<UiCommand>,
28    pub world_read_tx: mpsc::Sender<LoadResult<(World, ImportMode)>>,
29    pub file_write_tx: mpsc::Sender<WriteResult>,
30}

32/// Short definition for accessing the WORLD and SOURCE's data.
33impl ops::Deref for App {
34    type Target = Source;
35    fn deref(&self) -> &Self::Target {
36        &self.source
37    }
38}
39
40/// Short definition for accessing the SOURCE's data. Namely the [`Source::mutate`] method.
41impl ops::DerefMut for App {
42    fn deref_mut(&mut self) -> &mut Self::Target {
43        &mut self.source
44    }
45}

47impl App {
48    /// Creates a new app instance
49    pub fn new(ctx: &egui::Context) -> Self {
50        let (ui_command_tx, ui_command_rx) = mpsc::channel();
51        let (file_write_tx, file_write_rx) = mpsc::channel();
52        let (world_read_tx, world_read_rx) = mpsc::channel();
53        Self {
54            source: Source::new(),
55            timer: GlobalTimer::new(),
56            preferences: Preferences::new(ctx),
57            settings: Settings::default(),
58            selected_items: SelectedItems::None,
59            ui_command_tx,
60            ui_command_rx,
61            world_read_tx,
62            world_read_rx,
63            file_write_tx,
64            file_write_rx,
65        }
66    }
67    /// Initializes the custom WGPU resources used by the Marey tile's render callback.
68    ///
69    /// Call once at startup, once the WGPU backend exists. `msaa_samples` must match the render
70    /// pass egui uses (eframe's `NativeOptions::multisampling`).
71    pub fn init_gpu(&mut self, render_state: &eframe::egui_wgpu::RenderState, msaa_samples: u32) {
72        crate::tiles::marey::gpu_draw::init(render_state, msaa_samples);
73        crate::tiles::graph::gpu_draw::init(render_state, msaa_samples);
74    }
75    /// Apply UI commands and change the main ui state
76    pub(super) fn begin_frame(&mut self, ui: &mut Ui, mus: &mut UiTiles) {
77        puffin::profile_function!();
78        self.source.begin_frame();
79        for cmd in self.ui_command_rx.try_iter() {
80            match cmd {
81                UiCommand::OpenOrFocus(tile) => mus.open_or_focus(tile),
82            }
83        }
84        for result in self.file_write_rx.try_iter() {
85            ui.request_repaint();
86            match result {
87                WriteResult::Success(path) => info!("Saved `{}`", path.display()),
88                WriteResult::Error(error) => {
89                    error!("Failed to write file: {error}");
90                }
91                WriteResult::Aborted => {
92                    warn!("Write operation aborted")
93                }
94            }
95        }
96        for result in self.world_read_rx.try_iter() {
97            ui.request_repaint();
98            puffin::profile_scope!("file_load");
99            match result {
100                LoadResult::Aborted => {
101                    info!("Load aborted");
102                }
103                LoadResult::Success((new_world, mode)) => {
104                    let _ = self.source.import(new_world, mode);
105                }
106                LoadResult::Failed(e) => {
107                    error!("Failed to load world: {e}");
108                }
109            }
110        }
111    }
112    /// Send a UI command to the UI command queue.
113    fn send_ui_cmd(&self, cmd: UiCommand) {
114        self.ui_command_tx
115            .send(cmd)
116            .map_err(|e| format!("Failed to send UI message to app: {e}"))
117            .unwrap();
118    }
119    /// Open or focus a tile. Shortcut for most tile-opening operations
120    pub(crate) fn open_or_focus(&self, tile: impl Into<UiTile>) {
121        self.send_ui_cmd(UiCommand::OpenOrFocus(tile.into()));
122    }
123}
