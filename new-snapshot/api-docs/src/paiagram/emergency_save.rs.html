2use std::panic::{self, PanicHookInfo};
3use std::path::Path;
4
5use log::error;
6use paiagram_core::{SaveData, WORLD_BACKUP, World};
7
8/// Set up the emergency save pipeline
9pub fn setup_emergency_save() {
10    let default_hook = panic::take_hook();
11    panic::set_hook(Box::new(move |info| {
12        _setup_emergency_save(info);
13        default_hook(info);
14    }));
15}
16
17fn _setup_emergency_save(panic_info: &PanicHookInfo<'_>) {
18    error!("An unexpected error occurred. The app has panicked and crashed.");
19    error!("Panic info: {:?}", panic_info);
20
21    let Some(world_guard) = WORLD_BACKUP.load_full() else {
22        error!("Failed to create backup: no world stored in backup");
23        return;
24    };
25
26    let world: &World = &world_guard;
27    error!("Attempting to create emergency save...");
28
29    #[allow(unused_mut)]
30    let mut save = SaveData::from(world.clone());
31
32    #[cfg(target_arch = "wasm32")]
33    {
34        use paiagram_rw::ExportObject;
35        let filename = format!(
36            "paiagram-emergency-{}.{}",
37            jiff::Timestamp::now().to_string().replace(':', "-"),
38            save.extension(),
39        );
40        match save.write_to_vec(save.default_buf()) {
41            Ok(bytes) => match paiagram_rw::save::download_file(&filename, &bytes) {
42                Ok(()) => error!("Emergency save downloaded as `{filename}`"),
43                Err(e) => error!("Failed to download emergency save: {e:?}"),
44            },
45            Err(e) => error!("Failed to serialize emergency save: {e}"),
46        }
47        show_dialog(Path::new(&filename));
48    }
49
50    #[cfg(not(target_arch = "wasm32"))]
51    {
52        use paiagram_rw::{SaveCategory, WriteResult};
53        let (tx, rx) = std::sync::mpsc::channel();
54
55        SaveCategory::Emergency.save(save, tx);
56        match rx.recv_timeout(std::time::Duration::from_secs(30)) {
57            Ok(result) => match result {
58                WriteResult::Aborted => error!("Write was aborted"),
59                WriteResult::Error(e) => error!("Error while writing file: {}", e),
60                WriteResult::Success(path) => {
61                    error!("Successfully wrote to `{}`", path.display());
62                    show_dialog(&path);
63                }
64            },
65            Err(e) => {
66                error!("Failed to save file: {}", e)
67            }
68        }
69    }
70}
71
72fn show_dialog(path: &Path) {
73    let desc =
74        format!("The app has crashed. An emergency save at {} has been created.", path.display());
75    let title = "Emergency Save";
76    cfg_select! {
77        target_arch = "wasm32" => wasm_bindgen_futures::spawn_local(async move {
78            rfd::AsyncMessageDialog::new()
79                .set_title(title)
80                .set_description(desc)
81                .set_level(rfd::MessageLevel::Error)
82                .show()
83                .await;
84        }),
85        _ => {
86            rfd::MessageDialog::new()
87                .set_title(title)
88                .set_description(desc)
89                .set_level(rfd::MessageLevel::Error)
90                .show();
91        }
92    }
93}
