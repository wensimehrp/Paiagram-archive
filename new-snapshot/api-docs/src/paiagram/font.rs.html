22pub(crate) fn load_default_font(ctx: Context, font_name: Arc<Mutex<String>>) {
23    let mut definitions = FontDefinitions::default();
24    // custom timetable font
25    definitions
26        .families
27        .entry(FontFamily::Name("timetable font".into()))
28        .or_default()
29        .insert(0, "Paiagram-Timetable-Font".into());
30    // monospace fallback. DO NOT USE MONOSPACE FOR TIMETABLE STUFF
31    definitions
32        .families
33        .entry(FontFamily::Monospace)
34        .or_default()
35        .insert(0, "Paiagram-Timetable-Font".into());
36    // proportional fallback.
37    definitions
38        .families
39        .entry(FontFamily::Proportional)
40        .or_default()
41        .insert(0, "Paiagram-Timetable-Font".into());
42    ctx.all_styles_mut(|s| {
43        s.text_styles.insert(TIMETABLTE_TEXT_STYLE.clone(), TIMETABLE_FONT_ID.clone());
44    });
45    definitions.font_data.insert(
46        "Paiagram-Timetable-Font".into(),
47        Arc::new(egui::FontData::from_static(include_bytes!(
48            "../../../assets/fonts/Paiagram-Timetable-Font.ttf"
49        ))),
50    );
51    info!("Loaded Paiagram-Timetable-Font");
52    ctx.set_fonts(definitions.clone());
53    // egui_material_icons::initialize(&ctx);
54    // dynamic query
55    match FONT_DATABASE.read().query(&fontdb::Query {
56        families: &[
57            Family::Name("Sarasa UI SC"),
58            Family::Name("Sarasa Gothic SC"),
59            Family::SansSerif,
60        ],
61        ..Default::default()
62    }) {
63        Some(face_id) => {
64            load_font_name(face_id, font_name.clone());
65            load_font_to_egui(face_id, ctx.clone(), font_name.clone(), definitions);
66        }
67        None => {
68            warn!("Couldn't select font from system");
69            let mut name = font_name.lock();
70            name.clear();
71            name.push_str("Error");
72        }
73    };
74    #[cfg(target_arch = "wasm32")]
75    load_sarasa_cl(ctx.clone(), font_name.clone());
76}

79fn load_sarasa_cl(ctx: Context, font_name: Arc<Mutex<String>>) {
80    const WASM_FONT_NAME: &str = "SarasaUiCL-Regular.ttf";
81    wasm_bindgen_futures::spawn_local(async move {
82        let parse = async move || -> anyhow::Result<_> {
83            info!("Downloading `{WASM_FONT_NAME}`");
84            let window = web_sys::window().context("Browser window should be always present")?;
85            let document =
86                window.document().context("Browser document should be always present")?;
87            // Use the document's <base> URL instead of `location.href`, which also contains the
88            // query string and fragment (e.g. `#dev`) and would corrupt the font URL.
89            let base_url =
90                document.base_uri().ok().flatten().context("Document has no base URI")?;
91            let query_url = web_sys::Url::new_with_base(WASM_FONT_NAME, &base_url)
92                .map_err(|_| anyhow::anyhow!("Failed to resolve the font URL"))?
93                .href();
94            let res = reqwest::get(query_url.clone())
95                .await
96                .with_context(|| format!("Failed to get font from {query_url}"))?;
97            if !res.status().is_success() {
98                bail!("Response did not success, got code {}", res.status());
99            }
100            res.bytes().await.context("Failed to get bytes from response")
101        };
102        match parse().await {
103            Err(e) => {
104                warn!("Failed to fetch font {WASM_FONT_NAME}: {e}")
105            }
106            Ok(bytes) => {
107                let Some(face_id) = FONT_DATABASE
108                    .write()
109                    .load_font_source(fontdb::Source::Binary(Arc::new(bytes)))
110                    .into_iter()
111                    .next()
112                else {
113                    warn!("Failed to load font {WASM_FONT_NAME}: no valid font face");
114                    return;
115                };
116                load_font_to_egui(
117                    face_id,
118                    ctx.clone(),
119                    font_name,
120                    ctx.fonts(|r| r.definitions().clone()),
121                );
122            }
123        }
124    });
125}

127fn load_font_name(face_id: ID, font_name: Arc<Mutex<String>>) {
128    let post_script_name =
129        FONT_DATABASE.read().face(face_id).map_or("UNNAMED FONT".to_string(), |info| {
130            info.post_script_name.to_string()
131        });
132    *font_name.lock() = post_script_name;
133}

135pub(crate) fn load_font_to_egui(
136    face_id: ID,
137    ctx: Context,
138    font_name: Arc<Mutex<String>>,
139    mut definitions: FontDefinitions,
140) {
141    let Some(bytes) =
142        FONT_DATABASE.read().with_face_data(face_id, |font_bytes, _index| font_bytes.to_owned())
143    else {
144        warn!("Couldn't load font!");
145        return;
146    };
147    let bytes = Arc::new(bytes);
148    definitions.font_data.insert(
149        "system_font".into(),
150        Arc::new(egui::FontData::from_owned((*bytes).clone())),
151    );
152    definitions
153        .families
154        .entry(FontFamily::Proportional)
155        .or_default()
156        .insert(0, "system_font".into());
157    load_font_name(face_id, font_name);
158    ctx.set_fonts(definitions);
159    ctx.request_repaint();
160}
