48pub fn show_ui(
49    ui: &mut Ui,
50    app: &mut App,
51    ui_state: &mut UiState,
52    delta_time: std::time::Duration,
53) {
54    puffin::GlobalProfiler::lock().new_frame();
55    puffin::profile_function!();
56    app.begin_frame(ui, &mut ui_state.mus);
57
58    {
59        puffin::profile_scope!("command_palette");
60        ui_state.command_palette.show(ui.ctx(), app);
61    }
62
63    if ui.ctx().input_mut(|i| i.consume_key(Modifiers::COMMAND | Modifiers::SHIFT, Key::S)) {
64        app.save_as();
65    } else if ui.ctx().input_mut(|i| i.consume_key(Modifiers::COMMAND, Key::S)) {
66        app.save();
67    }
68
69    Panel::top("top panel").exact_size(32.0).show(ui, |ui| {
70        puffin::profile_scope!("top_panel");
71        ui.with_layout(egui::Layout::left_to_right(egui::Align::Center), |ui| {
72            Popup::menu(&ui.button("App")).show(|ui| {
73                if ui.add(Button::new("Command Palette").shortcut_text("Ctrl+P")).clicked() {
74                    ui_state.command_palette.visible ^= true;
75                }
76                if ui.button("Toggle Fullscreen").clicked() {
77                    cfg_select! {
78                        target_arch = "wasm32" => {
79                            let document = web_sys::window().unwrap().document().unwrap();
80                            if document.fullscreen_element().is_none() {
81                                let _ = document
82                                    .get_element_by_id("paiagram_canvas")
83                                    .unwrap()
84                                    .request_fullscreen();
85                            } else {
86                                document.exit_fullscreen();
87                            }
88                        }
89                        _ => {
90                            let is_fullscreen =
91                                ui.input(|i| i.viewport().fullscreen.unwrap_or(false));
92                            ui.send_viewport_cmd(egui::ViewportCommand::Fullscreen(!is_fullscreen));
93                        }
94                    }
95                }
96            });
97            Popup::menu(&ui.button("File")).show(|ui| {
98                for (button_display, category, import_type) in [
99                    ("Import OuDiaSecond", "OuDiaSecond", ImportType::OuDiaSecond),
100                    ("Import OuDia", "OuDia", ImportType::OuDia),
101                    ("Import qETRC/pyETRC", "pyetgr", ImportType::Pyetgr),
102                    ("Import GTFS", "GTFS", ImportType::Gtfs),
103                    ("Import .paia", "Paiagram", ImportType::PaiagramPaia),
104                    #[cfg(debug_assertions)]
105                    ("Import sample.oud2", "OuDiaSecond", ImportType::BuiltinOud2),
106                ] {
107                    ui.menu_button(button_display, |ui| {
108                        for (label, import_mode) in [
109                            ("Add to current world", ImportMode::Merge),
110                            ("Replace current world", ImportMode::Replace),
111                        ] {
112                            if !ui.button(label).clicked() {
113                                continue;
114                            }
115                            if matches!(import_mode, ImportMode::Replace) {
116                                app.save_path = None;
117                            }
118                            info!("Trying to read {category}");
119                            let dialog = AsyncFileDialog::new()
120                                .set_title(button_display)
121                                .add_filter(category, import_type.file_extensions());
122                            load::load_file(
123                                dialog,
124                                import_type,
125                                import_mode,
126                                app.world_read_tx.clone(),
127                            );
128                            ui.close();
129                        }
130                    });
131                }
132                ui.separator();
133                if ui.add(Button::new("Save").shortcut_text("Ctrl+S")).clicked() {
134                    app.save();
135                    ui.close();
136                }
137                if ui.add(Button::new("Save As...").shortcut_text("Ctrl+Shift+S")).clicked() {
138                    app.save_as();
139                    ui.close();
140                }
141                ui.separator();
142                if ui.button("Export .oud").clicked() {
143                    ExportOuDia {
144                        world: app.world().clone(),
145                        route: RouteKey::new(),
146                        is_oudia_second: false,
147                    }
148                    .write_via_interface(app.file_write_tx.clone());
149                }
150                if ui.button("Export GTFS").clicked() {
151                    ExportGtfs {
152                        world: app.world().clone(),
153                    }
154                    .write_via_interface(app.file_write_tx.clone());
155                }
156            });
157            Popup::menu(&ui.button(tr!("menu-about"))).show(|ui| {
158                if ui.button(tr!("menu-documentation")).clicked() {
159                    ui.ctx().open_url(OpenUrl::new_tab(if cfg!(target_arch = "wasm32") {
160                        "/docs"
161                    } else {
162                        "https://paiagram.com/docs"
163                    }));
164                }
165                if cfg!(target_arch = "wasm32") && ui.button(tr!("menu-legal")).clicked() {
166                    ui.ctx().open_url(OpenUrl::new_tab("./license.html"));
167                }
168            });
169            ui.with_layout(egui::Layout::right_to_left(egui::Align::Center), |ui| {
170                ui.menu_button("History", |ui| {
171                    ScrollArea::vertical().show_rows(
172                        ui,
173                        ui.spacing().interact_size.y,
174                        app.history().len(),
175                        |ui, range| {
176                            for (idx, _) in app.history()[range].iter().enumerate() {
177                                ui.button(idx.to_string());
178                            }
179                        },
180                    )
181                });
182                if ui.add_enabled(app.redoable(), Button::new("Redo")).clicked() {
183                    app.redo();
184                }
185                if ui.add_enabled(app.undoable(), Button::new("Undo")).clicked() {
186                    app.undo();
187                }
188                const GIT_REV_SHORT: &str = git_version::git_version!(fallback = "unknown");
189                const GH_LINK: &str = git_version::git_version!(
190                    args = ["--always", "--abbrev=40"],
191                    prefix = "https://github.com/wensimehrp/Paiagram/commit/",
192                    fallback = "https://github.com/wensimehrp/Paiagram/"
193                );
194                ui.hyperlink_to(GIT_REV_SHORT, GH_LINK);
195            });
196        })
197    });
198    Panel::bottom("bottom panel").exact_size(24.0).show(ui, |ui| {
199        puffin::profile_scope!("bottom_panel");
200        ui.horizontal_centered(|ui| {
201            let mut time = TimetableTime(app.timer.seconds() as i32);
202            ui.add_enabled(
203                !app.timer.sync_to_real_time,
204                egui::Checkbox::new(&mut app.timer.animation_playing, ""),
205            );
206            let time_response = ui.add(DragValue::new(&mut time));
207            ui.add_enabled(
208                !app.timer.sync_to_real_time,
209                egui::DragValue::new(&mut app.timer.animation_speed).fixed_decimals(1).suffix("×"),
210            );
211            egui::Popup::menu(&time_response).show(|ui| {
212                ui.checkbox(
213                    &mut app.timer.sync_to_real_time,
214                    tr!("menu-sync-system-clock"),
215                );
216            });
217            if !app.timer.sync_to_real_time
218                && time_response.dragged()
219                && let Some(key) = app.timer.try_lock()
220            {
221                app.timer.update_seconds(time.0 as f64, &key);
222                app.timer.unlock(key);
223            }
224            if app.timer.animation_playing || app.timer.sync_to_real_time {
225                ui.ctx().request_repaint();
226            }
227            app.timer.march(delta_time.as_secs_f64());
228        })
229    });
230    egui::CentralPanel::default().frame(Frame::default()).show(ui, |ui| {
231        puffin::profile_scope!("tiles");
232        let mut tile_viewer = TileViewer { app };
233        ui_state.mus.tree.ui(&mut tile_viewer, ui);
234    });
235}
