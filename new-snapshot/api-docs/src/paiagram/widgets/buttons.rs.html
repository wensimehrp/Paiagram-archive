13pub(crate) fn triangle_button_shape(
14    painter: &mut Painter,
15    center: Pos2,
16    base: f32,
17    stroke: Stroke,
18    fill_color: Color32,
19) {
20    let dx = base / 4.0 * 3.0_f32.sqrt();
21    let a = Pos2 { x: center.x - dx, y: center.y - base / 2.0 };
22    let b = Pos2 { x: center.x + dx, y: center.y };
23    let c = Pos2 { x: a.x, y: center.y + base / 2.0 };
24    painter.add(Shape::convex_polygon(vec![a, b, c], fill_color, stroke));
25}

44pub(crate) fn dash_button_shape(
45    painter: &mut Painter,
46    center: Pos2,
47    base: f32,
48    stroke: Stroke,
49    fill_color: Color32,
50) {
51    let a =
52        Pos2 { x: center.x - base / 2.0 - base / 2.0 / 3.0_f32.sqrt(), y: center.y - base / 2.0 };
53    let b = Pos2 { x: a.x + base, y: a.y };
54    let c = Pos2 { x: center.x + base / 2.0, y: center.y };
55    let d = Pos2 { x: b.x, y: center.y + base / 2.0 };
56    let e = Pos2 { x: a.x, y: d.y };
57    let f = Pos2 { x: center.x - base / 2.0, y: center.y };
58    painter.add(Shape::convex_polygon(vec![a, b, c, f], fill_color, Stroke::NONE));
59    painter.add(Shape::convex_polygon(vec![f, c, d, e], fill_color, Stroke::NONE));
60    painter.line_segment([c, f], Stroke { width: 2.0, color: fill_color });
61    painter.add(Shape::closed_line(vec![a, b, c, d, e, f], stroke));
