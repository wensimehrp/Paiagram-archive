1use std::debug_assert_matches;

3use jiff::ZonedDateTime;
4use paiagram_core::time::TimetableTime;

6/// Token used to unlock the timer.
7#[derive(Debug)]
8pub(crate) struct TimerLockKey(());
9
10pub(crate) struct GlobalTimer {
11    /// Progress of the timer
12    seconds: f64,
13    key: Option<TimerLockKey>,
14    /// Ignored when synced to real time.
15    pub animation_speed: f32,
16    pub animation_playing: bool,
17    /// Whether to sync the timer with the current time.
18    pub sync_to_real_time: bool,
19}
20
21impl GlobalTimer {
22    pub fn new() -> Self {
23        Self {
24            seconds: TimetableTime::from_hms(8, 0, 0).0 as f64,
25            key: Some(TimerLockKey(())),
26            animation_speed: 1.0,
27            animation_playing: false,
28            sync_to_real_time: false,
29        }
30    }
31}
32
33impl GlobalTimer {
34    pub fn seconds(&self) -> f64 {
35        self.seconds
36    }
37
38    pub fn update_seconds(&mut self, new_seconds: f64, _key: &TimerLockKey) {
39        self.seconds = new_seconds
40    }
41
42    /// Acquire the lock and return a key used to release it.
43    pub fn try_lock(&mut self) -> Option<TimerLockKey> {
44        self.key.take()
45    }
46
47    /// Release the lock associated with `key`.
48    pub fn unlock(&mut self, key: TimerLockKey) {
49        debug_assert_matches!(self.key, None);
50        self.key = Some(key);
51    }
52
53    /// The current time of day as seconds in f64 since midnight.
54    pub fn current_real_time_seconds() -> f64 {
55        let now = ZonedDateTime::now();
56        let time = now.datetime().time();
57        time.hour() as f64 * 3600.0
58            + time.minute() as f64 * 60.0
59            + time.second() as f64
60            + time.subsec_nanosecond() as f64 / 1e9
61    }
