49impl super::Tile for RouteTimetableTile {
50    const NAME: &'static str = "Route timetable";
51    fn main_display(&mut self, app: &mut crate::App, ui: &mut Ui) {
52        puffin::profile_function!();
53        let Some(route) = app.routes.get(&self.route_key) else {
54            ui.centered_and_justified(|ui| ui.heading("Route does not exist"));
55            return;
56        };
57
58        ui.horizontal(|ui| {
59            puffin::profile_scope!("timetable_toolbar");
60            if ui.button("refresh cache").clicked() {
61                self.cache = route.intervals.populate_route_timetable(&app.world);
62            }
63            if ui.button("Export .csv").clicked() {
64                puffin::profile_scope!("export_csv");
65                RouteTimetableExport(self.cache.clone(), app.world.clone())
66                    .write_via_interface(app.file_write_tx.clone());
67            }
68            ui.checkbox(&mut self.show_estimates, "Show estimates");
69            ui.checkbox(&mut self.show_inbound, "Show Inbound");
70        });
71
72        let cell_size: Vec2 = vec2(50.0, 20.0);
73
74        // remove spacing between buttons
75        ui.spacing_mut().item_spacing = Vec2::ZERO;
76        // force floating to avoid bouncing ui when scrolled to the end
77        ui.style_mut().spacing.scroll.floating = true;
78        // Remove button background
79        ui.visuals_mut().widgets.inactive.weak_bg_fill = Color32::TRANSPARENT;
80
81        let (station_records, columns) = if self.show_inbound {
82            let records = self.cache.station_records.make_mut().iter_mut();
83            let iter = records.map(|(stn, inbound_mode, _)| (&*stn, inbound_mode));
84            (Either::Left(iter), self.cache.inbound_columns.as_slice())
85        } else {
86            let records = self.cache.station_records.make_mut().iter_mut().rev();
87            let iter = records.map(|(stn, _, outbound_mode)| (&*stn, outbound_mode));
88            (Either::Right(iter), self.cache.outbound_columns.as_slice())
89        };
90
91        Panel::left("left panel")
92            .frame(Frame::NONE)
93            .drag_to_open(true)
94            .default_size(cell_size.x * 2.0)
95            .show(ui, |ui| {
96                puffin::profile_scope!("station_header");
97                self.scroll = show_station_header(app, ui, cell_size, station_records, self.scroll);
98            });
99
100        Panel::top("top panel").frame(Frame::NONE).show(ui, |ui| {
101            puffin::profile_scope!("trip_header");
102            self.scroll = show_trip_header(app, ui, cell_size, columns, self.scroll);
103        });
104
105        let station_records = if self.show_inbound {
106            let iter = self.cache.station_records.iter().map(|(r, mode, _)| (r, mode));
107            Either::Left(iter)
108        } else {
109            let iter = self.cache.station_records.iter().rev().map(|(r, _, mode)| (r, mode));
110            Either::Right(iter)
111        };
112
113        CentralPanel::no_frame().show(ui, |ui| {
114            puffin::profile_scope!("timetable_grid");
115            let (scroll, paint_params) = show_grid(
116                self.show_estimates,
117                app,
118                ui,
119                cell_size,
120                station_records,
121                columns,
122                self.scroll,
123            );
124            self.scroll = scroll;
125            if let Some((pos, record)) = paint_params {
126                let stroke = Stroke::new(2.0, Color32::RED);
127                ui.painter().vline(pos.x, ui.clip_rect().y_range(), stroke);
128                ui.painter().hline(ui.clip_rect().x_range(), pos.y, stroke);
129                let name = record
130                    .station_key(app.graph.nodes())
131                    .and_then(|key| app.stations.get(key))
132                    .map_or_default(|wfc| wfc.name.as_str());
133                ui.painter().text(
134                    pos,
135                    Align2::LEFT_BOTTOM,
136                    name,
137                    FontId::proportional(13.0),
138                    ui.visuals().text_color(),
139                );
140            }
141        });
142    }

145/// Left panel: the station names and their per-row display modes.
146fn show_station_header<'a, 'world>(
147    app: &'world crate::App,
148    ui: &mut Ui,
149    cell_size: Vec2,
150    station_records: impl Iterator<Item = (&'a StationRecord, &'a mut StationDisplayMode)>,
151    scroll: Vec2,
152) -> Vec2 {
153    puffin::profile_function!();
154    let cell_size = vec2(ui.available_width(), cell_size.y);
155    ui.add_sized(cell_size, Label::new("Stations").truncate());
156    let scroll_area = ScrollArea::vertical()
157        .auto_shrink(false)
158        .vertical_scroll_offset(scroll.y)
159        .scroll_bar_visibility(scroll_area::ScrollBarVisibility::AlwaysHidden)
160        .show(ui, |ui| {
161            for (record, display_mode) in station_records {
162                let station_name = record
163                    .station_key(app.world.graph.nodes())
164                    .and_then(|key| app.world.stations.get(key))
165                    .map_or_default(|wfc| wfc.name.as_str());
166                let mut cell_size = cell_size;
167                cell_size.y *= display_mode.bits().count_ones().at_least(1) as f32;
168                let row_type_atoms = ContainerAtom::new(Atoms::from_iter(
169                    display_mode
170                        .iter()
171                        .map(|mode| match mode {
172                            StationDisplayMode::ARRIVAL => Atom::from("Ar."),
173                            StationDisplayMode::TRACK => Atom::from("Tr."),
174                            StationDisplayMode::DEPARTURE => Atom::from("De."),
175                            _ => Atom::from("??"),
176                        })
177                        .map(|atm| {
178                            atm.atom_grow(false).atom_shrink(false).atom_align(Align2::RIGHT_TOP)
179                        }),
180                ))
181                .align2(Align2::RIGHT_CENTER)
182                .direction(Direction::TopDown);
183                let mut button = Button::new(Atoms::from_iter(iter::chain(
184                    iter::once(
185                        Atom::from(station_name).atom_grow(true).atom_align(Align2::LEFT_CENTER),
186                    ),
187                    row_type_atoms.into_atoms(),
188                )))
189                .truncate()
190                .min_size(cell_size)
191                .atom_ui(ui);
192                button.set_intrinsic_size(cell_size);
193                Tooltip::for_enabled(&button).show(|ui| ui.label(station_name));
194                Popup::menu(&button).close_behavior(PopupCloseBehavior::CloseOnClickOutside).show(
195                    |ui| {
196                        let mut disp_bigflag = |flag: StationDisplayMode, atoms: &str| {
197                            let mut flag_bool = display_mode.contains(flag);
198                            // disallow removing if it is the only activated flag
199                            ui.add_enabled(
200                                !(display_mode.bits().count_ones() == 1 && flag_bool),
201                                Checkbox::new(&mut flag_bool, atoms),
202                            );
203                            display_mode.set(flag, flag_bool);
204                        };
205                        disp_bigflag(StationDisplayMode::ARRIVAL, "Arrival");
206                        disp_bigflag(StationDisplayMode::TRACK, "Track");
207                        disp_bigflag(StationDisplayMode::DEPARTURE, "Departure");
208                    },
209                );
210            }
211        });
212    Vec2 {
213        x: scroll.x,
214        y: scroll_area.state.clamped_offset().y,
215    }
216}

218/// Top panel: trip names, following the grid's horizontal scroll.
219fn show_trip_header(
220    app: &App,
221    ui: &mut Ui,
222    cell_size: Vec2,
223    columns: &[(TripKey, EcoVec<RouteTimetableEntry>)],
224    scroll: Vec2,
225) -> Vec2 {
226    puffin::profile_function!();
227    let scroll_area = ScrollArea::horizontal()
228        .auto_shrink([false, true])
229        .scroll_bar_visibility(scroll_area::ScrollBarVisibility::AlwaysHidden)
230        .horizontal_scroll_offset(scroll.x);
231    let scroll_area = show_grid_viewport(
232        scroll_area,
233        ui,
234        cell_size,
235        1,
236        columns.len(),
237        |ui, rect, _row_range, col_range| {
238            for (col_idx, (key, _)) in columns[col_range].iter().enumerate() {
239                let name = app.world.trips.get(key).map_or_default(|wfc| wfc.name.as_str());
240                let fill = app
241                    .world
242                    .trips
243                    .get(key)
244                    .and_then(|wfc| wfc.service_class)
245                    .and_then(|class_key| app.world.service_classes.get(&class_key))
246                    .map(|wfc| wfc.style.color)
247                    .unwrap_or(ui.visuals().faint_bg_color);
248                let cell_rect = Rect::from_min_size(
249                    pos2(rect.left() + col_idx as f32 * cell_size.x, rect.top()),
250                    cell_size,
251                );
252                ui.visuals_mut().widgets.inactive.corner_radius = 0.into();
253                ui.visuals_mut().widgets.hovered.corner_radius = 0.into();
254                ui.visuals_mut().widgets.active.corner_radius = 0.into();
255                ui.visuals_mut().widgets.inactive.weak_bg_fill = fill;
256                ui.visuals_mut().widgets.hovered.weak_bg_fill = fill.gamma_multiply(0.5);
257                ui.visuals_mut().widgets.active.weak_bg_fill = fill.gamma_multiply(0.25);
258                ui.spacing_mut().button_padding = Vec2::ZERO;
259                // ui.visuals_mut().widgets.inactive.bg_stroke;
260                ui.visuals_mut().widgets.hovered.bg_stroke = Stroke {
261                    width: 1.0,
262                    color: fill,
263                };
264                ui.visuals_mut().widgets.active.bg_stroke = Stroke {
265                    width: 1.0,
266                    color: fill.gamma_multiply(0.5),
267                };
268                let button = Button::new(RichText::new(name).color(if fill.intensity() > 0.5 {
269                    Color32::BLACK
270                } else {
271                    Color32::WHITE
272                }))
273                .truncate();
274
275                if ui
276                    .place(cell_rect, button)
277                    .on_hover_text(format!("Click to open the trip\n{name}"))
278                    .clicked()
279                {
280                    app.open_or_focus(TripTile::new(*key));
281                }
282            }
283        },
284    );
285    Vec2 {
286        x: scroll_area.state.clamped_offset().x,
287        y: scroll.y,
288    }
289}

291/// Center panel: the timetable grid itself.
292fn show_grid<'a>(
293    show_estimates: bool,
294    app: &crate::App,
295    ui: &mut Ui,
296    cell_size: Vec2,
297    station_records: impl Iterator<Item = (&'a StationRecord, &'a StationDisplayMode)> + Clone,
298    columns: &[(TripKey, EcoVec<RouteTimetableEntry>)],
299    scroll: Vec2,
300) -> (Vec2, Option<(Pos2, &'a StationRecord)>) {
301    puffin::profile_function!();
302    // window stroke for drawing strokes
303    let stroke = ui.visuals().window_stroke;
304
305    let scroll_area = ScrollArea::both()
306        .auto_shrink(false)
307        .vertical_scroll_offset(scroll.y)
308        .horizontal_scroll_offset(scroll.x)
309        .scroll_bar_visibility(ScrollBarVisibility::AlwaysVisible);
310
311    let row_count =
312        station_records.clone().map(|(_, it)| it.bits().count_ones().at_least(1) as usize).sum();
313
314    let scroll_area = show_grid_viewport(
315        scroll_area,
316        ui,
317        cell_size,
318        row_count,
319        columns.len(),
320        |ui, rect, row_range, col_range| {
321            puffin::profile_scope!("grid_cells");
322            for idx in station_records
323                .clone()
324                .flat_map(|(_, disp_mode)| disp_mode.iter())
325                .skip(row_range.start)
326                .take(row_range.clone().count())
327                .tuple_windows()
328                .enumerate()
329                .filter_map(|(idx, (prev, curr))| {
330                    use StationDisplayMode as Sdm;
331                    match [prev, curr] {
332                        [Sdm::ARRIVAL, Sdm::ARRIVAL] => false,
333                        [Sdm::ARRIVAL, Sdm::TRACK] => true,
334                        [Sdm::ARRIVAL, Sdm::DEPARTURE] => true,
335                        [Sdm::DEPARTURE, Sdm::ARRIVAL] => false,
336                        [Sdm::DEPARTURE, Sdm::TRACK] => true,
337                        [Sdm::DEPARTURE, Sdm::DEPARTURE] => false,
338                        [Sdm::TRACK, _] => true,
339                        _ => false,
340                    }
341                    .then_some(idx)
342                })
343            {
344                let y = rect.top() + (idx + 1) as f32 * cell_size.y;
345                ui.painter().hline(rect.x_range(), y, stroke);
346            }
347            let col_iter = columns[col_range].iter().enumerate();
348            col_iter.fold(None, |paint_pos, (col_idx, (_, section))| {
349                let left = rect.left() + col_idx as f32 * cell_size.x;
350                let entry_iter = izip!(
351                    station_records.clone(),
352                    chain!(
353                        section.iter().copied().map(Either::Left),
354                        iter::once(Either::Right(())),
355                        iter::repeat(Either::Left(RouteTimetableEntry::NotInOperation)),
356                    ),
357                )
358                .flat_map(|((station_record, display_mode), entry)| {
359                    display_mode.iter().map(move |disp_mode| (station_record, disp_mode, entry))
360                })
361                .scan(None, |disp_progress, (station_record, disp_mode, entry)| {
362                    let ret = if disp_mode == StationDisplayMode::TRACK
363                        && let Either::Left(RouteTimetableEntry::Active((_, entry))) = entry
364                    {
365                        let track_name = app
366                            .world
367                            .graph
368                            .nodes()
369                            .get(&entry.node_key())
370                            .map_or("?".into(), |wfc| wfc.name.as_str());
371                        Atom::from(track_name).atom_shrink(true)
372                    } else {
373                        let text_to_display = match entry {
374                            Either::Left(RouteTimetableEntry::Active((es, entry))) => {
375                                *disp_progress = Some(true);
376                                match [show_estimates, disp_mode == StationDisplayMode::ARRIVAL] {
377                                    [true, true] => es.arr.to_oud2_string(false),
378                                    [true, false] => es.dep.to_oud2_string(false),
379                                    [false, true] => entry.to_oud2_string_arr(false),
380                                    [false, false] => entry.to_oud2_string_dep(false),
381                                }
382                            }
383                            Either::Right(()) if *disp_progress == Some(true) => {
384                                *disp_progress = Some(false);
385                                "=====".into()
386                            }
387                            Either::Left(RouteTimetableEntry::NotInOperation) | _ => {
388                                if *disp_progress == Some(true) {
389                                    "||".into()
390                                } else {
391                                    "..".into()
392                                }
393                            }
394                        };
395                        Atom::from(
396                            WidgetText::Text(text_to_display)
397                                .text_style(TIMETABLTE_TEXT_STYLE.clone()),
398                        )
399                        .atom_align(Align2::RIGHT_CENTER)
400                    };
401                    Some((station_record, ret))
402                })
403                .skip(row_range.start)
404                .take(row_range.clone().count());
405
406                // offset by one cell
407                ui.painter().vline(left + cell_size.x, rect.y_range(), stroke);
408
409                entry_iter.enumerate().fold(paint_pos, |paint_pos, (row, (stn_record, atoms))| {
410                    let top = rect.top() + row as f32 * cell_size.y;
411                    let cell_rect = Rect::from_min_size(pos2(left, top), cell_size);
412                    ui.spacing_mut().button_padding = Vec2::ZERO;
413                    ui.place(cell_rect, WidgetAtom::new(atoms));
414                    paint_pos.or_else(|| {
415                        ui.input(|r| r.pointer.hover_pos())
416                            .map_or(false, |pos| cell_rect.contains(pos))
417                            .then_some((pos2(left + cell_size.x, top + cell_size.y), stn_record))
418                    })
419                })
420            })
421        },
422    );
423    (scroll_area.state.clamped_offset(), scroll_area.inner)
424}

426/// Like [`ScrollArea::show_viewport`], but specialized for a grid of uniformly-sized cells.
427///
428/// The callback only has to lay out the visible cells: it receives the screen-space rect
429/// covering the visible range together with the row and column index ranges that intersect the
430/// viewport.
431fn show_grid_viewport<R>(
432    scroll_area: ScrollArea,
433    ui: &mut Ui,
434    cell_size: Vec2,
435    total_rows: usize,
436    total_cols: usize,
437    add_contents: impl FnOnce(&mut Ui, Rect, Range<usize>, Range<usize>) -> R,
438) -> ScrollAreaOutput<R> {
439    puffin::profile_function!();
440    let row_pitch = cell_size.y;
441    let col_pitch = cell_size.x;
442
443    scroll_area.show_viewport(ui, |ui, viewport| {
444        ui.set_height(row_pitch * total_rows as f32);
445        ui.set_width(col_pitch * total_cols as f32);
446
447        let row_range = visible_range(viewport.y_range(), row_pitch, total_rows);
448        let col_range = visible_range(viewport.x_range(), col_pitch, total_cols);
449
450        let origin = ui.max_rect().min;
451        let rect = Rect::from_min_max(
452            origin
453                + vec2(
454                    col_range.start as f32 * col_pitch,
455                    row_range.start as f32 * row_pitch,
456                ),
457            origin
458                + vec2(
459                    col_range.end as f32 * col_pitch,
460                    row_range.end as f32 * row_pitch,
461                ),
462        );
463
464        ui.scope_builder(UiBuilder::new().max_rect(rect), |ui| {
465            add_contents(ui, rect, row_range, col_range)
466        })
467        .inner
468    })
469}

471/// The indices intersecting `viewport` for evenly spaced cells of the given pitch, padded by one
472/// cell on either side so that partially visible cells are included.
473fn visible_range(viewport: Rangef, pitch: f32, total: usize) -> Range<usize> {
474    let mut min = (viewport.min / pitch).floor().max(0.0) as usize;
475    let mut max = (viewport.max / pitch).ceil() as usize + 1;
476    if max > total {
477        let diff = max.saturating_sub(min);
478        max = total;
479        min = total.saturating_sub(diff);
480    }
481    min..max
