45impl Tile for MapTile {
46    const NAME: &'static str = "Map";
47    fn main_display(&mut self, app: &mut App, ui: &mut Ui) {
48        puffin::profile_function!();
49        Frame::canvas(ui.style())
50            .inner_margin(Margin::ZERO)
51            .outer_margin(Margin::ZERO)

59/// MapPlugin runs before [`super::select_content::SelectPlugin`]
60pub struct MapPlugin<'a> {
61    /// Closure to work around with lifetimes
62    pub app: &'a mut App,
63    pub trip_icon: TextureId,
64}
65
66impl<'a> walkers::Plugin for MapPlugin<'a> {
67    fn run(
68        self: Box<Self>,
69        ui: &mut Ui,
70        response: &Response,
71        projector: &Projector,
72        _map_memory: &MapMemory,
73    ) {
74        let MapPlugin { app, trip_icon } = *self;
75        map_content::render_map_contents(app, trip_icon, ui, response, projector);
76        display_selected_content::display_selected_content(app, ui, response, projector);
77        select_content::select_map_contents(app, ui, response, projector);
78    }
79}

81fn main_display(tile: &mut MapTile, app: &mut App, ui: &mut Ui) {
82    puffin::profile_function!();
83    let viewport = ui.available_size();
84    let tile_window = Window::new("foo bar")
85        .id(ui.scope_id().with("foo bar"))
86        .auto_sized()
87        .title_bar(false)
88        .fixed_pos(ui.clip_rect().shrink(8.0).right_top())
89        .pivot(Align2::RIGHT_TOP);
90    let mut coord_type = app.world().graph.coord_type;
91    tile_window.show(ui.ctx(), |ui| {
92        if ui.button("Auto layout").clicked() {
93            app.auto_layout();
94            // Coordinates changed, so refit the viewport on the next frame.
95            tile.map_memory = None;
96        }
97        if ui.button("Center viewport").clicked()
98            && let Some(map_memory) = tile.map_memory.as_mut()
99        {
100            *map_memory = viewport::fit_viewport_to_nodes(app, viewport);
101        }
102        ComboBox::from_id_salt("coordinate type and projection")
103            .selected_text(coord_type.name())
104            .truncate()
105            .show_ui(ui, |ui| {
106                for option in CoordType::ALL {
107                    if ui.selectable_value(&mut coord_type, *option, option.name()).changed() {
108                        let _ = app.mutate(|mut world| {
109                            world.graph.coord_type = *option;
110                            Ok(world)
111                        });
112                        // Both the tile source and the fitted viewport follow the projection.
113                        tile.tiles = None;
114                        tile.map_memory = None;
115                    }
116                }
117            });
118        ui.label({
119            let zoom = tile.map_memory.as_ref().map_or(0.0, |mem| mem.zoom());
120            format!("{:.3}", zoom)
121        });
122    });
123    let coord_type = app.world().graph.coord_type;
124    let is_autonavi = matches!(coord_type, CoordType::Wgs(WgsProjection::GcjAndMercator));
125    if tile.tiles.is_none() {
126        let ctx = ui.ctx().clone();
127        tile.tiles = Some(if is_autonavi {
128            HttpTiles::new(AutoNavi, ctx)
129        } else {
130            HttpTiles::new(OpenStreetMap, ctx)
131        });
132    }
133    let map_memory =
134        tile.map_memory.get_or_insert_with(|| viewport::fit_viewport_to_nodes(app, viewport));
135    let trip_icon =
136        tile.trip_icon.get_or_insert_with(|| map_content::load_trip_icon(ui.ctx())).id();
137    let projection = XyProjection;
138    let map = Map::new(
139        match tile.tiles.as_mut() {
140            // map::new wants dyn, thus rebind
141            Some(tiles) => Some(tiles),
142            None => None,
143        },
144        map_memory,
145        XyProjection::to_position(coord_type.project(Coord::from_wgs84(-123.1139, 49.2612))),
146    )
147    .with_projection(&projection)
148    .with_plugin(MapPlugin { app, trip_icon });
149    ui.add(map);
150}
