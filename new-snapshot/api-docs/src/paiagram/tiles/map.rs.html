55/// MapPlugin runs before [`super::select_content::SelectPlugin`]
56pub struct MapPlugin<'a> {
57    /// Closure to work around with lifetimes
58    pub app: &'a mut App,
59    pub trip_icon: TextureId,
60}
61
62impl<'a> walkers::Plugin for MapPlugin<'a> {
63    fn run(
64        self: Box<Self>,
65        ui: &mut Ui,
66        response: &Response,
67        projector: &Projector,
68        map_memory: &MapMemory,
69    ) {
70        let MapPlugin { app, trip_icon } = *self;
71        map_content::render_map_contents(app, trip_icon, ui, response, projector, map_memory);
72        select_content::select_map_contents(app, ui, response, projector, map_memory);
73    }
74}

76fn main_display(tile: &mut MapTile, app: &mut App, ui: &mut Ui) {
77    puffin::profile_function!();
78    let viewport = ui.available_size();
79    let tile_window = Window::new("foo bar")
80        .id(ui.scope_id().with("foo bar"))
81        .auto_sized()
82        .title_bar(false)
83        .fixed_pos(ui.clip_rect().shrink(8.0).right_top())
84        .pivot(Align2::RIGHT_TOP);
85    tile_window.show(ui.ctx(), |ui| {
86        if ui.button("Auto layout").clicked() {
87            app.auto_layout();
88            // Coordinates changed, so refit the viewport on the next frame.
89            tile.map_memory = None;
90        }
91        if ui.button("Center viewport").clicked()
92            && let Some(map_memory) = tile.map_memory.as_mut()
93        {
94            *map_memory = viewport::fit_viewport_to_nodes(app, tile.is_autonavi, viewport);
95        }
96        if ui.checkbox(&mut tile.is_autonavi, "使用高德地图").changed() || tile.tiles.is_none()
97        {
98            let ctx = ui.ctx().clone();
99            tile.tiles = Some(if tile.is_autonavi {
100                HttpTiles::new(AutoNavi, ctx)
101            } else {
102                HttpTiles::new(OpenStreetMap, ctx)
103            });
104        }
105        ui.label({
106            let zoom = tile.map_memory.as_ref().map_or(0.0, |mem| mem.zoom());
107            format!("{:.3}", zoom)
108        });
109    });
110    let map_memory = tile
111        .map_memory
112        .get_or_insert_with(|| viewport::fit_viewport_to_nodes(&app, tile.is_autonavi, viewport));
113    let trip_icon =
114        tile.trip_icon.get_or_insert_with(|| map_content::load_trip_icon(ui.ctx())).id();
115    let map = Map::new(
116        match tile.tiles.as_mut() {
117            // map::new wants dyn, thus rebind
118            Some(tiles) => Some(tiles),
119            None => None,
120        },
121        map_memory,
122        lon_lat(0.0, 0.0),
123    )
124    .with_projection(if tile.is_autonavi {
125        &MercatorProjectionGcj02
126    } else {
127        &MercatorProjection
128    })
129    .with_plugin(MapPlugin { app, trip_icon });
130    ui.add(map);
131}
