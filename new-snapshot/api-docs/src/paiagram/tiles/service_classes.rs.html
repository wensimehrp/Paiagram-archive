24impl Tile for ServiceClassesTile {
25    const NAME: &'static str = "Service Classes";
26    fn main_display(&mut self, app: &mut App, ui: &mut Ui) {
27        puffin::profile_function!();
28        let mut class_trip_count = ServiceClassKeyHashMap::default();
29        for (_, trip) in &app.trips {
30            let Some(service_class_key) = trip.service_class else {
31                return;
32            };
33            *class_trip_count.entry(service_class_key).or_insert(0usize) += 1usize;
34        }
35        Panel::left("service classes left panel").show(ui, |ui| {
36            puffin::profile_scope!("service_class_list");
37            let spacing = ui.spacing().interact_size.y;
38            let rows = app.service_classes.len();
39            let scroll_area = ScrollArea::vertical().auto_shrink(false);
40            scroll_area.show_rows(ui, spacing, rows, |ui, row_range| {
41                let entries =
42                    (&app.service_classes).iter().skip(row_range.start).take(row_range.count());
43                let show_entries = |ui: &mut Ui, selected| {
44                    entries.fold(selected, |selected, (&sc_key, service_class)| {
45                        let (stroke_atom, name_atom) = service_class_button_atoms(service_class);
46                        Button::selectable(
47                            selected == Some(sc_key),
48                            (
49                                stroke_atom,
50                                name_atom.atom_grow(true).atom_shrink(true),
51                                class_trip_count
52                                    .get(&sc_key)
53                                    .copied()
54                                    .unwrap_or(0)
55                                    .to_string()
56                                    .atom_align(Align2::RIGHT_CENTER),
57                            ),
58                        )
59                        .truncate()
60                        .atom_ui(ui)
61                        .clicked()
62                        .then_some(sc_key)
63                        .or(selected)
64                    })
65                };
66                self.selected_class = ui
67                    .with_layout(Layout::top_down_justified(egui::Align::Min), |ui| {
68                        show_entries(ui, self.selected_class)
69                    })
70                    .inner;
71            });
72        });
73        CentralPanel::default_margins().show(ui, |ui| {
74            puffin::profile_scope!("service_class_details");
75            let Some(selected_class_key) = self.selected_class else {
76                ui.centered_and_justified(|ui| ui.heading("Nothing focused"));
77                return;
78            };
79            let Some(selected_class) = app.service_classes.get(&selected_class_key) else {
80                ui.centered_and_justified(|ui| ui.heading("Service class does not exist"));
81                return;
82            };
83            ui.heading(selected_class.name.as_str());
84            Grid::new(selected_class.name.as_str()).num_columns(2).show(ui, |ui| {
85                ui.label("Creation time");
86                let time_string = jiff::Zoned::try_from(selected_class_key.creation_time())
87                    .map_or("Unknown".into(), |zoned| {
88                        format!("{}", zoned.strftime("%F %T %:z"))
89                    });
90                ui.label(time_string);
91                ui.end_row();
92            });
93            let mut class_trips: Vec<_> = app
94                .trips
95                .iter()
96                .filter(|(_, trip)| trip.service_class == Some(selected_class_key))
97                .filter_map(|(&trip_key, trip)| {
98                    if let Some(entry) = trip.schedule.entries().first()
99                        && let Some(t) = match (entry.arr_or_pass, entry.dep) {
100                            (TravelMode::At(t), _) => Some(t),
101                            (_, Some(TravelMode::At(t))) => Some(t),
102                            _ => None,
103                        }
104                    {
105                        Some((trip_key, trip.name.as_str(), t))
106                    } else {
107                        None
108                    }
109                })
110                .collect();
111            class_trips.sort_unstable_by_key(|(_, _, t)| *t);
112
113            // ugly solution, but interesting
114            let scroll_area = ScrollArea::vertical().show(ui, |ui| {
115                let chunks = class_trips.chunk_by(|(_, _, a), (_, _, b)| a.hour() == b.hour());
116                chunks.fold(None, |acc, chunk| {
117                    let Some(first) = chunk.first() else {
118                        return acc;
119                    };
120                    ui.horizontal(|ui| {
121                        ui.label(format!("{}:00", first.2.hour()));
122                        ui.add(Separator::default().horizontal());
123                    });
124                    ui.horizontal_wrapped(|ui| {
125                        chunk.iter().fold(acc, |acc, &(trip_key, name, time)| {
126                            let resp = ui.button(name).clicked();
127                            acc.or(resp.then(|| TripTile::new(trip_key)))
128                        })
129                    })
130                    .inner
131                })
132            });
133            if let Some(tile) = scroll_area.inner {
134                app.open_or_focus(tile);
135            }
136        });
137    }
138}

140pub fn service_class_button_atoms<'a>(
141    service_class: &'a Wfc<ServiceClass, ServiceClassCache>,
142) -> (Atom<'a>, Atom<'a>) {
143    let service_class_str = service_class.name.as_str();
144    let stroke_atom = Atom::paint([35.0, 15.0], |ui, args| {
145        let rect = args.rect;
146        let stroke = Stroke::new(
147            service_class.style.width as f32 * 1.0,
148            service_class.style.color,
149        );
150        ui.painter().line_segment([rect.left_center(), rect.right_center()], stroke);
151        let stroke = Stroke::new(2.0, service_class.style.color);
152        ui.painter().line_segment([rect.left_top(), rect.left_bottom()], stroke);
153    })
154    .atom_grow(false)
155    .atom_shrink(false)
156    .atom_align(Align2::LEFT_CENTER);
157    let class_name_atom = Atom::from(service_class_str)
158        .atom_grow(true)
159        .atom_shrink(true)
160        .atom_align(Align2::LEFT_CENTER);
161    (stroke_atom, class_name_atom)
162}
