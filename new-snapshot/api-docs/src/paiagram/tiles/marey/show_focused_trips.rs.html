19pub fn show_focused_trip<'a>(
20    trip_schedule: &TripSchedule,
21    stroke: Stroke,
22    source: &mut Source,
23    animation_progress: f32,
24    ui: &mut Ui,
25    tile: &mut super::MareyTile,
26    station_heights: &[CanvasLength],
27    painter: &mut Painter,
28    polylines: TripBucket,
29    trip_key: TripKey,
30    estimate_base: Option<TEstimate>,
31) -> Option<(TEntryId, TEstimate, Either<TDuration, TDuration>)> {
32    puffin::profile_function!();
33    let text_color = ui.visuals().text_color().gamma_multiply(animation_progress);
34
35    polylines.iter().fold(None, |acc, polyline| {
36        let points = draw_polyline(&tile.navi, polyline, station_heights, painter, stroke);
37        let ret = show_trip_handles(
38            trip_schedule,
39            tile,
40            source,
41            polyline,
42            &points,
43            text_color,
44            ui,
45            painter,
46            trip_key,
47            estimate_base,
48        );
49        acc.or(ret)
50    })
51}

53/// Draws the path of one trip, without any labels or interactive handles.
54///
55/// Returns the screen-space positions of the arrival/departure pairs, so that
56/// [`show_trip_handles`] can place its handles on top of the line.
57pub fn draw_polyline(
58    navi: &MareyTileNavigation,
59    polyline: &[BucketPoint],
60    station_heights: &[CanvasLength],
61    painter: &mut Painter,
62    stroke: Stroke,
63) -> Vec<Pos2> {
64    puffin::profile_scope!("trip line");
65    let points: Vec<_> = polyline
66        .iter()
67        .flat_map(|&(estimate, _entry, idx, progress)| {
68            let x1 = navi.logical_x_to_screen_x(estimate.arr.to_ticks());
69            let x2 = navi.logical_x_to_screen_x(estimate.dep.to_ticks());
70            let idx = idx as usize;
71            let curr_y = navi.logical_y_to_screen_y(station_heights[idx]);
72            let next_y = navi.logical_y_to_screen_y(station_heights[idx + 1]);
73            let y = lerp(curr_y..=next_y, progress);
74            [Pos2::new(x1, y), Pos2::new(x2, y)]
75        })
76        .collect();
77    painter.line(points.clone(), stroke);
78    points
79}

81/// Draws the arrival/departure labels and the interactive handles of one trip.
82///
83/// Returns the time patch produced by dragging a handle, if any.
84fn show_trip_handles(
85    trip_schedule: &TripSchedule,
86    tile: &super::MareyTile,
87    source: &mut Source,
88    polyline: &[BucketPoint],
89    points: &[Pos2],
90    text_color: Color32,
91    ui: &mut Ui,
92    mut painter: &mut Painter,
93    trip_key: TripKey,
94    estimate_base: Option<TEstimate>,
95) -> Option<(TEntryId, TEstimate, Either<TDuration, TDuration>)> {
96    puffin::profile_scope!("trip handles");
97    let (points, _) = points.as_chunks::<2>();
98    let handle_iter = iter::zip(
99        polyline,
100        chain!(iter::once(None), points.iter().map(Some), iter::once(None)).tuple_windows(),
101    );
102
103    handle_iter.fold(None, |ret, (a, b)| {
104        let (estimate_visual, entry, ..) = a;
105        let (prev, curr, next) = b;
106        let estimate_entry = estimate_base.unwrap_or(*estimate_visual);
107        let &[pos_curr_arr, pos_curr_dep] = curr.unwrap();
108        let pos_prev_dep = prev.map_or(pos_curr_arr, |&[_, p]| p);
109        let pos_next_arr = next.map_or(pos_curr_dep, |&[p, _]| p);
110        // arrival text
111        let ((arr_align, arr_direction), (dep_align, dep_direction)) =
112            if estimate_visual.arr == estimate_visual.dep {
113                let (arr_align, arr_direction) =
114                    label_placement(pos_prev_dep, pos_curr_dep, pos_next_arr);
115                (
116                    (arr_align, arr_direction),
117                    (arr_align.flipped(), -arr_direction),
118                )
119            } else {
120                let (dep_align, dep_direction) =
121                    label_placement(pos_curr_arr, pos_curr_dep, pos_next_arr);
122                (
123                    label_placement(pos_prev_dep, pos_curr_arr, pos_curr_dep),
124                    (dep_align.flipped(), -dep_direction),
125                )
126            };
127
128        painter.text(
129            pos_curr_arr + arr_direction * 10.0,
130            arr_align,
131            estimate_visual.arr,
132            TIMETABLE_FONT_ID.clone(),
133            ui.visuals().strong_text_color(),
134        );
135
136        if estimate_visual.arr != estimate_visual.dep {
137            painter.text(
138                pos_curr_dep + dep_direction * 10.0,
139                dep_align,
140                estimate_visual.dep,
141                TIMETABLE_FONT_ID.clone(),
142                ui.visuals().strong_text_color(),
143            );
144        }
145
146        let EstimateEntry::Pinned(entry) = entry else {
147            return ret;
148        };
149
150        let button_radius = 8.0;
151        let circle_handle_size: f32 = 7.0 / 12.0 * button_radius * 2.0;
152        let triangle_handle_size: f32 = 10.0 / 12.0 * button_radius * 2.0;
153        let dash_handle_size: f32 = 9.0 / 12.0 * button_radius * 2.0;
154        let raw_delta = (button_radius * 2.0)
155            - (pos_curr_dep.x - pos_curr_arr.x).clamp(-button_radius * 2.0, button_radius * 2.0);
156        let delta_x = (raw_delta / 2.0) * 0.8;
157        let extreme_bg_color = ui.visuals().extreme_bg_color;
158        let bg_stroke = if ui.visuals().dark_mode {
159            Stroke::new(1.0, ui.visuals().text_color())
160        } else {
161            Stroke::new(2.5, Color32::BLACK)
162        };
163        let style = ui.style_mut();
164        style.visuals.widgets.inactive = WidgetVisuals {
165            bg_fill: extreme_bg_color,
166            bg_stroke,
167            ..style.visuals.widgets.inactive
168        };
169        style.visuals.widgets.hovered = WidgetVisuals {
170            bg_fill: extreme_bg_color.lerp_to_gamma(text_color, 0.25),
171            bg_stroke,
172            ..style.visuals.widgets.inactive
173        };
174        style.visuals.widgets.active = WidgetVisuals {
175            bg_fill: extreme_bg_color.lerp_to_gamma(text_color, 0.5),
176            bg_stroke,
177            ..style.visuals.widgets.inactive
178        };
179        let ret = {
180            let arr_pos = pos2(pos_curr_arr.x - delta_x, pos_curr_arr.y);
181            let curr_arr_response = ui.allocate_rect(
182                Rect::from_pos(arr_pos).expand(button_radius),
183                Sense::click_and_drag(),
184            );
185            let arr_visuals = ui.style().interact(&curr_arr_response);
186            let (paint_fn, handle_size): (fn(&mut Painter, Pos2, f32, Stroke, Color32) -> (), f32) =
187                match entry.arr_or_pass {
188                    TravelMode::At(..) => (buttons::circle_button_shape, circle_handle_size),
189                    TravelMode::For(..) => (buttons::dash_button_shape, dash_handle_size),
190                    TravelMode::Flexible => (buttons::triangle_button_shape, triangle_handle_size),
191                };
192            paint_fn(
193                &mut painter,
194                curr_arr_response.rect.center(),
195                handle_size,
196                arr_visuals.bg_stroke,
197                arr_visuals.bg_fill,
198            );
199            let delta_t = ui.input(|r| r.pointer.latest_pos()).map_or(TDuration::ZERO, |p| {
200                tile.navi.screen_x_to_logical_x(p.x).to_timetable_time() - estimate_entry.arr
201            });
202            if curr_arr_response.drag_stopped() {
203                let _ = source.mutate(|mut world| {
204                    let trip = world.trips.get_mut(&trip_key).ok_or("wtf")?;
205                    trip.schedule.shift_arr_or_pass(entry.id, delta_t);
206                    Ok(world)
207                });
208            }
209            Popup::menu(&curr_arr_response).align(RectAlign::outside(arr_align.flipped())).show(
210                |ui| {
211                    let [show_at, show_for, show_flexible] = [
212                        !matches!(entry.arr_or_pass, TravelMode::At(_)),
213                        !matches!(entry.arr_or_pass, TravelMode::For(_)),
214                        !matches!(entry.arr_or_pass, TravelMode::Flexible),
215                    ];
216                    let btn = Button::new("At").right_text(
217                        RichText::from(format!("{}", estimate_visual.arr))
218                            .font(TIMETABLE_FONT_ID.clone()),
219                    );
220                    if ui.add_enabled(show_at, btn).clicked() {
221                        let _ = source.mutate(|mut world| {
222                            let trip = world.trips.get_mut(&trip_key).ok_or("wtf")?;
223                            trip.schedule.set_arr_or_pass_mode(
224                                entry.id,
225                                TravelMode::At(estimate_visual.arr),
226                            );
227                            Ok(world)
228                        });
229                    }
230                    let duration =
231                        trip_schedule.estimates(&source.graph, |estimates| {
232                            trip_schedule.dur_since_prev_stable(estimates, entry.id)
233                        });
234                    let btn = Button::new("For").right_text(
235                        RichText::from(
236                            duration.map_or("--:--:--".into(), |d| d.to_string_no_arrow()),
237                        )
238                        .font(TIMETABLE_FONT_ID.clone()),
239                    );
240                    if ui.add_enabled(show_for, btn).clicked() {
241                        let _ = source.mutate(|mut world| {
242                            let trip = world.trips.get_mut(&trip_key).ok_or("wtf")?;
243                            trip.schedule.set_arr_or_pass_mode(
244                                entry.id,
245                                TravelMode::For(duration.unwrap_or_default()),
246                            );
247                            Ok(world)
248                        });
249                    }
250                    if ui.add_enabled(show_flexible, Button::new("Flexible")).clicked() {
251                        let _ = source.mutate(|mut world| {
252                            let trip = world.trips.get_mut(&trip_key).ok_or("wtf")?;
253                            trip.schedule.set_arr_or_pass_mode(entry.id, Flexible);
254                            Ok(world)
255                        });
256                    }
257                },
258            );
259            ret.or(curr_arr_response.dragged().then_some((
260                entry.id,
261                estimate_entry,
262                Either::Left(delta_t),
263            )))
264        };
265        {
266            let dep_pos = pos2(pos_curr_dep.x + delta_x, pos_curr_dep.y);
267            let curr_dep_response = ui.allocate_rect(
268                Rect::from_pos(dep_pos).expand(button_radius),
269                Sense::click_and_drag(),
270            );
271            let dep_visuals = ui.style().interact(&curr_dep_response);
272            let (paint_fn, handle_size): (fn(&mut Painter, Pos2, f32, Stroke, Color32) -> (), f32) =
273                match entry.dep {
274                    None => (buttons::double_triangle, dash_handle_size),
275                    Some(TravelMode::At(..)) => (buttons::circle_button_shape, circle_handle_size),
276                    Some(TravelMode::For(..)) => (buttons::dash_button_shape, dash_handle_size),
277                    Some(TravelMode::Flexible) => {
278                        (buttons::triangle_button_shape, triangle_handle_size)
279                    }
280                };
281            paint_fn(
282                &mut painter,
283                curr_dep_response.rect.center(),
284                handle_size,
285                dep_visuals.bg_stroke,
286                dep_visuals.bg_fill,
287            );
288            let delta_t = ui.input(|r| r.pointer.latest_pos()).map_or(TDuration::ZERO, |p| {
289                tile.navi.screen_x_to_logical_x(p.x).to_timetable_time() - estimate_entry.dep
290            });
291            if curr_dep_response.drag_stopped() {
292                let _ = source.mutate(|mut world| {
293                    let trip = world.trips.get_mut(&trip_key).ok_or("wtf")?;
294                    trip.schedule.shift_dep(entry.id, delta_t);
295                    Ok(world)
296                });
297            }
298            Popup::menu(&curr_dep_response).align(RectAlign::outside(arr_align.flipped())).show(
299                |ui| {
300                    let [show_at, show_for, show_flexible, show_non_stop] = [
301                        !matches!(entry.dep, Some(TravelMode::At(_))),
302                        !matches!(entry.dep, Some(TravelMode::For(_))),
303                        !matches!(entry.dep, Some(TravelMode::Flexible)),
304                        !matches!(entry.dep, None),
305                    ];
306                    let btn = Button::new("At").right_text(
307                        RichText::from(format!("{}", estimate_visual.dep))
308                            .font(TIMETABLE_FONT_ID.clone()),
309                    );
310                    if ui.add_enabled(show_at, btn).clicked() {
311                        let _ = source.mutate(|mut world| {
312                            let trip = world.trips.get_mut(&trip_key).ok_or("wtf")?;
313                            trip.schedule
314                                .set_dep_mode(entry.id, Some(TravelMode::At(estimate_visual.dep)));
315                            Ok(world)
316                        });
317                    }
318                    let duration = estimate_visual.duration();
319                    let btn = Button::new("For").right_text(
320                        RichText::from(duration.to_string_no_arrow())
321                            .font(TIMETABLE_FONT_ID.clone()),
322                    );
323                    if ui.add_enabled(show_for, btn).clicked() {
324                        let _ = source.mutate(|mut world| {
325                            let trip = world.trips.get_mut(&trip_key).ok_or("wtf")?;
326                            trip.schedule.set_dep_mode(entry.id, Some(TravelMode::For(duration)));
327                            Ok(world)
328                        });
329                    }
330                    if ui.add_enabled(show_flexible, Button::new("Flexible")).clicked() {
331                        let _ = source.mutate(|mut world| {
332                            let trip = world.trips.get_mut(&trip_key).ok_or("wtf")?;
333                            trip.schedule.set_dep_mode(entry.id, Some(Flexible));
334                            Ok(world)
335                        });
336                    }
337                    if ui.add_enabled(show_non_stop, Button::new("Non-Stop")).clicked() {
338                        let _ = source.mutate(|mut world| {
339                            let trip = world.trips.get_mut(&trip_key).ok_or("wtf")?;
340                            trip.schedule.set_dep_mode(entry.id, None);
341                            Ok(world)
342                        });
343                    }
344                },
345            );
346            ret.or(curr_dep_response.dragged().then_some((
347                entry.id,
348                estimate_entry,
349                Either::Right(delta_t),
350            )))
351        }
352    })
