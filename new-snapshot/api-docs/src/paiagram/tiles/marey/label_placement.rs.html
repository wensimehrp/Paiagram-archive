3pub fn label_placement(p0: Pos2, p1: Pos2, p2: Pos2) -> (Align2, Vec2) {
4    let dir0 = p0 - p1;
5    let dir1 = p2 - p1;
6
7    let len_sq0 = dir0.length_sq();
8    let len_sq1 = dir1.length_sq();
9
10    let offset_dir = if len_sq0 < 1e-6 && len_sq1 < 1e-6 {
11        Vec2::LEFT
12    } else if len_sq0 < 1e-6 {
13        -dir1.normalized()
14    } else if len_sq1 < 1e-6 {
15        -dir0.normalized()
16    } else {
17        let v0 = dir0 / len_sq0.sqrt();
18        let v1 = dir1 / len_sq1.sqrt();
19        let sum = v0 + v1;
20
21        let raw_offset = if sum.length_sq() < 1e-6 {
22            v0.rot90()
23        } else {
24            -sum.normalized()
25        };
26
27        let raw_travel = p2 - p0;
28        let travel_dir = if raw_travel.y > 0.0 || (raw_travel.y == 0.0 && raw_travel.x < 0.0) {
29            -raw_travel
30        } else {
31            raw_travel
32        };
33
34        let cross = travel_dir.x * raw_offset.y - travel_dir.y * raw_offset.x;
35        if cross >= f32::EPSILON {
36            -raw_offset
37        } else {
38            raw_offset
39        }
40    };
41
42    let deg = offset_dir.angle().to_degrees().rem_euclid(360.0).round() as u16;
43
44    let align = match deg {
45        0..=5 | 355..=359 => Align2::LEFT_CENTER, // Right (+x)
46        6..=84 => Align2::LEFT_TOP,               // Down-Right (+x, +y)
47        85..=95 => Align2::CENTER_TOP,            // Down (+y)
48        96..=174 => Align2::RIGHT_TOP,            // Down-Left (-x, +y)
49        175..=185 => Align2::RIGHT_CENTER,        // Left (-x)
50        186..=264 => Align2::RIGHT_BOTTOM,        // Up-Left (-x, -y)
51        265..=275 => Align2::CENTER_BOTTOM,       // Up (-y)
52        _ => Align2::LEFT_BOTTOM,                 // Up-Right (+x, -y)
53    };
54
55    (align, offset_dir)
56}
