2//! Tile provider module for AutoNavi (Amap, or 高德)
3
4use walkers::TileId;
5use walkers::sources::{Attribution, TileSource};
6
7pub struct AutoNavi;
8
9impl TileSource for AutoNavi {
10    fn tile_url(&self, tile_id: TileId) -> String {
11        let z = tile_id.zoom;
12        let x = tile_id.x;
13        let y = tile_id.y;
14
15        // Cycle through subdomains 01-04 based on the x coordinate
16        let subdomain = (x % 4) + 1;
17
18        // style=8 is the standard road map
19        format!(
20            "https://webrd0{subdomain}.is.autonavi.com/appmaptile?lang=zh_cn&size=1&scale=1&style=8&x={x}&y={y}&z={z}"
21        )
22    }
23    fn attribution(&self) -> Attribution {
24        Attribution {
25            text: "AutoNavi (Amap)",
26            url: "https://www.amap.com/",
27            logo_dark: None,
28            logo_light: None,
29        }
30    }
31    fn max_zoom(&self) -> u8 {
32        18
33    }
34}
