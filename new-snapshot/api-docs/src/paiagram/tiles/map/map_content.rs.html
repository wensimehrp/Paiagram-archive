17/// Decodes the embedded trip icon and uploads it as an egui texture. Every trip uses the same
18/// texture for now, so egui batches all icon quads into one draw call.
19pub fn load_trip_icon(ctx: &Context) -> TextureHandle {
20    let decoded = image::load_from_memory(include_bytes!("../../../../../assets/raicho_485.png"))
21        .expect("the embedded trip icon must be a valid PNG")
22        .into_rgba8();
23
24    // The icon is drawn from one square quad, so pad it with transparent pixels instead of
25    // stretching it. This preserves the original aspect ratio.
26    let side = decoded.width().max(decoded.height());
27    let mut rgba = image::RgbaImage::new(side, side);
28    let x = i64::from((side - decoded.width()) / 2);
29    let y = i64::from((side - decoded.height()) / 2);
30    image::imageops::overlay(&mut rgba, &decoded, x, y);
31
32    let image = ColorImage::from_rgba_unmultiplied([side as usize, side as usize], rgba.as_raw());
33    ctx.load_texture("graph_trip_icon", image, TextureOptions::LINEAR)
34}

36pub fn render_map_contents(
37    app: &App,
38    trip_icon: TextureId,
39    ui: &mut Ui,
40    response: &Response,
41    projector: &Projector,
42) {
43    puffin::profile_function!();
44
45    const DISPLAY_ZOOM_THRESHOLD: f64 = 8.0;
46    const DISPLAY_ZOOM_HARD_THRESHOLD: f64 = 7.0;
47    let strength = ui.animate_bool_responsive(
48        ui.scope_id().with("text display animation"),
49        projector.zoom() > DISPLAY_ZOOM_THRESHOLD,
50    );
51    let strength = if projector.zoom() > DISPLAY_ZOOM_HARD_THRESHOLD {
52        strength
53    } else {
54        0.0
55    };
56
57    let pos_to_xy = |pos: Pos2| XyProjection::pos_to_xy(projector, pos);
58    let xy_to_pos = |xy: XyPos| XyProjection::xy_to_pos(projector, xy);
59
60    let xy_min = pos_to_xy(response.rect.left_top());
61    let xy_max = pos_to_xy(response.rect.right_bottom());
62
63    // Intervals
64    let mut callback = GraphCallback::new();
65    {
66        puffin::profile_scope!("map_intervals");
67        callback.populate_uniforms(response.rect);
68        for IntervalRTreeObject {
69            interval_key: _,
70            nodes,
71        } in app.spatial().intervals.get(xy_min, xy_max)
72        {
73            let Some(((last, _), rest)) = nodes.split_last() else {
74                continue;
75            };
76            callback.interval_points.extend(
77                rest.iter()
78                    .copied()
79                    .map(|(xy, _)| xy)
80                    .map(xy_to_pos)
81                    .map(|pos| IntervalPoint::new((pos - response.rect.left_top()).into(), 1)),
82            );
83            let pos = xy_to_pos(*last) - response.rect.left_top();
84            callback.interval_points.push(IntervalPoint::new(pos.into(), 0));
85        }
86    }
87
88    // Nodes
89    ui.scope(|ui| {
90        puffin::profile_scope!("draw_nodes");
91        ui.set_opacity(strength);
92        ui.visuals_mut().text_options.color_transfer_function =
93            epaint::FontColorTransferFunction::Off;
94        for NodeRTreeObject { node_key, xy } in app.spatial().nodes.get(xy_min, xy_max) {
95            let Some(node) = app.graph.nodes().get(&node_key) else {
96                continue;
97            };
98            let node_pos = xy_to_pos(xy);
99            callback.node_points.push(NodePoint::new(
100                (node_pos - response.rect.left_top()).into(),
101                u32::from_le_bytes(Color32::GRAY.to_array()),
102            ));
103            if strength > 0.1 {
104                let galley = ui.painter().layout(
105                    node.name.to_string(),
106                    FontId::proportional(13.0),
107                    Color32::BLACK,
108                    200.0,
109                );
110                let text_pos = node_pos + vec2(0.0, 4.0) - vec2(galley.size().x / 2.0, 0.0);
111                ui.painter().rect_filled(
112                    Rect::from_min_size(text_pos, galley.size()).expand(2.0),
113                    2.0,
114                    Color32::from_white_alpha(128),
115                );
116                ui.painter().galley(text_pos, galley, Color32::WHITE);
117            };
118        }
119    });
120
121    // Trip legs
122    {
123        puffin::profile_scope!("draw_trip_legs");
124        let map_trip_legs = app
125            .spatial()
126            .trip_legs
127            .get(xy_min, xy_max, TimetableTime(app.timer.seconds() as i32))
128            .filter_map(|trip_leg| {
129                app.trips.get(&trip_leg.trip_key).zip(trip_leg_current_pos(
130                    trip_leg,
131                    app.timer.seconds(),
132                    xy_to_pos,
133                ))
134            })
135            .map(|(trip, draw_pos)| {
136                (
137                    trip,
138                    draw_pos,
139                    app.get_service_class_stroke_or_default(trip.service_class).color,
140                )
141            });
142        if strength > 0.1 {
143            ui.painter().add(callback.paint_callback(response.rect));
144            // simply use egui's image features
145            ui.scope(|ui| {
146                ui.set_opacity(strength);
147                ui.visuals_mut().text_options.color_transfer_function =
148                    epaint::FontColorTransferFunction::Off;
149                for (trip, trip_pos, fill) in map_trip_legs {
150                    ui.painter().image(
151                        trip_icon,
152                        Rect::from_center_size(trip_pos, Vec2::splat(ICON_HALF_SIZE * 2.0)),
153                        Rect::from_min_max(pos2(0.0, 0.0), pos2(1.0, 1.0)),
154                        Color32::WHITE,
155                    );
156                    let galley = ui.painter().layout(
157                        trip.name.to_string(),
158                        TIMETABLE_FONT_ID.clone(),
159                        fill,
160                        100.0,
161                    );
162                    let text_pos = trip_pos + Vec2::new(0.0, -ICON_HALF_SIZE)
163                        - vec2(galley.size().x / 2.0, galley.size().y);
164                    ui.painter().rect_filled(
165                        Rect::from_min_size(text_pos, galley.size()).expand(2.0),
166                        2.0,
167                        Color32::from_white_alpha(128),
168                    );
169                    ui.painter().galley(text_pos, galley, fill);
170                }
171            });
172        } else {
173            for (_, draw_pos, fill) in map_trip_legs {
174                callback.node_points.push(NodePoint::new(
175                    (draw_pos - response.rect.left_top()).into(),
176                    u32::from_le_bytes(fill.to_array()),
177                ));
178            }
179            ui.painter().add(callback.paint_callback(response.rect));
180        }
181    }
182}

184pub fn trip_leg_current_pos(
185    TripLegRTreeObject {
186        interval,
187        is_hi_to_lo,
188        trip_key: _,
189        curr_arr: _,
190        curr_dep,
191        next_arr,
192    }: TripLegRTreeObject,
193    current_secs: f64,
194    xy_to_pos: impl Fn(XyPos) -> Pos2,
195) -> Option<Pos2> {
196    let curr_progress = {
197        let t1 = curr_dep.0 as f64;
198        let t2 = next_arr.0 as f64;
199        // curr arr time is simply out of range.
200        // Just clamp the value so it stays at the station...
201        inverse_lerp(t1..=t2, current_secs).unwrap_or(0.0).at_least(0.0) as f32
202    };
203    // current_secs is a float while t1 and t2 are i32
204    // this leaves a ~1 in-app second gap that duplicates entries.
205    // simply remove the entry when progress > 1.0.
206    if curr_progress > 1.0 {
207        return None;
208    }
209    let curr_progress = if is_hi_to_lo {
210        curr_progress
211    } else {
212        1.0 - curr_progress
213    };
214    // Simple O(n) find
215    let prev_idx: usize = interval
216        .iter()
217        .enumerate()
218        .find_map(|(idx, &(_, interval_progress))| {
219            (curr_progress >= interval_progress).then_some(idx)
220        })
221        .unwrap_or(0);
222
223    let (prev_xy, prev_progress) = interval[prev_idx];
224    let (next_xy, next_progress) = interval[prev_idx + 1];
225    let normalized_progress =
226        inverse_lerp(prev_progress..=next_progress, curr_progress).unwrap_or(0.0);
227    let prev_pos = xy_to_pos(prev_xy);
228    let next_pos = xy_to_pos(next_xy);
229    Some(prev_pos.lerp(next_pos, normalized_progress))
230}
