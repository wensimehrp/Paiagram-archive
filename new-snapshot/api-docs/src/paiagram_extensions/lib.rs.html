21pub struct Metadata {
22    pub id: String,
23    pub name: Option<String>,

29impl Metadata {

31        rquickjs_serde::from_value(value).map_err(|err| {
32            Error::new_from_js_message("rquickjs-serde", "Metadata", err.to_string())
33        })

37#[derive(Clone, Debug)]
38pub struct ByteCodes {
39    inner: Arc<[u8]>,
40    pub metadata: Metadata,
41}
42
43impl ByteCodes {
44    pub fn compile(script_str: &str) -> Result<Self> {
45        let ctx = Context::full(&RQUICKJS_RUNTIME).unwrap();
46        ctx.with(|ctx| {
47            install_console(&ctx)?;
48            let script_error = |err: rquickjs::CaughtError<'_>| {
49                Error::new_from_js_message("script", "extension", err.to_string())
50            };
51            let module = Module::declare(ctx.clone(), "extension.js", script_str)
52                .catch(&ctx)
53                .map_err(script_error)?;
54            let bytes = module.write(WriteOptions::default()).catch(&ctx).map_err(script_error)?;
55            let (module, promise) = module.eval().catch(&ctx).map_err(script_error)?;
56            promise.finish::<()>().catch(&ctx).map_err(script_error)?;
57            let metadata_val: Value = module.get("metadata").catch(&ctx).map_err(script_error)?;
58            let metadata = Metadata::parse(&ctx, metadata_val)?;
59            Ok(Self { inner: bytes.into(), metadata })
60        })
61    }
62}

64/// Compile the script into bytecodes and extract its metadata
65pub fn compile_script(script_str: Arc<str>, sender: mpsc::Sender<Result<ByteCodes>>) {
66    rayon::spawn(move || {
67        let _ = sender.send(ByteCodes::compile(&script_str));
68    });
69}

82static RQUICKJS_RUNTIME: LazyLock<Runtime> = LazyLock::new(|| Runtime::new().unwrap());

84/// Evaluate the config
85pub fn eval_config(
86    runtime_data: JsRuntimeData,
87    status: Arc<Mutex<ConfigRunningStatus>>,
88    bytecodes: ByteCodes,
89) {
90    *status.lock() = ConfigRunningStatus::Running;
91    rayon::spawn(move || {
92        let result = with_context(runtime_data, None, &bytecodes, |ctx, module, world, _io| {
93            let entry: Function = module.get("config_ui")?;
94            let value: Value = entry.call((world,))?;
95            ConfigUi::parse_many(ctx, value)
96        });
97        *status.lock() = ConfigRunningStatus::Finished(result);
98    });
99}

101/// Execute the extension's `run` entrypoint
102pub fn run_script(
103    runtime_data: JsRuntimeData,
104    io: JsIo,
105    return_values: ReturnValueMap,
106    status: Arc<Mutex<ScriptRunningStatus>>,
107    bytecodes: ByteCodes,
108) {
109    *status.lock() = ScriptRunningStatus::Running;
110    rayon::spawn(move || {
111        let result = with_context(runtime_data, Some(io), &bytecodes, |_ctx, module, world, io| {
112            let entry: Function = module.get("run")?;
113            let io = io.expect("run always receives io");
114            entry.call::<_, ()>((world.clone(), return_values, io))?;
115            Ok(Some(std::mem::take(&mut world.borrow_mut().world)))
116        });
117        *status.lock() = ScriptRunningStatus::Finished(result);
118    });
119}

121fn with_context<T>(
122    runtime_data: JsRuntimeData,
123    io: Option<JsIo>,
124    bytecodes: &ByteCodes,
125    f: impl for<'js> FnOnce(
126        &Ctx<'js>,
127        Module<'js, rquickjs::module::Evaluated>,
128        Class<'js, JsRuntimeData>,
129        Option<Class<'js, JsIo>>,
130    ) -> Result<T>,
131) -> Result<T> {
132    let ctx = Context::full(&RQUICKJS_RUNTIME).unwrap();
133    ctx.with(|ctx| {
134        install_console(&ctx)?;
135        let world = Class::instance(ctx.clone(), runtime_data)?;
136        let io = io.map(|io| Class::instance(ctx.clone(), io)).transpose()?;
137
138        let script_error = |err: rquickjs::CaughtError<'_>| {
139            Error::new_from_js_message("script", "extension", err.to_string())
140        };
141        // SAFETY: `bytecodes.inner` was produced by `Module::write` on a valid declared module.
142        let module = unsafe { Module::load(ctx.clone(), &bytecodes.inner) }
143            .catch(&ctx)
144            .map_err(script_error)?;
145        let (module, promise) = module.eval().catch(&ctx).map_err(script_error)?;
146        promise.finish::<()>().catch(&ctx).map_err(script_error)?;
147
148        f(&ctx, module, world, io)
149            .catch(&ctx)
150            .map_err(|err| Error::new_from_js_message("entry point", "extension", err.to_string()))
151    })
152}

154/// JavaScript factory for the global `console` object. It takes the Rust log bridge as a
155/// parameter, so no extra globals are needed.
156const CONSOLE_SHIM: &str = r#"
157((log) => {
158    const fmt = (value) => {
159        if (typeof value === "string") return value;
160        if (value === null) return "null";
161        if (value === undefined) return "undefined";
162        if (typeof value === "object") {
163            try { return JSON.stringify(value); } catch (_) { return String(value); }
164        }
165        return String(value);
166    };
167    const make = (level) => (...args) => log(level, args.map(fmt).join(" "));
168    return {
169        log: make(1),
170        info: make(1),
171        debug: make(2),
172        warn: make(3),
173        error: make(4),
174        trace: make(5),
175    };
176})
177"#;
178
179/// Installs the standard `console` global, whose methods forward to the `log` crate.
180fn install_console<'js>(ctx: &Ctx<'js>) -> Result<()> {
181    let log_fn = Function::new(ctx.clone(), |level: u32, message: String| {
182        let level = match level {
183            2 => log::Level::Debug,
184            3 => log::Level::Warn,
185            4 => log::Level::Error,
186            5 => log::Level::Trace,
187            _ => log::Level::Info,
188        };
189        log::log!(target: "script", level, "{message}");
190    })?;
191    let factory: Function = ctx.eval(CONSOLE_SHIM)?;
192    let console: Object = factory.call((log_fn,))?;
193    ctx.globals().set("console", console)?;
194    Ok(())
195}

205    #[test]
206    fn selected_items_serialize_keys_exactly() {
207        // Allocate a key with nonzero counter bits so f64 precision loss would be visible.
208        let _ = TripKey::new();
209        let trip = TripKey::new();
210        let data = JsRuntimeData {
211            world: World::default(),
212            selected_items: SelectedItem::Trip(trip.into()).into(),
213        };
214
215        let runtime = Runtime::new().unwrap();
216        let context = Context::full(&runtime).unwrap();
217        context.with(|ctx| {
218            let value: Value = data.selected_items(ctx.clone()).unwrap();
219            let object = value.into_object().unwrap();
220            let trips: Array = object.get("Trips").unwrap();
221            let first: BigInt = trips.get(0).unwrap();
222            assert_eq!(first.to_i64().unwrap() as u64, trip.to_bits());
223        });
224    }
