10#[derive(Clone, Debug, Deserialize)]

49impl ConfigUi {
50    /// Decodes a UI definition from a JavaScript value (an array of element objects).
51    /// `undefined` yields no elements.
52    pub fn parse_many<'js>(_ctx: &Ctx<'js>, value: Value<'js>) -> rquickjs::Result<Vec<ConfigUi>> {
53        if value.is_undefined() {
54            return Ok(Vec::new());
55        }
56        rquickjs_serde::from_value(value).map_err(|err| {
57            rquickjs::Error::new_from_js_message("rquickjs-serde", "ConfigUi", err.to_string())
58        })
59    }
60
61    fn desc(&self) -> Option<&String> {
62        match self {
63            Self::Text { desc } => Some(desc),
64            Self::Separator => None,
65            // special case for collapsing header as we want to display it on the header
66            Self::CollapsingHeader { .. } => None,
67            Self::Radio { desc, .. } => Some(desc),
68            Self::Tickbox { .. } => None,
69            Self::TextInput { desc, .. } => Some(desc),
70            Self::DragValue { desc, .. } => Some(desc),
71        }
72    }
73
74    fn get_return_value(&self) -> Option<(ReturnVariableName, ConfigUiReturnValue)> {
75        match self {
76            Self::Radio { variable, default, .. } => {
77                Some((variable.clone(), ConfigUiReturnValue::Number(*default as f64)))
78            }
79            Self::Tickbox { variable, default, .. } => {
80                Some((variable.clone(), ConfigUiReturnValue::Boolean(*default)))
81            }
82            Self::TextInput { variable, default, .. } => Some((
83                variable.clone(),
84                ConfigUiReturnValue::String(default.as_ref().map_or_default(String::clone)),
85            )),
86            Self::DragValue { variable, default, .. } => {
87                Some((variable.clone(), ConfigUiReturnValue::Number(*default)))
88            }
89            Self::CollapsingHeader { variable, items, .. } => Some((
90                variable.clone(),
91                ConfigUiReturnValue::Map(
92                    items.iter().filter_map(ConfigUi::get_return_value).collect(),
93                ),
94            )),
95            _ => None,
96        }
97    }
98}

100#[derive(Clone, Debug)]
101pub enum ConfigUiReturnValue {
102    String(String),
103    Number(f64),
104    Boolean(bool),
105    Map(ReturnValueMap),
106}

108impl<'js> IntoJs<'js> for ConfigUiReturnValue {
109    fn into_js(self, ctx: &Ctx<'js>) -> rquickjs::Result<Value<'js>> {
110        match self {
111            Self::String(value) => value.into_js(ctx),
112            Self::Number(value) => value.into_js(ctx),
113            Self::Boolean(value) => value.into_js(ctx),
114            Self::Map(map) => map.into_js(ctx),
115        }
116    }
117}

119/// Builds the initial return values (the widgets' defaults) for a UI definition.
120pub fn default_return_values(ui_definition: &[ConfigUi]) -> ReturnValueMap {
121    ui_definition.iter().filter_map(ConfigUi::get_return_value).collect()
122}

124pub struct ConfigUiList<'a> {
125    pub ui_definition: &'a [ConfigUi],
126    pub return_value: &'a mut ReturnValueMap,
127}
128
129impl<'a> ConfigUiList<'a> {
130    pub fn show_ui(mut self, ui: &mut Ui) {
131        for elem in self.ui_definition {
132            if let Some(desc) = elem.desc() {
133                ui.label(desc);
134                Frame::NONE
135                    .inner_margin(Margin { left: 12, ..Default::default() })
136                    .show(ui, |ui| self.show_inner(ui, elem));
137            } else {
138                self.show_inner(ui, elem);
139            }
140        }
141    }
142    fn show_inner(&mut self, ui: &mut Ui, elem: &ConfigUi) {
143        match elem {
144            ConfigUi::Text { .. } => {
145                // already handled
146            }
147            ConfigUi::Separator => {
148                ui.separator();
149            }
150            ConfigUi::CollapsingHeader { variable, desc, items, .. } => {
151                let Some(ConfigUiReturnValue::Map(map)) = self.return_value.get_mut(variable)
152                else {
153                    unreachable!();
154                };
155                ui.collapsing(desc, |ui| {
156                    ConfigUiList { ui_definition: items, return_value: map }.show_ui(ui);
157                });
158            }
159            ConfigUi::Radio { variable, items, .. } => {
160                let Some(ConfigUiReturnValue::Number(current_value)) =
161                    self.return_value.get_mut(variable)
162                else {
163                    unreachable!()
164                };
165                ui.horizontal_wrapped(|ui| {
166                    for (idx, item) in items.iter().enumerate() {
167                        ui.radio_value(current_value, idx as f64, item);
168                    }
169                });
170            }
171            ConfigUi::Tickbox { variable, desc, .. } => {
172                let Some(ConfigUiReturnValue::Boolean(current_value)) =
173                    self.return_value.get_mut(variable)
174                else {
175                    unreachable!()
176                };
177                ui.checkbox(current_value, desc);
178            }
179            ConfigUi::TextInput { variable, placeholder, .. } => {
180                let Some(ConfigUiReturnValue::String(current_value)) =
181                    self.return_value.get_mut(variable)
182                else {
183                    unreachable!()
184                };
185                TextEdit::singleline(current_value)
186                    .hint_text(placeholder.as_ref().map_or_default(String::as_str))
187                    .show(ui);
188            }
189            ConfigUi::DragValue { variable, integer, min, max, .. } => {
190                let Some(ConfigUiReturnValue::Number(current_value)) =
191                    self.return_value.get_mut(variable)
192                else {
193                    unreachable!()
194                };
195                let mut widget =
196                    DragValue::new(current_value).range(*min..=*max).clamp_existing_to_range(true);
197                if *integer {
198                    widget = widget.min_decimals(0).max_decimals(0)
199                }
200                ui.add(widget);
201            }
202        }
203    }
