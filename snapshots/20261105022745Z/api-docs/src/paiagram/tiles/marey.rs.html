67impl MareyTile {
68    pub fn new(key: RouteKey) -> Self {
69        Self {
70            key,
71            navi: MareyTileNavigation::default(),
72            cache: MareyCache::default(),
73            callback: gpu_draw::MareyCallback::new(key),
74            prev_world: None,
75        }
76    }
77    fn update_cache(&mut self, curr_world: &World) -> bool {
78        puffin::profile_function!();
79        let Some(curr_route) = curr_world.routes.get(&self.key) else {
80            return false;
81        };
82        // TODO: reuse MareyCache
83        let (_rebuild_route, trips_to_rebuild) = match self.prev_world.as_ref() {
84            None => (
85                true,
86                Either::Left(curr_route.intervals.all_passing_trips(curr_world).into_iter()),
87            ),
88            Some(prev_world) => {
89                let rebuild_route = prev_world.graph.nodes().ptr_eq(curr_world.graph.nodes())
90                    || prev_world.graph.intervals().ptr_eq(curr_world.graph.intervals())
91                    || prev_world.routes.get(&self.key) != Some(curr_route);
92                let trips_to_rebuild =
93                    prev_world.trips.diff(&curr_world.trips).map(|diff| match diff {
94                        DiffItem::Add(key, _)
95                        | DiffItem::Remove(key, _)
96                        | DiffItem::Update { old: (key, _), .. } => *key,
97                    });
98                (rebuild_route, Either::Right(trips_to_rebuild))
99            }
100        };
101        curr_route.intervals.populate_trips(curr_world, &mut self.cache, trips_to_rebuild);
102        self.prev_world = Some(curr_world.clone());
103        self.callback.populate_entry_segments(&self.cache, curr_world);
104        true
105    }
106}

115struct MareyTileNavigation {

117    y_offset: CanvasLength,
118    zoom: Vec2,
119    #[serde(skip, default = "default_visible_rect")]

125impl Default for MareyTileNavigation {
126    fn default() -> Self {
127        // Default horizontal scale: five minutes of time span one centimetre of canvas.
128        let five_minutes_in_ticks = TimetableTime::from_hms(0, 5, 0).to_ticks().0 as f32;
129        let one_cm_in_pts = CanvasLength::from_cm(1.0).to_egui_pts();
130        Self {
131            x_offset: Tick(0),
132            y_offset: CanvasLength(0.0),
133            zoom: vec2(one_cm_in_pts / five_minutes_in_ticks, 1.0),
134            visible_rect: Rect::NOTHING,
135            max_height: CanvasLength(0.0),
136        }
137    }
138}

144impl Navigatable for MareyTileNavigation {
145    type XOffset = Tick;
146    type YOffset = CanvasLength;
147
148    fn zoom_x(&self) -> f32 {
149        self.zoom.x
150    }
151    fn zoom_y(&self) -> f32 {
152        self.zoom.y * CanvasLength::egui_pts_per_mm() as f32
153    }
154    fn set_zoom(&mut self, zoom_x: f32, zoom_y: f32) {
155        self.zoom = Vec2::new(zoom_x, zoom_y / CanvasLength::egui_pts_per_mm() as f32);
156    }
157    fn offset_x(&self) -> f64 {
158        self.x_offset.0 as f64
159    }
160    fn offset_y(&self) -> f64 {
161        self.y_offset.0
162    }
163    fn set_offset(&mut self, offset_x: f64, offset_y: f64) {
164        self.x_offset = Tick(offset_x.round() as i64);
165        self.y_offset = CanvasLength(offset_y);
166    }
167    fn visible_rect(&self) -> egui::Rect {
168        self.visible_rect
169    }
170    fn allow_axis_zoom(&self) -> bool {
171        true
172    }
173    fn clamp_zoom(&self, zoom_x: f32, zoom_y: f32) -> (f32, f32) {
174        let ppmm = CanvasLength::egui_pts_per_mm() as f32;
175        (
176            zoom_x.clamp(0.00005, 0.4),
177            zoom_y.clamp(ppmm * 0.1, ppmm * 2048.0),
178        )
179    }
180    fn post_navigation(&mut self, response: &egui::Response) {
181        let max_tick = Tick::from_timetable_time(TimetableTime(366 * 86400)).0;
182        self.x_offset = Tick(self.x_offset.0.clamp(
183            -max_tick,
184            max_tick - (response.rect.width() as f64 / self.zoom.x as f64) as i64,
185        ));
186        const TOP_BOTTOM_PADDING: f32 = 30.0;
187        let v = if response.rect.height() / self.zoom_y()
188            > (self.max_height.0 as f32 + TOP_BOTTOM_PADDING * 2.0 / self.zoom_y())
189        {
190            ((-response.rect.height() / self.zoom_y() + self.max_height.0 as f32) / 2.0) as f64
191        } else {
192            self.y_offset.0.clamp(
193                (-TOP_BOTTOM_PADDING / self.zoom_y()) as f64,
194                (self.max_height.0 as f32 - response.rect.height() / self.zoom_y()
195                    + TOP_BOTTOM_PADDING / self.zoom_y()) as f64,
196            )
197        };
198        self.y_offset.0 = v;
199    }
200}

255fn draw_time_lines(painter: &mut Painter, navi: &MareyTileNavigation) {
256    puffin::profile_function!();
257    const MAX_SCREEN_WIDTH: f64 = 64.0;
258    const MIN_SCREEN_WIDTH: f64 = 32.0;
259    let sizes = [
260        Tick::from_timetable_time(TimetableTime(1)).0, // 1 second
261        Tick::from_timetable_time(TimetableTime(10)).0, // 10 seconds
262        Tick::from_timetable_time(TimetableTime(30)).0, // 30 seconds
263        Tick::from_timetable_time(TimetableTime(60)).0, // 1 minute
264        Tick::from_timetable_time(TimetableTime(60 * 5)).0, // 5 minutes
265        Tick::from_timetable_time(TimetableTime(60 * 10)).0, // 10 minutes
266        Tick::from_timetable_time(TimetableTime(60 * 30)).0, // 30 minutes
267        Tick::from_timetable_time(TimetableTime(60 * 60)).0, // 1 hour
268        Tick::from_timetable_time(TimetableTime(60 * 60 * 4)).0, // 4 hours
269        Tick::from_timetable_time(TimetableTime(60 * 60 * 24)).0, // 1 day
270    ];
271    let visible_ticks = navi.visible_x();
272    let ticks_per_screen_unit = navi.x_per_screen_unit_f64();
273    let screen_rect = navi.visible_rect();
274    let pixels_per_point = painter.pixels_per_point();
275    let mut drawn: Vec<i64> = Vec::with_capacity(30);
276
277    // align the first tick to a spacing boundary that is <= visible start.
278    let first_visible_position = sizes
279        .iter()
280        .position(|s| *s as f64 / ticks_per_screen_unit * 1.5 > MIN_SCREEN_WIDTH)
281        .unwrap_or(0);
282    let visible = &sizes[first_visible_position..];
283    for (i, spacing) in visible.iter().enumerate().rev() {
284        let first = visible_ticks.start.0 - visible_ticks.start.0.rem_euclid(*spacing) - spacing;
285        let mut tick = first;
286        let strength = (((*spacing as f64 / ticks_per_screen_unit * 1.5) - MIN_SCREEN_WIDTH)
287            / (MAX_SCREEN_WIDTH - MIN_SCREEN_WIDTH))
288            .clamp(0.0, 1.0);
289        if strength < 0.1 {
290            continue;
291        }
292        // TODO: make stroke depend on current theme
293        let mut current_stroke = Stroke {
294            width: 0.6,
295            color: Color32::GRAY,
296        };
297        if strength.is_finite() {
298            // strange bug here
299            current_stroke.color = current_stroke.color.gamma_multiply(strength as f32);
300        }
301        current_stroke.width = 0.5;
302        while tick <= visible_ticks.end.0 {
303            tick += *spacing;
304            if drawn.contains(&tick) {
305                continue;
306            }
307            let mut x = navi.logical_x_to_screen_x(Tick(tick));
308            current_stroke.round_center_to_pixel(pixels_per_point, &mut x);
309            painter.vline(x, screen_rect.top()..=screen_rect.bottom(), current_stroke);
310            drawn.push(tick);
311            let time = Tick(tick).to_timetable_time();
312            let mut offset = screen_rect.top();
313            let text = match i + first_visible_position {
314                0..=2 => time.to_hmsd().2.to_string(),
315                3..=8 => format!("{}:{:02}", time.to_hmsd().0, time.to_hmsd().1),
316                9 => {
317                    offset += 13.0;
318                    time.to_string()
319                }
320                _ => unreachable!(),
321            };
322            painter.text(
323                Pos2 { x, y: offset },
324                Align2::CENTER_TOP,
325                text,
326                FontId::proportional(13.0),
327                current_stroke.color,
328            );
329        }
330    }
331}

333fn main_display(tile: &mut MareyTile, app: &mut App, ui: &mut Ui) {
334    puffin::profile_function!();
335    let (response, mut painter) =
336        ui.allocate_painter(ui.available_size_before_wrap(), Sense::click_and_drag());
337    tile.navi.visible_rect = response.rect;
338    tile.navi.handle_navigation(ui, &response);
339
340    draw_time_lines(&mut painter, &tile.navi);
341
342    let Some(route) = app.routes.get(&tile.key).cloned() else {
343        return;
344    };
345
346    let (station_heights, max_height) = route.intervals.0.iter().fold(
347        (
348            Vec::with_capacity(route.intervals.0.len()),
349            CanvasLength::ZERO,
350        ),
351        |(mut acc, height_so_far), interval| {
352            let Some(station) = interval
353                .station_record
354                .station_key(app.graph.nodes())
355                .and_then(|key| app.stations.get(key))
356            else {
357                return (acc, height_so_far);
358            };
359            let station_line_stroke = Stroke {
360                width: 0.6,
361                color: Color32::GRAY,
362            };
363            let mut y = tile.navi.logical_y_to_screen_y(height_so_far);
364            station_line_stroke.round_center_to_pixel(ui.pixels_per_point(), &mut y);
365            painter.hline(response.rect.x_range(), y, station_line_stroke);
366            painter.text(
367                pos2(response.rect.left(), y),
368                Align2::LEFT_BOTTOM,
369                station.name.as_str(),
370                FontId::proportional(13.0),
371                station_line_stroke.color,
372            );
373            acc.push(height_so_far);
374            (
375                acc,
376                height_so_far + interval.canvas_length.unwrap_or(CanvasLength::from_cm(1.5)),
377            )
378        },
379    );
380
381    tile.navi.max_height = max_height;
382
383    // paint the station lines and Marey lines
384    tile.callback.populate_stations(station_heights.iter().copied());
385    tile.callback.populate_uniforms(&tile.navi, response.rect);
386    painter.add(tile.callback.clone().paint_callback(response.rect));
387
388    // current time overlay
389    {
390        let seconds = app.timer.seconds();
391        let mut x = tile.navi.logical_x_to_screen_x(Tick::from_seconds_f64(seconds));
392        let stroke = Stroke::new(2.0, Color32::RED);
393        stroke.round_center_to_pixel(ui.pixels_per_point(), &mut x);
394        painter.vline(x, response.rect.y_range(), stroke);
395        let range = Rangef::point(x).expand(5.0);
396        painter.hline(range, response.rect.bottom() - stroke.width / 2.0, stroke);
397        painter.hline(range, response.rect.top() + stroke.width / 2.0, stroke);
398    }
399
400    // selected item overlay
401    let animation_progress = ui.animate_bool_responsive(
402        ui.scope_id().with("ui animation progress"),
403        !matches!(app.selected_items, SelectedItems::None),
404    );
405    painter.rect_filled(
406        response.rect,
407        0,
408        ui.visuals().extreme_bg_color.gamma_multiply(lerp(0.0..=0.5, animation_progress)),
409    );
410
411    // interactive zone
412    if response.clicked()
413        && let Some(pointer_pos) = response.interact_pointer_pos()
414        // this makes sure that when already selecting item
415        // the next action always deselects the item instead of focusing on new item
416        && !matches!(app.selected_items, SelectedItems::TripExtension(_)) // it has its own state
417        && let SelectedItems::None = app.selected_items.take()
418    {
419        if let Some(trip_key) = select_trip_from_pos(tile, pointer_pos, &station_heights) {
420            app.selected_items.replace(SelectedItem::Trip(trip_key.into()));
421        } else if response.double_clicked()
422            // TODO: also show the route when hovering in another trip panel?
423            && let Some(station_idx) = find_nearest_station_idx_from_pos(&station_heights, pointer_pos, &tile.navi)
424            && let interval = &route.intervals.0[station_idx]
425            && let Some(station_key) = interval.station_record.station_key(app.graph.nodes())
426            && let Some(station) = app.stations.get(station_key)
427            && let Some(node) = station.cache.nodes.first()
428        {
429            let t = tile.navi.screen_x_to_logical_x(pointer_pos.x).to_timetable_time();
430            let trip_key = TripKey::new();
431            let id = TEntryId::new();
432            let new_entry = TEntry {
433                node: *node,
434                arr_or_pass: TravelMode::At(t),
435                dep: None,
436                id,
437            };
438            let _ = app.mutate(|mut world| {
439                world.trips.insert(
440                    trip_key,
441                    Wfc::new(Trip {
442                        name: "New Trip".into(),
443                        schedule: TripSchedule::new(eco_vec![new_entry]),
444                        service_class: None,
445                        vehicles: Default::default(),
446                    }),
447                );
448                Ok(world)
449            });
450            app.selected_items = SelectedItems::TripExtension((trip_key, None));
451        }
452    }
453
454    // display and edit zone
455    if matches!(app.selected_items, SelectedItems::Trips(_))
456        && let SelectedItems::Trips(mut trips) = app.selected_items.take()
457    {
458        for selected in trips.iter_mut() {
459            let trip_key = &selected.trip_key;
460            let Some(trip) = app.trips.get(trip_key) else {
461                continue;
462            };
463            let style = trip
464                .service_class
465                .and_then(|key| app.service_classes.get(&key))
466                .map_or_default(|class| class.style);
467            let stroke = Stroke {
468                width: lerp(style.width as f32..=5.0, animation_progress),
469                color: style.color,
470            };
471            let schedule = trip.schedule.clone().with_entry_shift(selected.editing_offset);
472            let Some(polylines) = route.intervals.populate_schedule(&app, &schedule) else {
473                continue;
474            };
475            let polylines = polylines.clone();
476            let ret = show_focused_trips::show_focused_trip(
477                &schedule,
478                stroke,
479                &mut *app,
480                animation_progress,
481                ui,
482                tile,
483                &station_heights,
484                &mut painter,
485                polylines,
486                *trip_key,
487                selected.editing_offset.map(|it| it.1),
488            );
489            if response.contains_pointer() {
490                selected.editing_offset = ret;
491            }
492        }
493        app.selected_items = SelectedItems::Trips(trips);
494    } else if let &SelectedItems::TripExtension((trip_key, focused_pos)) = &app.selected_items
495        && let Some(trip) = app.trips.get(&trip_key)
496    {
497        let stroke = Stroke::new(3.0, Color32::RED);
498
499        let focused_pos = if response.contains_pointer()
500            && let Some(pointer_pos) = ui.pointer_hover_pos()
501            && let Some(station_idx) =
502                find_nearest_station_idx_from_pos(&station_heights, pointer_pos, &tile.navi)
503            && let Some(interval) = route.intervals.0.get(station_idx)
504            && let Some(station_key) = interval.station_record.station_key(app.graph.nodes())
505            && let Some(station) = app.stations.get(station_key)
506            && let Some(node) = station.cache.nodes.first()
507        {
508            Some((
509                *node,
510                tile.navi.screen_x_to_logical_x(pointer_pos.x).to_timetable_time(),
511            ))
512        } else {
513            focused_pos
514        };
515
516        let schedule = trip.schedule.clone().with_pushed_entry(focused_pos);
517
518        if response.double_clicked()
519            && let Some((focused_node, focused_time)) = focused_pos
520        {
521            let _ = app.mutate(|mut world| {
522                let trip = world.trips.get_mut(&trip_key).ok_or("wtf?")?;
523                if let Some(prev_entry) = trip.schedule.entries().last()
524                    && prev_entry.node == focused_node
525                {
526                    // merge if same node
527                    let prev_id = prev_entry.id;
528                    trip.schedule.set_dep_mode(prev_id, Some(TravelMode::At(focused_time)));
529                } else {
530                    // create new entry if different nodes
531                    trip.schedule.push(TEntry {
532                        node: focused_node,
533                        arr_or_pass: TravelMode::At(focused_time),
534                        dep: None,
535                        id: TEntryId::new(),
536                    })
537                };
538                Ok(world)
539            });
540        }
541
542        route.intervals.populate_schedule(&app, &schedule).into_iter().flatten().for_each(
543            |polyline| {
544                draw_polyline(
545                    &tile.navi,
546                    &polyline,
547                    &station_heights,
548                    &mut painter,
549                    stroke,
550                );
551            },
552        );
553
554        app.selected_items = SelectedItems::TripExtension((trip_key, focused_pos));
555
556        if response.secondary_clicked() {
557            app.selected_items = SelectedItems::None;
558        }
559    }
560    if let Some(pos) = ui.pointer_hover_pos()
561        && ui.ui_contains_pointer()
562    {
563        paint_time_cursor(
564            &tile.navi,
565            &station_heights,
566            pos,
567            ui,
568            response.rect,
569            route,
570            app,
571        );
572    }
573}

575fn select_trip_from_pos(
576    tile: &MareyTile,
577    ptr_pos: Pos2,
578    station_heights: &[CanvasLength],
579) -> Option<TripKey> {
580    puffin::profile_function!();
581    const SELECT_RADIUS: f32 = 7.0;
582    // The same point mapping as when drawing (see the display zone in `main_display`), so that
583    // what is selected matches what is drawn. Each entry spans `arr`..`dep` at one placement.
584    let points = |estimate: &TEstimate, idx: u32, progress: f32| {
585        let x1 = tile.navi.logical_x_to_screen_x(estimate.arr.to_ticks());
586        let x2 = tile.navi.logical_x_to_screen_x(estimate.dep.to_ticks());
587        let curr_y = tile.navi.logical_y_to_screen_y(station_heights[idx as usize]);
588        let next_y = tile.navi.logical_y_to_screen_y(station_heights[idx as usize + 1]);
589        let y = lerp(curr_y..=next_y, progress);
590        [Pos2::new(x1, y), Pos2::new(x2, y)]
591    };
592
593    let hit = |p1: Pos2, p2: Pos2| {
594        distance_sq_to_line((p1, p2), ptr_pos) <= SELECT_RADIUS * SELECT_RADIUS
595    };
596
597    tile.cache
598        .map
599        .iter()
600        .flat_map(|(k, v)| v.iter().map(move |l| (k, l)))
601        .flat_map(|(k, l)| l.array_windows().map(move |w| (k, w)))
602        .find_map(|(key, [curr, next])| {
603            let [curr_arr, curr_dep] = points(&curr.0, curr.2, curr.3);
604            let [next_arr, next_dep] = points(&next.0, next.2, next.3);
605            // The segments the drawing produces between two consecutive entries: the dwell at
606            // the first, the travel between them, and the dwell at the second.
607            (hit(curr_arr, curr_dep) || hit(curr_dep, next_arr) || hit(next_arr, next_dep))
608                .then_some(*key)
609        })
610}

612fn find_nearest_station_idx_from_pos(
613    station_heights: &[CanvasLength],
614    pos: Pos2,
615    navi: &MareyTileNavigation,
616) -> Option<usize> {
617    if station_heights.is_empty() {
618        return None;
619    }
620    let target_y = navi.screen_y_to_logical_y(pos.y);
621    let idx = station_heights.partition_point(|&height| height < target_y);
622
623    if idx == 0 {
624        Some(0)
625    } else if idx >= station_heights.len() {
626        Some(station_heights.len() - 1)
627    } else {
628        let prev = station_heights[idx - 1];
629        let next = station_heights[idx];
630        let mid = CanvasLength((prev.0 + next.0) / 2.0);
631
632        if target_y <= mid {
633            Some(idx - 1)
634        } else {
635            Some(idx)
636        }
637    }
638}

640fn paint_time_cursor(
641    navi: &MareyTileNavigation,
642    station_heights: &[CanvasLength],
643    pointer_pos: Pos2,
644    ui: &mut Ui,
645    rect: Rect,
646    route: Wfc<Route, RouteCache>,
647    app: &App,
648) {
649    let stroke = Stroke::new(1.5, Color32::RED);
650    let (pointer_t, pointer_distance) = navi.screen_pos_to_xy(pointer_pos);
651    let Some(station_idx) = find_nearest_station_idx_from_pos(station_heights, pointer_pos, navi)
652    else {
653        return;
654    };
655    let dt = ui.input(|input| input.stable_dt).at_most(0.1);
656    let mut requires_repaint = false;
657    let new_logical_y = station_heights[station_idx];
658    let smoothed_logical_y = ui.data_mut(|data| {
659        let smoothed: &mut CanvasLength =
660            data.get_temp_mut_or(ui.scope_id().with("pointer animation"), new_logical_y);
661        let t = emath::exponential_smooth_factor(0.9, 0.05, dt);
662        *smoothed = CanvasLength(lerp(smoothed.0..=new_logical_y.0, t as f64));
663        if (*smoothed - new_logical_y).0.abs() > 0.1 {
664            requires_repaint = true;
665        }
666        *smoothed
667    });
668    if requires_repaint {
669        ui.request_repaint();
670    }
671    // time axis snaps, station axis animates
672    let mut pos =
673        navi.xy_to_screen_pos(pointer_t.to_timetable_time().to_ticks(), smoothed_logical_y);
674    // stupid API here...
675    stroke.round_center_to_pixel(ui.pixels_per_point(), &mut pos.x);
676    stroke.round_center_to_pixel(ui.pixels_per_point(), &mut pos.y);
677    ui.painter().hline(rect.x_range(), pos.y, stroke);
678    ui.painter().vline(pos.x, rect.y_range(), stroke);
679    ui.painter().text(
680        pos + vec2(-3.0, -3.0),
681        Align2::RIGHT_BOTTOM,
682        pointer_t.to_timetable_time(),
683        TIMETABLE_FONT_ID.clone(),
684        ui.visuals().strong_text_color(),
685    );
686    ui.painter().text(
687        pos + vec2(3.0, -3.0),
688        Align2::LEFT_BOTTOM,
689        "Double click to create trip",
690        FontId::proportional(13.0),
691        ui.visuals().strong_text_color(),
692    );
693    let station_name = route.intervals.0[station_idx]
694        .station_record
695        .station_key(app.graph.nodes())
696        .and_then(|key| app.stations.get(key))
697        .map_or("Unknown", |wfc| wfc.name.as_str());
698    ui.painter().text(
699        pos + vec2(-3.0, 3.0),
700        Align2::RIGHT_TOP,
701        station_name,
702        FontId::proportional(13.0),
703        ui.visuals().strong_text_color(),
704    );
