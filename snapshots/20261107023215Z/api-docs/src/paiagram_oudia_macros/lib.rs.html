9#[derive(FromMeta)]
10enum OuDiaType {
11    /// [key => Val]
12    /// return Val
13    SinglePairSingleEntry(String),
14    /// [key => Val1, Val2, Val3, ...]
15    /// return Vec<Val>
16    SinglePairManyEntries(String),
17    /// [Struct::key => Struct]
18    /// return Struct
19    SingleStruct(String),
20    /// [Struct::key => Struct, Struct::key => Struct, ...]
21    /// return Vec<Struct>
22    ManyStructs(String),
23    /// [key => [Struct::key => Struct, Struct::key => Struct, ...]]
24    /// return Vec<Struct>
25    SingleStructManyEntries(String),
26    /// [key1 => [Struct::key => Struct, ...], key2 => [Struct::key => Struct, ...]]
27    /// return Vec<Struct>
28    TwinStructMultipleEntries { first: String, second: String },
29}

31#[derive(FromMeta)]
32#[darling(derive_syn_parse)]
33struct StructArgs {
34    key: Option<String>,
35    alias: Option<String>,
36}

38/// A field-level `default` annotation, mirroring serde's `#[serde(default)]`.
39///
40/// `#[oudia(default)]` falls back to `Default::default()`, while
41/// `#[oudia(default = <expr>)]` uses the given expression.
42enum DefaultValue {
43    UseDefault,
44    Value(syn::Expr),
45}
46
47impl FromMeta for DefaultValue {
48    fn from_word() -> darling::Result<Self> {
49        Ok(DefaultValue::UseDefault)
50    }
51    fn from_expr(expr: &syn::Expr) -> darling::Result<Self> {
52        Ok(DefaultValue::Value(expr.clone()))
53    }
54}

56#[derive(FromField)]
57#[darling(attributes(oudia))]
58struct FieldOpts {
59    #[darling(rename = "type")]
60    kind: OuDiaType,
61    alias: Option<String>,
62    parse_fn: Option<syn::Expr>,
63    silence_fn: Option<syn::Expr>,
64    serialize_fn: Option<syn::Expr>,
65    default: Option<DefaultValue>,
66}

68enum OuterType<'a> {
69    Option(&'a Type),
70    Vec(&'a Type),
71    Plain(&'a Type),
72}
73
74impl<'a> OuterType<'a> {
75    fn into_inner(&self) -> &Type {
76        match *self {
77            Self::Option(ty) => ty,
78            Self::Vec(ty) => ty,
79            Self::Plain(ty) => ty,
80        }
81    }
82}

84fn inspect_type(ty: &Type) -> OuterType<'_> {
85    if let Type::Path(TypePath {
86        qself: None, path, ..
87    }) = ty
88        && let Some(segment) = path.segments.last()
89        && let PathArguments::AngleBracketed(args) = &segment.arguments
90        && let Some(GenericArgument::Type(inner_ty)) = args.args.first()
91    {
92        if segment.ident == "Option" {
93            return OuterType::Option(inner_ty);
94        } else if segment.ident == "Vec" {
95            return OuterType::Vec(inner_ty);
96        }
97    }
98
99    OuterType::Plain(ty)
100}

102fn is_bool_type(ty: &Type) -> bool {
103    if let Type::Path(TypePath {
104        qself: None, path, ..
105    }) = ty
106    {
107        return path.is_ident("bool");
108    }
109    false
110}

112fn doc_aliases(names: &[String], skip: &str) -> Vec<syn::Attribute> {
113    let names: Vec<&String> = names.iter().filter(|name| name.as_str() != skip).collect();
114    if names.is_empty() {
115        return Vec::new();
116    }
117
118    let first = names[0];
119    let rest = &names[1..];
120    let doc_desc = quote! {
121        concat!{
122            "\n\nAlso known as `",
123            #first,
124            #( "`, `", #rest, )*
125            "`."
126        }
127    };
128
129    let mut attrs = vec![parse_quote!(#[doc = #doc_desc])];
130    for name in names {
131        attrs.push(parse_quote!(#[doc(alias = #name)]));
132    }
133    attrs
134}

136#[proc_macro_attribute]
137pub fn oudia(args: TokenStream, input: TokenStream) -> TokenStream {
138    let args = parse_macro_input!(args as StructArgs);
139    let mut input = parse_macro_input!(input as DeriveInput);
140
141    let struct_ident = input.ident.clone();
142    let (impl_generics, ty_generics, where_clause) = input.generics.split_for_impl();
143
144    let struct_oudia_key = args.key.clone().unwrap_or_else(|| struct_ident.to_string());
145
146    let mut struct_names = Vec::new();
147    if let Some(key) = &args.key {
148        if !key.is_empty() {
149            struct_names.push(key.clone());
150        }
151    }
152    if let Some(alias) = &args.alias {
153        struct_names.push(alias.clone());
154    }
155    input.attrs.extend(doc_aliases(&struct_names, &struct_ident.to_string()));
156
157    let mut initializers = proc_macro2::TokenStream::new();
158    let mut matchers = proc_macro2::TokenStream::new();
159    let mut silencers = proc_macro2::TokenStream::new();
160    let mut assembler = proc_macro2::TokenStream::new();
161    let mut fixups = proc_macro2::TokenStream::new();
162    let mut serializers = proc_macro2::TokenStream::new();
163
164    // Fields used to synthesize a `Default` impl. We only emit one when every
165    // field has a resolvable default.
166    let mut all_fields_default = true;
167    let mut default_fields = proc_macro2::TokenStream::new();
168
169    {
170        let Data::Struct(data_struct) = &mut input.data else {
171            return Error::new_spanned(&input.ident, "You should apply `oudia` on a struct")
172                .to_compile_error()
173                .into();
174        };
175
176        for field in &mut data_struct.fields {
177            let opts = match FieldOpts::from_field(&*field) {
178                Ok(opts) => opts,
179                Err(e) => {
180                    return Error::new_spanned(
181                        &*field,
182                        format!("failed to parse oudia attributes on field: {e}"),
183                    )
184                    .to_compile_error()
185                    .into();
186                }
187            };
188
189            let field_ident = field.ident.as_ref();
190            let outer = inspect_type(&field.ty);
191            let inner_ty = outer.into_inner();
192            let is_bool = is_bool_type(inner_ty);
193            let has_default = opts.default.is_some();
194
195            // The value used for both deserialization fallback and the `Default` impl.
196            let default_expr = match &opts.default {
197                Some(DefaultValue::UseDefault) => Some(quote! { Default::default() }),
198                Some(DefaultValue::Value(expr)) => Some(quote! { #expr }),
199                None => match &outer {
200                    OuterType::Option(_) => Some(quote! { None }),
201                    OuterType::Vec(_) => Some(quote! { Vec::new() }),
202                    OuterType::Plain(_) => None,
203                },
204            };
205
206            if let (Some(ident), Some(expr)) = (field_ident, &default_expr) {
207                default_fields.extend(quote! {
208                    #ident: #expr,
209                });
210            } else {
211                all_fields_default = false;
212            }
213
214            if let Some(parse_fn) = &opts.parse_fn {
215                initializers.extend(quote! {
216                    let #field_ident = (#parse_fn)(input)?;
217                });
218            } else {
219                // How to parse a single string value (a `&str`) into `inner_ty`.
220                let parse_value = |value: &syn::Expr| {
221                    if is_bool {
222                        quote! { (#value == "1") }
223                    } else {
224                        quote! { #value.parse::<#inner_ty>()? }
225                    }
226                };
227
228                // Uniform initializer: every field accumulates into `Option<T>`
229                // or `Vec<T>`; defaults are applied later, in the fixups.
230                match (&outer, &opts.kind) {
231                    (
232                        OuterType::Option(_) | OuterType::Plain(_),
233                        OuDiaType::SingleStruct(..) | OuDiaType::SinglePairSingleEntry(..),
234                    ) => {
235                        initializers.extend(quote! {
236                            let mut #field_ident: Option<#inner_ty> = None;
237                        });
238                    }
239                    (
240                        OuterType::Vec(_),
241                        OuDiaType::ManyStructs(..)
242                        | OuDiaType::SinglePairManyEntries(..)
243                        | OuDiaType::SingleStructManyEntries(..)
244                        | OuDiaType::TwinStructMultipleEntries { .. },
245                    ) => {
246                        initializers.extend(quote! {
247                            let mut #field_ident: Vec<#inner_ty> = Vec::new();
248                        });
249                    }
250                    _ => {
251                        return Error::new_spanned(
252                            &field.ty,
253                            "OuDia type don't match with field type.",
254                        )
255                        .to_compile_error()
256                        .into();
257                    }
258                }
259
260                matchers.extend(match &opts.kind {
261                    OuDiaType::SingleStruct(key) => quote! {
262                        crate::ast::Structure::Struct(k, v) if k == #key => {
263                            #field_ident = Some(<#inner_ty as crate::OuDiaIo>::from_structure(v)?);
264                        }
265                    },
266                    OuDiaType::SinglePairSingleEntry(key) => {
267                        let first = parse_quote!(first);
268                        let parse_expr = parse_value(&first);
269                        quote! {
270                            crate::ast::Structure::Pair(k, v) if k == #key && let Some(first) = v.first() => {
271                                #field_ident = Some(#parse_expr);
272                            }
273                        }
274                    }
275                    OuDiaType::ManyStructs(key) => quote! {
276                        crate::ast::Structure::Struct(k, v) if k == #key => {
277                            #field_ident.push(<#inner_ty as crate::OuDiaIo>::from_structure(v)?);
278                        }
279                    },
280                    OuDiaType::SinglePairManyEntries(key) => {
281                        let val = parse_quote!(val);
282                        let parse_expr = parse_value(&val);
283                        quote! {
284                            crate::ast::Structure::Pair(k, v) if k == #key => for val in v {
285                                #field_ident.push(#parse_expr);
286                            }
287                        }
288                    }
289                    OuDiaType::SingleStructManyEntries(key) => quote! {
290                        crate::ast::Structure::Struct(k, v) if k == #key => for node in v {
291                            match node {
292                                crate::ast::Structure::Struct(k2, v2) if k2 == <#inner_ty as crate::OuDiaIo>::OUDIA_KEY => {
293                                    #field_ident.push(<#inner_ty as crate::OuDiaIo>::from_structure(v2)?);
294                                }
295                                // Unknown field. Give error in log.
296                                crate::ast::Structure::Pair(k2, _) | crate::ast::Structure::Struct(k2, _) => {
297                                    log::debug!("Encountered unknown field `{k2}` when parsing `{}`", std::any::type_name::<Self>());
298                                }
299                            }
300                        }
301                    },
302                    OuDiaType::TwinStructMultipleEntries { first, second } => quote! {
303                        crate::ast::Structure::Struct(k, v) if k == #first || k == #second => for node in v {
304                            match node {
305                                crate::ast::Structure::Struct(k2, v2) if k2 == <#inner_ty as crate::OuDiaIo>::OUDIA_KEY => {
306                                    #field_ident.push(<#inner_ty as crate::OuDiaIo>::from_structure(v2)?);
307                                }
308                                // Unknown field. Give error in log.
309                                crate::ast::Structure::Pair(k2, _) | crate::ast::Structure::Struct(k2, _) => {
310                                    log::debug!("Encountered unknown field `{k2}` when parsing `{}`", std::any::type_name::<Self>());
311                                }
312                            }
313                        }
314                    },
315                });
316
317                // Fixups are the only place defaults / required-ness differ.
318                match (&outer, &opts.kind) {
319                    (
320                        OuterType::Plain(_),
321                        OuDiaType::SingleStruct(..) | OuDiaType::SinglePairSingleEntry(..),
322                    ) => {
323                        if has_default {
324                            let expr = default_expr.as_ref().unwrap();
325                            fixups.extend(quote! {
326                                let #field_ident = #field_ident.unwrap_or(#expr);
327                            });
328                        } else {
329                            let missing = match &opts.kind {
330                                OuDiaType::SingleStruct(key) => quote! { #key },
331                                OuDiaType::SinglePairSingleEntry(key) => quote! { #key },
332                                _ => unreachable!(),
333                            };
334                            fixups.extend(quote! {
335                                let Some(#field_ident) = #field_ident else {
336                                    return Err(crate::IrConversionError::MissingField {
337                                        processing: std::any::type_name::<Self>(),
338                                        missing: #missing,
339                                    });
340                                };
341                            });
342                        }
343                    }
344                    (OuterType::Vec(_), _) => {
345                        if has_default {
346                            let expr = default_expr.as_ref().unwrap();
347                            fixups.extend(quote! {
348                                if #field_ident.is_empty() {
349                                    #field_ident = #expr;
350                                }
351                            });
352                        }
353                    }
354                    _ => {}
355                }
356            }
357
358            if let Some(silence_fn) = &opts.silence_fn {
359                silencers.extend(quote! {
360                    crate::ast::Structure::Pair(k, _) | crate::ast::Structure::Struct(k, _) if (#silence_fn)(k)=> {}
361                })
362            }
363
364            if let Some(serialize_fn) = &opts.serialize_fn {
365                serializers.extend(quote! {
366                    (#serialize_fn)(&mut __oudia_items);
367                });
368            } else {
369                serializers.extend(match (&outer, &opts.kind) {
370                    (OuterType::Option(_), OuDiaType::SinglePairSingleEntry(key)) => {
371                        if is_bool {
372                            quote! {
373                                if let Some(value) = &self.#field_ident {
374                                    __oudia_items.push(crate::pair!(#key => if *value { "1" } else { "0" }));
375                                }
376                            }
377                        } else {
378                            quote! {
379                                if let Some(value) = &self.#field_ident {
380                                    __oudia_items.push(crate::pair!(#key => value.to_string()));
381                                }
382                            }
383                        }
384                    }
385                    (OuterType::Plain(_), OuDiaType::SinglePairSingleEntry(key)) => {
386                        if is_bool {
387                            quote! {
388                                __oudia_items.push(crate::pair!(#key => if self.#field_ident { "1" } else { "0" }));
389                            }
390                        } else {
391                            quote! {
392                                __oudia_items.push(crate::pair!(#key => self.#field_ident.to_string()));
393                            }
394                        }
395                    }
396                    (OuterType::Vec(_), OuDiaType::SinglePairManyEntries(key)) => {
397                        if is_bool {
398                            quote! {
399                                __oudia_items.push(crate::pair!(#key => .. self.#field_ident.iter().map(|value| if *value { "1" } else { "0" })));
400                            }
401                        } else {
402                            quote! {
403                                __oudia_items.push(crate::pair!(#key => .. self.#field_ident.iter().map(|value| value.to_string())));
404                            }
405                        }
406                    }
407                    (OuterType::Option(_), OuDiaType::SingleStruct(..)) => quote! {
408                        if let Some(value) = &self.#field_ident {
409                            __oudia_items.push(value.to_structure());
410                        }
411                    },
412                    (OuterType::Plain(_), OuDiaType::SingleStruct(..)) => quote! {
413                        __oudia_items.push(self.#field_ident.to_structure());
414                    },
415                    (OuterType::Vec(_), OuDiaType::ManyStructs(..)) => quote! {
416                        __oudia_items.extend(self.#field_ident.iter().map(|value| value.to_structure()));
417                    },
418                    (OuterType::Vec(_), OuDiaType::SingleStructManyEntries(key)) => quote! {
419                        __oudia_items.push(crate::structure!(#key => .. self.#field_ident.iter().map(|value| value.to_structure())));
420                    },
421                    // We don't have a generic way to split the values back into the two
422                    // keys, so serialize everything under the first key for now.
423                    (OuterType::Vec(_), OuDiaType::TwinStructMultipleEntries { first, .. }) => quote! {
424                        __oudia_items.push(crate::structure!(#first => .. self.#field_ident.iter().map(|value| value.to_structure())));
425                    },
426                    _ => unreachable!(),
427                });
428            }
429
430            assembler.extend(quote! {
431                #field_ident,
432            });
433
434            field.attrs.retain(|attr| !attr.path().is_ident("oudia"));
435
436            let mut field_names = Vec::new();
437            if let Some(key) = match &opts.kind {
438                OuDiaType::SinglePairSingleEntry(key)
439                | OuDiaType::SinglePairManyEntries(key)
440                | OuDiaType::SingleStruct(key)
441                | OuDiaType::ManyStructs(key)
442                | OuDiaType::SingleStructManyEntries(key) => Some(key.clone()),
443                OuDiaType::TwinStructMultipleEntries { .. } => None,
444            } {
445                field_names.push(key);
446            }
447            if let Some(alias) = &opts.alias {
448                field_names.push(alias.clone());
449            }
450            field.attrs.extend(doc_aliases(
451                &field_names,
452                &field_ident.map(|ident| ident.to_string()).unwrap_or_default(),
453            ));
454        }
455    }
456
457    let default_impl = if all_fields_default {
458        quote! {
459            impl #impl_generics Default for #struct_ident #ty_generics #where_clause {
460                fn default() -> Self {
461                    Self {
462                        #default_fields
463                    }
464                }
465            }
466        }
467    } else {
468        proc_macro2::TokenStream::new()
469    };
470
471    let expanded = quote! {
472        #input
473
474        #default_impl
475
476        impl #impl_generics crate::OuDiaIo for #struct_ident #ty_generics #where_clause {
477            const OUDIA_KEY: &'static str = #struct_oudia_key;
478
479            fn from_structure(input: &[crate::ast::Structure<'_>]) -> Result<Self, crate::IrConversionError> {
480                #initializers
481                for node in input {
482                    match node {
483                        #matchers
484                        #silencers
485                        // Unknown field. Give error in log.
486                        crate::ast::Structure::Pair(k, _) | crate::ast::Structure::Struct(k, _) => {
487                            log::debug!("Encountered unknown field `{k}` when parsing `{}`", std::any::type_name::<Self>());
488                        }
489                    }
490                }
491                #fixups
492                Ok(Self { #assembler })
493            }
494
495            fn to_structure(&self) -> crate::ast::Structure<'static> {
496                let mut __oudia_items: Vec<crate::ast::Structure<'static>> = Vec::new();
497                #serializers
498                crate::ast::Structure::Struct(Self::OUDIA_KEY.into(), __oudia_items)
499            }
500        }
501    };
502
503    TokenStream::from(expanded)
504}
