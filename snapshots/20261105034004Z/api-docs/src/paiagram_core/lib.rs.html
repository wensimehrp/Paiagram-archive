43pub trait Key: Clone + Copy {
44    /// Return the key in bits
45    fn to_bits(self) -> u64;
46    /// Return the creation time of the key
47    fn creation_time(self) -> std::time::SystemTime {
48        crate::key_gen::creation_time(self.to_bits())
49    }
50    /// Return the id of the client that allocated the key
51    fn client(self) -> u8 {
52        crate::key_gen::client(self.to_bits())
53    }
54    /// Return the generation
55    fn generation(self) -> u16 {
56        crate::key_gen::generation(self.to_bits())
57    }
58}

170make_type! {
171    /// An interval is a directed edge, so the ordered pair of
172    /// its endpoints, [`IntervalKey`], uniquely identifies it. Parallel edges are not allowed.
173    Interval,
174    key = graph::SortedKey,
175    data {
176        /// The nodes between the two nodes this interval connects.
177        /// Stored in high -> low order
178        intermediate_nodes: EcoVec<LonLat>,
179        /// The length of the interval. If the length is None, then it is calculated from
180        /// the intermediate nodes and the endpoint node positions.
181        length: Option<NonZeroU32>,
182        direction: graph::IntervalDirection,
183    }
184    cache {
185        /// trips passing this interval
186        trips: EcoVec<TripKey>,
187    }
188}

190impl Interval {
191    /// The full polyline of the interval, from its high-key endpoint through the
192    /// intermediate positions to its low-key endpoint.
193    pub fn positions(
194        &self,
195        IntervalKey { hi, lo }: IntervalKey,
196        nodes: &NodeCollection,
197    ) -> impl Iterator<Item = LonLat> + DoubleEndedIterator {
198        chain!(
199            nodes.get(&hi).map(|wfc| wfc.pos),
200            self.intermediate_nodes.iter().cloned(),
201            nodes.get(&lo).map(|wfc| wfc.pos)
202        )
203    }
204}

228#[derive(Serialize, Deserialize, Default, Clone, Debug, PartialEq)]
229pub struct World {
230    pub trips: TripCollection,
231    pub vehicles: VehicleCollection,
232    pub stations: StationCollection,
233    pub graph: graph::Graph,
234    pub service_classes: ServiceClassCollection,
235    pub routes: RouteCollection,
236}
237
238impl World {
239    /// Insert every entity from `other`, keeping the entities already present in `self`.
240    fn merged(mut self, other: &Self) -> Self {
241        puffin::profile_function!();
242        for (key, value) in &other.vehicles {
243            self.vehicles.insert(*key, value.clone());
244        }
245        for (key, value) in &other.stations {
246            self.stations.insert(*key, value.clone());
247        }
248        for (key, value) in &other.graph.nodes {
249            self.graph.nodes.insert(*key, value.clone());
250        }
251        for (key, value) in &other.graph.intervals {
252            self.graph.insert_interval(*key, value.clone());
253        }
254        for (key, value) in &other.trips {
255            self.trips.insert(*key, value.clone());
256        }
257        for (key, value) in &other.service_classes {
258            self.service_classes.insert(*key, value.clone());
259        }
260        for (key, value) in &other.routes {
261            self.routes.insert(*key, value.clone());
262        }
263        self
264    }
265
266    /// Get the serivce class stroke style of a trip, or return the default style
267    pub fn get_service_class_stroke_or_default(
268        &self,
269        service_class_key: Option<ServiceClassKey>,
270    ) -> egui::Stroke {
271        service_class_key
272            .and_then(|key| self.service_classes.get(&key))
273            .map_or_default(|wfc| wfc.style)
274            .to_egui_stroke()
275    }
276}

278/// The truth of the application. This structure holds a write-only log and a set of undos and
279/// redos, as well as the current world.
280///
281/// The source is not clonable, and should not be cloned anyways.
282pub struct Source {
283    /// The current world
284    world: World,
285    history: Vec<World>,
286    /// the amount of redos available at this point
287    redo_len: usize,
288    revision: u64,
289    spatial: SpatialCache,
290    spatial_tx: mpsc::Sender<(u64, SpatialCache)>,
291    spatial_rx: mpsc::Receiver<(u64, SpatialCache)>,
292}
293
294impl std::ops::Deref for Source {
295    type Target = World;
296    fn deref(&self) -> &Self::Target {
297        &self.world
298    }
299}
300
301impl Source {
302    /// Changes after every successful command, undo, and redo.
303    pub fn revision(&self) -> u64 {
304        self.revision
305    }
306    /// The source's spatial cache
307    pub fn spatial(&self) -> &SpatialCache {
308        &self.spatial
309    }
310    /// Available history
311    pub fn history(&self) -> &[World] {
312        &self.history
313    }
314    /// The underlying world
315    pub fn world(&self) -> &World {
316        &self.world
317    }
318
319    /// Saves the current world as the previous frame's. Call once at the start of each frame.
320    pub fn begin_frame(&mut self) {
321        puffin::profile_function!();
322        if let Ok((build_idx, new_spatial)) = self.spatial_rx.try_recv() {
323            if build_idx == self.revision() {
324                self.spatial = new_spatial;
325            } else {
326                self.spawn_spatial_build();
327            }
328        }
329    }
330
331    /// Build the spatial cache on the rayon pool.
332    fn spawn_spatial_build(&self) {
333        let revision = self.revision;
334        let world = self.world.clone();
335        let sender = self.spatial_tx.clone();
336        rayon::spawn(move || {
337            puffin::profile_scope!("spatial_cache_build");
338            let cache = SpatialCache::build(&world);
339            let _ = sender.send((revision, cache));
340        });
341    }
342
343    /// Mutates the world and makes a backup
344    pub fn mutate(&mut self, f: impl FnOnce(World) -> Result<World, String>) -> Result<(), String> {
345        puffin::profile_function!();
346        let new_world = f(self.world.clone())?;
347
348        if self.redo_len > 0 {
349            let active_len = self.history.len() - self.redo_len;
350            self.history.truncate(active_len);
351            self.redo_len = 0;
352        }
353
354        let old = self.world.clone();
355        self.history.push(old);
356        let spatial_dirty = self.world.update(|_old_world| new_world);
357        self.revision += 1;
358        if spatial_dirty {
359            self.spawn_spatial_build();
360        }
361        Ok(())
362    }
363
364    /// Applies an imported world, either on top of the current one or replacing it.
365    pub fn import(&mut self, world: World, mode: ImportMode) -> Result<(), String> {
366        puffin::profile_function!();
367        self.mutate(move |existing| {
368            Ok(match mode {
369                ImportMode::Merge => existing.merged(&world),
370                ImportMode::Replace => world,
371            })
372        })
373    }
374
375    pub fn undoable(&self) -> bool {
376        self.history.len() > self.redo_len
377    }
378    pub fn redoable(&self) -> bool {
379        self.redo_len > 0
380    }
381
382    pub fn undo(&mut self) -> Result<(), String> {
383        puffin::profile_function!();
384        if !self.undoable() {
385            return Err("No undos available".into());
386        }
387        let target_idx = self.history.len() - 1 - self.redo_len;
388        std::mem::swap(&mut self.world, &mut self.history[target_idx]);
389        self.redo_len += 1;
390        self.revision += 1;
391        self.spawn_spatial_build();
392        Ok(())
393    }
394
395    pub fn redo(&mut self) -> Result<(), String> {
396        puffin::profile_function!();
397        if !self.redoable() {
398            return Err("No redos available".into());
399        }
400        self.redo_len -= 1;
401        let target_idx = self.history.len() - 1 - self.redo_len;
402        std::mem::swap(&mut self.world, &mut self.history[target_idx]);
403        self.revision += 1;
404        self.spawn_spatial_build();
405        Ok(())
406    }
407}

409impl Source {
410    pub fn new() -> Self {
411        let (spatial_tx, spatial_rx) = mpsc::channel();
412        Self {
413            world: World::default(),
414            history: Vec::new(),
415            redo_len: 0,
416            revision: 0,
417            spatial: SpatialCache::default(),
418            spatial_tx,
419            spatial_rx,
420        }
421    }
