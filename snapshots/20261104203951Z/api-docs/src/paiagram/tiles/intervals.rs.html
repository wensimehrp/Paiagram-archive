1use egui::*;
2use paiagram_core::{IntervalDirection, IntervalKey};
3use serde::{Deserialize, Serialize};
4
5use crate::App;
6use crate::tiles::trip::TripTile;
7
8#[derive(Serialize, Deserialize, Clone, Default)]
9pub struct IntervalsTile {
10    focused: Option<IntervalKey>,
11}
12
13impl PartialEq for IntervalsTile {
14    fn eq(&self, _other: &Self) -> bool {
15        true
16    }
17}
18
19impl super::Tile for IntervalsTile {
20    const NAME: &'static str = "Intervals";
21    fn main_display(&mut self, app: &mut App, ui: &mut Ui) {
22        puffin::profile_function!();
23        Panel::left("interval left panel").show(ui, |ui| {
24            puffin::profile_scope!("interval_list");
25            let spacing = ui.spacing().interact_size.y;
26            let total_rows = app.graph.intervals().len();
27            let scroll_area = ScrollArea::vertical().auto_shrink(false);
28            scroll_area.show_rows(ui, spacing, total_rows, |ui, row_range| {
29                ui.with_layout(Layout::top_down_justified(Align::Min), |ui| {
30                    for (&interval_key, interval) in app
31                        .source
32                        .graph
33                        .intervals()
34                        .iter()
35                        .skip(row_range.start)
36                        .take(row_range.count())
37                    {
38                        let IntervalKey { hi, lo } = interval_key;
39                        let hi_text =
40                            app.graph.nodes().get(&hi).map_or("Unknown", |wfc| wfc.name.as_str());
41                        let lo_text =
42                            app.graph.nodes().get(&lo).map_or("Unknown", |wfc| wfc.name.as_str());
43                        let di_text = match interval.direction {
44                            IntervalDirection::Both => "<=>",
45                            IntervalDirection::HiToLo => "->",
46                            IntervalDirection::LoToHi => "<-",
47                        };
48                        let response = Button::selectable(
49                            self.focused == Some(interval_key),
50                            format!("{hi_text} {di_text} {lo_text}"),
51                        )
52                        .truncate()
53                        .atom_ui(ui);
54                        if response.clicked() {
55                            self.focused = Some(interval_key);
56                        }
57                    }
58                });
59            });
60        });
61        CentralPanel::default_margins().show(ui, |ui| {
62            puffin::profile_scope!("interval_details");
63            let Some(interval) = self.focused else {
64                ui.vertical_centered_justified(|ui| ui.heading("Nothing focused"));
65                return;
66            };
67            let Some(interval) = app.source.graph.intervals().get(&interval) else {
68                ui.vertical_centered_justified(|ui| ui.heading("Interval does not exist"));
69                return;
70            };
71            ScrollArea::vertical().auto_shrink(false).show(ui, |ui| {
72                for trip_key in &interval.cache.trips {
73                    let Some(trip) = app.trips.get(trip_key) else {
74                        continue;
75                    };
76                    if ui.button(trip.name.as_str()).clicked() {
77                        app.open_or_focus(TripTile::new(*trip_key));
78                    }
79                }
80            });
81        });
82    }
83}
