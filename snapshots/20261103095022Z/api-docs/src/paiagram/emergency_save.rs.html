2use std::panic::{self, PanicHookInfo};
3use std::path::Path;
4use std::sync::Arc;

6use arc_swap::ArcSwapOption;
7use log::error;
8use paiagram_core::{SaveData, World};
9
10/// A backup for the current world. The content stored in this variable would be serialized and
11/// stored upon a panic.
12static WORLD_BACKUP: ArcSwapOption<World> = ArcSwapOption::const_empty();
13
14/// Updates the backup emergency save at [`WORLD_BACKUP`].
15pub fn update_emergency_save(new_world: World) {
16    WORLD_BACKUP.store(Some(Arc::new(new_world)));
17}
18
19/// Set up the emergency save pipeline
20pub fn setup_emergency_save() {
21    let default_hook = panic::take_hook();
22    panic::set_hook(Box::new(move |info| {
23        _setup_emergency_save(info);
24        default_hook(info);
25    }));
26}
27
28fn _setup_emergency_save(panic_info: &PanicHookInfo<'_>) {
29    error!("An unexpected error occurred. The app has panicked and crashed.");
30    error!("Panic info: {:?}", panic_info);
31
32    let Some(world_guard) = WORLD_BACKUP.load_full() else {
33        error!("Failed to create backup: no world stored in backup");
34        return;
35    };
36
37    let world: &World = &world_guard;
38    error!("Attempting to create emergency save...");
39
40    #[allow(unused_mut)]
41    let mut save = SaveData::from(world.clone());
42
43    #[cfg(target_arch = "wasm32")]
44    {
45        use paiagram_rw::ExportObject;
46        let filename = format!(
47            "paiagram-emergency-{}.{}",
48            jiff::Timestamp::now().to_string().replace(':', "-"),
49            save.extension(),
50        );
51        match save.write_to_vec(save.default_buf()) {
52            Ok(bytes) => match paiagram_rw::save::download_file(&filename, &bytes) {
53                Ok(()) => error!("Emergency save downloaded as `{filename}`"),
54                Err(e) => error!("Failed to download emergency save: {e:?}"),
55            },
56            Err(e) => error!("Failed to serialize emergency save: {e}"),
57        }
58        show_dialog(Path::new(&filename));
59    }
60
61    #[cfg(not(target_arch = "wasm32"))]
62    {
63        use paiagram_rw::{SaveCategory, WriteResult};
64        let (tx, rx) = std::sync::mpsc::channel();
65
66        SaveCategory::Emergency.save(save, tx);
67        match rx.recv_timeout(std::time::Duration::from_secs(30)) {
68            Ok(result) => match result {
69                WriteResult::Aborted => error!("Write was aborted"),
70                WriteResult::Error(e) => error!("Error while writing file: {}", e),
71                WriteResult::Success(path) => {
72                    error!("Successfully wrote to `{}`", path.display());
73                    show_dialog(&path);
74                }
75            },
76            Err(e) => {
77                error!("Failed to save file: {}", e)
78            }
79        }
80    }
81}
82
83fn show_dialog(path: &Path) {
84    let desc = format!(
85        "The app has crashed. An emergency save at {} has been created.",
86        path.display()
87    );
88    let title = "Emergency Save";
89    cfg_select! {
90        target_arch = "wasm32" => wasm_bindgen_futures::spawn_local(async move {
91            rfd::AsyncMessageDialog::new()
92                .set_title(title)
93                .set_description(desc)
94                .set_level(rfd::MessageLevel::Error)
95                .show()
96                .await;
97        }),
98        _ => {
99            rfd::MessageDialog::new()
100                .set_title(title)
101                .set_description(desc)
102                .set_level(rfd::MessageLevel::Error)
103                .show();
104        }
105    }
106}
