15/// Decodes the embedded trip icon and uploads it as an egui texture. Every trip uses the same
16/// texture for now, so egui batches all icon quads into one draw call.
17pub fn load_trip_icon(ctx: &Context) -> TextureHandle {
18    let decoded = image::load_from_memory(include_bytes!("../../../../../assets/raicho_485.png"))
19        .expect("the embedded trip icon must be a valid PNG")
20        .into_rgba8();
21
22    // The icon is drawn from one square quad, so pad it with transparent pixels instead of
23    // stretching it. This preserves the original aspect ratio.
24    let side = decoded.width().max(decoded.height());
25    let mut rgba = image::RgbaImage::new(side, side);
26    let x = i64::from((side - decoded.width()) / 2);
27    let y = i64::from((side - decoded.height()) / 2);
28    image::imageops::overlay(&mut rgba, &decoded, x, y);
29
30    let image = ColorImage::from_rgba_unmultiplied([side as usize, side as usize], rgba.as_raw());
31    ctx.load_texture("graph_trip_icon", image, TextureOptions::LINEAR)
32}

34pub fn render_map_contents(
35    app: &App,
36    trip_icon: TextureId,
37    ui: &mut Ui,
38    response: &Response,
39    projector: &Projector,
40    map_memory: &MapMemory,
41) {
42    puffin::profile_function!();
43
44    const DISPLAY_ZOOM_THRESHOLD: f64 = 8.0;
45    let strength = ui.animate_bool_responsive(
46        ui.scope_id().with("text display animation"),
47        map_memory.zoom() > DISPLAY_ZOOM_THRESHOLD,
48    );
49
50    let pos_to_lon_lat = |pos: Pos2| -> LonLat {
51        let coor = projector.unproject(pos.to_vec2());
52        Wgs84LonLat {
53            lon: coor.x(),
54            lat: coor.y(),
55        }
56        .into()
57    };
58
59    let lon_lat_to_pos = |coor: LonLat| {
60        let wgs84_lon_lat: Wgs84LonLat = coor.into();
61        projector.project(lon_lat(wgs84_lon_lat.lon, wgs84_lon_lat.lat)).to_pos2()
62    };
63
64    let coor_min = pos_to_lon_lat(response.rect.left_top());
65    let coor_max = pos_to_lon_lat(response.rect.right_bottom());
66
67    // Intervals
68    let mut callback = GraphCallback::new();
69    {
70        puffin::profile_scope!("map_intervals");
71        callback.populate_uniforms(response.rect);
72        for IntervalRTreeObject {
73            interval_key: _,
74            nodes,
75        } in app.spatial().intervals.get(coor_min, coor_max)
76        {
77            let Some((last, rest)) = nodes.split_last() else {
78                continue;
79            };
80            callback.interval_points.extend(
81                rest.iter()
82                    .copied()
83                    .map(lon_lat_to_pos)
84                    .map(|pos| IntervalPoint::new((pos - response.rect.left_top()).into(), 1)),
85            );
86            let pos = lon_lat_to_pos(*last) - response.rect.left_top();
87            callback.interval_points.push(IntervalPoint::new(pos.into(), 0));
88        }
89    }
90
91    // Nodes
92    {
93        puffin::profile_scope!("draw_nodes");
94        let node_text_color = ui.visuals().strong_text_color().gamma_multiply(strength);
95        for NodeRTreeObject { node_key, coor } in app.spatial().nodes.get(coor_min, coor_max) {
96            let Some(node) = app.graph.nodes().get(&node_key) else {
97                continue;
98            };
99            let draw_pos = lon_lat_to_pos(coor);
100            callback.node_points.push(NodePoint::new(
101                (draw_pos - response.rect.left_top()).into(),
102                u32::from_le_bytes(Color32::GRAY.to_array()),
103            ));
104            if strength > 0.1 {
105                ui.painter().text(
106                    draw_pos + vec2(0.0, 4.0),
107                    Align2::CENTER_TOP,
108                    node.name.as_str(),
109                    FontId::proportional(13.0),
110                    node_text_color,
111                );
112            }
113        }
114    }
115
116    // Trip legs
117    {
118        let tint = Color32::WHITE.gamma_multiply(strength);
119        puffin::profile_scope!("draw_trip_legs");
120        let map_trip_legs = app
121            .spatial()
122            .trip_legs
123            .get(
124                coor_min,
125                coor_max,
126                TimetableTime(app.timer.seconds() as i32),
127            )
128            .filter_map(|trip_leg| {
129                app.trips.get(&trip_leg.trip_key).zip(trip_leg_current_pos(
130                    trip_leg,
131                    app.timer.seconds(),
132                    lon_lat_to_pos,
133                ))
134            })
135            .map(|(trip, draw_pos)| {
136                (
137                    trip,
138                    draw_pos,
139                    app.get_service_class_stroke_or_default(trip.service_class).color,
140                )
141            });
142        if strength > 0.1 {
143            ui.painter().add(callback.paint_callback(response.rect));
144            // simply use egui's image features
145            for (trip, draw_pos, fill) in map_trip_legs {
146                ui.painter().image(
147                    trip_icon,
148                    Rect::from_center_size(draw_pos, Vec2::splat(ICON_HALF_SIZE * 2.0)),
149                    Rect::from_min_max(pos2(0.0, 0.0), pos2(1.0, 1.0)),
150                    tint,
151                );
152                ui.painter().text(
153                    draw_pos + Vec2::new(0.0, -ICON_HALF_SIZE),
154                    Align2::CENTER_BOTTOM,
155                    trip.name.as_str(),
156                    FontId::proportional(13.0),
157                    fill.gamma_multiply(strength),
158                );
159            }
160        } else {
161            for (_, draw_pos, fill) in map_trip_legs {
162                callback.node_points.push(NodePoint::new(
163                    (draw_pos - response.rect.left_top()).into(),
164                    u32::from_le_bytes(fill.to_array()),
165                ));
166            }
167            ui.painter().add(callback.paint_callback(response.rect));
168        }
169    }
170}

172pub fn trip_leg_current_pos(
173    TripLegRTreeObject {
174        interval,
175        is_hi_to_lo,
176        trip_key: _,
177        curr_arr: _,
178        curr_dep,
179        next_arr,
180    }: TripLegRTreeObject,
181    current_secs: f64,
182    lon_lat_to_pos: impl Fn(LonLat) -> Pos2,
183) -> Option<Pos2> {
184    let curr_progress = {
185        let t1 = curr_dep.0 as f64;
186        let t2 = next_arr.0 as f64;
187        // curr arr time is simply out of range.
188        // Just clamp the value so it stays at the station...
189        inverse_lerp(t1..=t2, current_secs).unwrap_or(0.0).at_least(0.0) as f32
190    };
191    // current_secs is a float while t1 and t2 are i32
192    // this leaves a ~1 in-app second gap that duplicates entries.
193    // simply remove the entry when progress > 1.0.
194    if curr_progress > 1.0 {
195        return None;
196    }
197    let curr_progress = if is_hi_to_lo {
198        curr_progress
199    } else {
200        1.0 - curr_progress
201    };
202    // Simple O(n) find
203    let prev_idx: usize = interval
204        .iter()
205        .enumerate()
206        .find_map(|(idx, &(_, interval_progress))| {
207            (curr_progress >= interval_progress).then_some(idx)
208        })
209        .unwrap_or(0);
210
211    let (prev_coor, prev_progress) = interval[prev_idx];
212    let (next_coor, next_progress) = interval[prev_idx + 1];
213    let normalized_progress =
214        inverse_lerp(prev_progress..=next_progress, curr_progress).unwrap_or(0.0);
215    let prev_pos = lon_lat_to_pos(prev_coor);
216    let next_pos = lon_lat_to_pos(next_coor);
217    Some(prev_pos.lerp(next_pos, normalized_progress))
218}
