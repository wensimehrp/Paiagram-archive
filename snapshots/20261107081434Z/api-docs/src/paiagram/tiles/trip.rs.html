52fn show_trip(tile: &mut TripTile, app: &mut App, ui: &mut Ui) {
53    puffin::profile_function!();
54
55    let get_trip_sch = |key| app.trips.get(&key).map(|wfc| (key, wfc, wfc.schedule.clone()));
56
57    let Some((trip_key, trip, base_schedule)) =
58        tile.trip_key.and_then(get_trip_sch).or_else(|| match &app.selected_items {
59            SelectedItems::Trips(t) => get_trip_sch(t.first().trip_key)
60                .map(|(key, wfc, sch)| (key, wfc, sch.with_entry_shift(t.first().editing_offset))),
61            SelectedItems::TripExtension((key, focused_pos)) => get_trip_sch(*key)
62                .map(|(key, wfc, sch)| (key, wfc, sch.with_pushed_entry(*focused_pos))),
63            _ => None,
64        })
65    else {
66        ui.heading("Select a trip to display the schedule");
67        select_a_trip(ui, &mut tile.trip_key, &app);
68        return;
69    };
70
71    // apply tile local editing offset to schedule
72    let visual_schedule = base_schedule.with_entry_shift(tile.editing_offset.map(|(a, b)| {
73        (
74            a,
75            TEstimate {
76                arr: TTime::ZERO,
77                dep: TTime::ZERO,
78            },
79            b,
80        )
81    }));
82
83    let trip_name = trip.name.clone();
84    let current_service_class = trip.service_class;
85
86    ui.horizontal(|ui| {
87        if let Some(buf) = &mut tile.name_edit_buf {
88            ui.text_edit_singleline(buf);
89            if ui.button("Finish").clicked() {
90                let _ = app.mutate(|mut world| {
91                    let trip = world.trips.get_mut(&trip_key).ok_or("wtf!")?;
92                    trip.name.clear();
93                    trip.name.push_str(buf);
94                    Ok(world)
95                });
96                tile.name_edit_buf = None;
97            }
98        } else {
99            ui.heading(trip_name.as_str());
100            if ui.button("Edit").clicked() {
101                tile.name_edit_buf = Some(trip_name.to_string());
102            }
103        }
104    });
105
106    ui.horizontal_wrapped(|ui| {
107        select_a_trip(ui, &mut tile.trip_key, &app);
108        let selected_text_atoms = current_service_class
109            .and_then(|key| app.service_classes.get(&key))
110            .map_or("Unknown".into_atoms(), |wfc| {
111                service_class_button_atoms(wfc).into_atoms()
112            });
113        let mut new_service_class = current_service_class;
114        ComboBox::from_id_salt("change current trip service class")
115            .selected_text(selected_text_atoms)
116            .show_ui(ui, |ui| {
117                ui.selectable_value(&mut new_service_class, None, "Unknown");
118                let spacing = ui.spacing().interact_size.y;
119                let total_rows = app.service_classes.len();
120                ScrollArea::vertical().show_rows(ui, spacing, total_rows, |ui, range| {
121                    let iter = app.service_classes.iter().skip(range.start).take(range.count());
122                    for (key, service_class) in iter {
123                        let atoms = service_class_button_atoms(service_class);
124                        ui.selectable_value(&mut new_service_class, Some(*key), atoms);
125                    }
126                });
127            });
128        if new_service_class != current_service_class {
129            let _ = app.mutate(|mut world| {
130                let trip = world.trips.get_mut(&trip_key).ok_or("wtf??")?;
131                trip.service_class = new_service_class;
132                Ok(world)
133            });
134        }
135    });
136
137    ScrollArea::vertical().auto_shrink(false).show(ui, |ui| {
138        Grid::new("trip ui").num_columns(3).striped(true).show(ui, |ui| {
139            ui.label("Station");
140            ui.label("Arrival/Pass");
141            ui.label("Departure");
142            ui.end_row();
143            // Remove button background
144            ui.visuals_mut().widgets.inactive.weak_bg_fill = Color32::TRANSPARENT;
145            let new_editing_offset = visual_schedule.estimates(&app.graph, |estimates| {
146                show_cursor_to_insert_at_idx(ui, 0);
147                let iter = estimates.into_iter().copied().enumerate();
148                iter.fold(None, |acc, (idx, (estimate, entry))| {
149                    let ret = row_ui(
150                        &visual_schedule,
151                        estimates,
152                        estimate,
153                        tile.editing_offset,
154                        entry,
155                        &app,
156                        ui,
157                    );
158                    ui.end_row();
159                    show_cursor_to_insert_at_idx(ui, idx + 1);
160                    acc.or(ret)
161                })
162            });
163            if new_editing_offset.is_none()
164                && let Some((id, offset)) = tile.editing_offset
165            {
166                let _ = app.mutate(|mut world| {
167                    let trip = world.trips.get_mut(&trip_key).ok_or("??")?;
168                    match offset {
169                        Either::Left(d) => trip.schedule.shift_arr_or_pass(id, d),
170                        Either::Right(d) => trip.schedule.shift_dep(id, d),
171                    }
172                    Ok(world)
173                });
174            };
175            tile.editing_offset = new_editing_offset;
176            if matches!(app.selected_items, SelectedItems::TripExtension(_))
177                && !ui.ui_contains_pointer()
178            {
179                ui.label("Extending...")
180                    .scroll_to_me_animation(Some(Align::Max), style::ScrollAnimation::none());
181            }
182        });
183    });
184}

186fn select_a_trip(ui: &mut Ui, existing_key: &mut Option<TripKey>, world: &World) {
187    let selected_text = existing_key.map_or("Current selected trip", |key| {
188        world.trips.get(&key).map_or("Unknown", |wfc| &wfc.name)
189    });
190    let combo_box = ComboBox::from_id_salt("Selected existing trip").selected_text(selected_text);
191    combo_box.show_ui(ui, |ui| {
192        ui.selectable_value(existing_key, None, "Current selected trip");
193        ScrollArea::vertical().show_rows(
194            ui,
195            ui.spacing().interact_size.y,
196            world.trips.len(),
197            |ui, range| {
198                for (key, trip) in world.trips.iter().skip(range.start).take(range.count()) {
199                    ui.selectable_value(existing_key, Some(*key), trip.name.as_str());
200                }
201            },
202        );
203    });
204}

206fn row_ui(
207    schedule: &TripSchedule,
208    estimates: &[(Option<TEstimate>, EstimateEntry)],
209    estimate: Option<TEstimate>,
210    edit_offset: Option<(TEntryId, Either<TDuration, TDuration>)>,
211    entry: EstimateEntry,
212    world: &World,
213    ui: &mut Ui,
214) -> Option<(TEntryId, Either<TDuration, TDuration>)> {
215    // TODO: change this to station name instead
216    let Some(node_name) = world.graph.nodes().get(&entry.node_key()).map(|wfc| wfc.name.as_str())
217    else {
218        ui.label("Unknown Station");
219        return None;
220    };
221    ui.label(node_name);
222    let Some(estimate) = estimate else {
223        ui.label("No estimate");
224        return None;
225    };
226    fn fmt_time_label(time: TimetableTime) -> WidgetText {
227        WidgetText::from(time.to_string()).text_style(TIMETABLTE_TEXT_STYLE.clone()).weak()
228    }
229    let EstimateEntry::Pinned(entry) = entry else {
230        ui.label(fmt_time_label(estimate.arr));
231        ui.label(RichText::from("⇂").weak());
232        return None;
233    };
234    let (mut arr_or_pass_offset, mut dep_offset) =
235        match edit_offset.and_then(|(id, offset)| (id == entry.id).then_some(offset)) {
236            Some(Either::Left(inner)) => (inner, TDuration::ZERO),
237            Some(Either::Right(inner)) => (TDuration::ZERO, inner),
238            None => (TDuration::ZERO, TDuration::ZERO),
239        };
240    let arr_or_pass_resp = match entry.arr_or_pass {
241        TravelMode::At(t) => ui.add(SeparateDisplayDragValue {
242            modified: &mut arr_or_pass_offset,
243            displayed: t,
244        }),
245        TravelMode::For(d) => ui.add(SeparateDisplayDragValue {
246            modified: &mut arr_or_pass_offset,
247            displayed: d,
248        }),
249        TravelMode::Flexible => ui.label(fmt_time_label(estimate.arr)),
250    };
251    let dep_resp = match entry.dep {
252        Some(TravelMode::At(t)) => ui.add(SeparateDisplayDragValue {
253            modified: &mut dep_offset,
254            displayed: t,
255        }),
256        Some(TravelMode::For(d)) => ui.add(SeparateDisplayDragValue {
257            modified: &mut dep_offset,
258            displayed: d,
259        }),
260        Some(TravelMode::Flexible) => ui.label(fmt_time_label(estimate.arr)),
261        None => ui.label(RichText::from("⇂").weak()),
262    };
263    arr_or_pass_resp
264        .dragged()
265        .then_some(Either::Left(arr_or_pass_offset))
266        .or(dep_resp.dragged().then_some(Either::Right(dep_offset)))
267        .map(|offset| (entry.id, offset))
268}

270fn show_cursor_to_insert_at_idx(ui: &mut Ui, idx: usize) -> Option<NodeKey> {
271    let left = ui.min_rect().left_bottom();
272    let right = ui.min_rect().right_bottom() + vec2(10.0, 0.0);
273    let rect = Rect::from_pos(right).expand(8.0);
274    let resp = ui
275        .interact(
276            rect,
277            ui.scope_id().with(("insert entry", idx)),
278            Sense::click(),
279        )
280        .on_hover_text("Insert entry to timetable at this position");
281    let mut stroke = Stroke::new(2.0, Color32::TRANSPARENT);
282    if resp.contains_pointer() {
283        stroke.color = Color32::RED;
284    }
285    if resp.is_pointer_button_down_on() {
286        stroke.color = Color32::DARK_RED;
287    }
288    {
289        let stroke = if stroke.color == Color32::TRANSPARENT {
290            Stroke::new(1.0, ui.visuals().strong_text_color())
291        } else {
292            stroke
293        };
294        let rect = rect.shrink(4.0);
295        ui.painter().line_segment([rect.center_top(), rect.center_bottom()], stroke);
296        ui.painter().line_segment([rect.right_center(), rect.left_center()], stroke);
297    }
298    ui.painter().line_segment([left, rect.left_center()], stroke);
299    None
